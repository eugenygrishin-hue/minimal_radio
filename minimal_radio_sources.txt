#pragma once
#include "driver/i2c_master.h"
#include "esp_log.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include <cstring>

#define HMI_ADDR             0x41
#define ENCODER_REG          0x00
#define INCREMENT_REG        0x10
#define BUTTON_REG           0x20
#define LED_REG              0x30
#define RESET_COUNTER_REG    0x40
#define FIRMWARE_VERSION_REG 0xFE
#define I2C_ADDRESS_REG      0xFF

class M5ModuleHMI_ESPIDF {
public:
    bool begin(i2c_port_t port_num, int sda_pin, int scl_pin, uint32_t speed = 100000) {
        _port = port_num;
        i2c_master_bus_config_t bus_cfg = {};
        bus_cfg.i2c_port = port_num;
        bus_cfg.clk_source = I2C_CLK_SRC_DEFAULT;
        bus_cfg.sda_io_num = (gpio_num_t)sda_pin;
        bus_cfg.scl_io_num = (gpio_num_t)scl_pin;
        bus_cfg.glitch_ignore_cnt = 7;
        bus_cfg.flags.enable_internal_pullup = true;

        esp_err_t err = i2c_new_master_bus(&bus_cfg, &_bus_handle);
        if (err != ESP_OK && err != ESP_ERR_INVALID_STATE) {
            ESP_LOGE("HMI", "i2c_new_master_bus failed: %s", esp_err_to_name(err));
            return false;
        }

        err = i2c_master_probe(_bus_handle, HMI_ADDR, 1000 / portTICK_PERIOD_MS);
        if (err != ESP_OK) {
            ESP_LOGE("HMI", "Device 0x%02X not found", HMI_ADDR);
            if (_bus_handle) i2c_del_master_bus(_bus_handle);
            _bus_handle = nullptr;
            return false;
        }

        i2c_device_config_t dev_cfg = {};
        dev_cfg.dev_addr_length = I2C_ADDR_BIT_LEN_7;
        dev_cfg.device_address = HMI_ADDR;
        dev_cfg.scl_speed_hz = speed;

        err = i2c_master_bus_add_device(_bus_handle, &dev_cfg, &_dev_handle);
        if (err != ESP_OK) {
            ESP_LOGE("HMI", "i2c_master_bus_add_device failed: %s", esp_err_to_name(err));
            if (_bus_handle) i2c_del_master_bus(_bus_handle);
            _bus_handle = nullptr;
            return false;
        }

        ESP_LOGI("HMI", "HMI Module initialized on pins SDA:%d, SCL:%d", sda_pin, scl_pin);
        return true;
    }

    int32_t getEncoderValue() {
        uint8_t buf[4] = {0};
        if (!readRegister(ENCODER_REG, buf, 4)) return 0;
        return (int32_t)(buf[0] | (buf[1] << 8) | (buf[2] << 16) | (buf[3] << 24));
    }

    int32_t getIncrementValue() {
        uint8_t buf[4] = {0};
        if (!readRegister(INCREMENT_REG, buf, 4)) return 0;
        return (int32_t)(buf[0] | (buf[1] << 8) | (buf[2] << 16) | (buf[3] << 24));
    }

    bool getButtonS() {
        uint8_t val = 0;
        if (!readRegister(BUTTON_REG, &val, 1)) return false;
        return val == 0;
    }

    bool getButton1() {
        uint8_t val = 0;
        if (!readRegister(BUTTON_REG + 1, &val, 1)) return false;
        return val == 0;
    }

    bool getButton2() {
        uint8_t val = 0;
        if (!readRegister(BUTTON_REG + 2, &val, 1)) return false;
        return val == 0;
    }

    void resetCounter() {
        uint8_t dummy = 0;
        writeRegister(RESET_COUNTER_REG, &dummy, 1);
    }

    void setLEDStatus(uint8_t index, uint8_t status) {
        uint8_t buf[1] = {status};
        // Исправлено: регистры светодиодов начинаются с 0x30
        uint8_t reg = LED_REG + index - 1;  // 0x30 + 1 - 1 = 0x30, 0x30 + 2 - 1 = 0x31
        writeRegister(reg, buf, 1);
    }

private:
    i2c_master_bus_handle_t _bus_handle = nullptr;
    i2c_master_dev_handle_t _dev_handle = nullptr;
    i2c_port_t _port;

    bool readRegister(uint8_t reg, uint8_t* buffer, size_t length) {
        uint8_t write_buf[1] = {reg};
        esp_err_t err = i2c_master_transmit_receive(
            _dev_handle, write_buf, 1, buffer, length, 1000 / portTICK_PERIOD_MS
        );
        return (err == ESP_OK);
    }

    bool writeRegister(uint8_t reg, const uint8_t* buffer, size_t length) {
        uint8_t* write_buf = (uint8_t*)malloc(length + 1);
        if (!write_buf) return false;
        write_buf[0] = reg;
        if (buffer) memcpy(write_buf + 1, buffer, length);

        esp_err_t err = i2c_master_transmit(_dev_handle, write_buf, length + 1, 1000 / portTICK_PERIOD_MS);
        free(write_buf);
        return (err == ESP_OK);
    }
};

static i2s_chan_handle_t tx_chan = NULL;
static RingbufHandle_t s_ringbuf = NULL;

static uint32_t g_rb_in = 0, g_rb_out = 0, g_rb_overflow = 0, g_rb_starve = 0, g_dec_buf = 0;
static volatile bool g_playback_started = false, g_prefill_active = false;
static int g_current_volume = 75;
static bool g_is_muted = false;

// Новые переменные для отслеживания состояния I2S
static uint32_t g_current_sample_rate = 0;
static bool g_i2s_channel_created = false;
static bool g_i2s_enabled = false;

/**
 * @brief Создаёт и инициализирует I2S канал (вызывается только один раз)
 */
static esp_err_t i2s_ensure_channel(void) {
    if (g_i2s_channel_created) {
        return ESP_OK;
    }

    i2s_chan_config_t chan_cfg = I2S_CHANNEL_DEFAULT_CONFIG(I2S_NUM_0, I2S_ROLE_MASTER);
    chan_cfg.auto_clear = true;
    chan_cfg.dma_desc_num = 24;
    chan_cfg.dma_frame_num = 1023;

    esp_err_t err = i2s_new_channel(&chan_cfg, &tx_chan, NULL);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Failed to create I2S channel: %s", esp_err_to_name(err));
        return err;
    }

    i2s_std_config_t std_cfg = {};
    // Временная частота 44100, будет перезаписана при первом фрейме
    std_cfg.clk_cfg = I2S_STD_CLK_DEFAULT_CONFIG(44100);
    std_cfg.clk_cfg.mclk_multiple = I2S_MCLK_MULTIPLE_256;
    std_cfg.slot_cfg = I2S_STD_PHILIP_SLOT_DEFAULT_CONFIG(I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_STEREO);

    i2s_std_gpio_config_t gpio_cfg = {};
    gpio_cfg.mclk = I2S_MCLK_PIN;
    gpio_cfg.bclk = I2S_BCK_PIN;
    gpio_cfg.ws = I2S_WS_PIN;
    gpio_cfg.dout = I2S_DOUT_PIN;
    gpio_cfg.din = I2S_DIN_PIN;
    std_cfg.gpio_cfg = gpio_cfg;

    err = i2s_channel_init_std_mode(tx_chan, &std_cfg);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Failed to init I2S std mode: %s", esp_err_to_name(err));
        return err;
    }

    g_i2s_channel_created = true;
    ESP_LOGI(TAG, "I2S channel created successfully");
    return ESP_OK;
}

/**
* @brief Динамически обновляет частоту дискретизации I2S на лету
*/
static esp_err_t i2s_update_sample_rate(uint32_t sample_rate) {
    if (g_current_sample_rate == sample_rate) {
        return ESP_OK; // Частота не изменилась, ничего не делаем
    }
    if (!g_i2s_channel_created) {
        return ESP_ERR_INVALID_STATE;
    }

    // 1. Сначала выключаем канал перед перенастройкой
    i2s_channel_disable(tx_chan);

    // 2. Перенастраиваем тактирование
    i2s_std_clk_config_t clk_cfg = I2S_STD_CLK_DEFAULT_CONFIG(sample_rate);
    clk_cfg.mclk_multiple = I2S_MCLK_MULTIPLE_256;

    esp_err_t err = i2s_channel_reconfig_std_clock(tx_chan, &clk_cfg);

    if (err == ESP_OK) {
        ESP_LOGI(TAG, ">>> I2S sample rate updated to %lu Hz", (unsigned long)sample_rate);
        g_current_sample_rate = sample_rate;

        // 3. КРИТИЧЕСКИ ВАЖНО: Включаем канал обратно после успешной перенастройки!
        i2s_channel_enable(tx_chan);
        g_i2s_enabled = true; // Гарантируем, что флаг установлен
    } else {
        ESP_LOGE(TAG, "Failed to reconfig I2S clock to %lu Hz: %s", (unsigned long)sample_rate, esp_err_to_name(err));
        // На всякий случай включаем канал обратно, чтобы устройство не зависло в тишине при ошибке
        i2s_channel_enable(tx_chan);
    }

    return err;
}

static void es8388_init() {
    // Пробуждение и базовая инициализация ES8388
    M5.In_I2C.writeRegister8(ES8388_ADDR, 0x1A, 0xC0, 100000);
    M5.In_I2C.writeRegister8(ES8388_ADDR, 0x1B, 0xC0, 100000);
    vTaskDelay(pdMS_TO_TICKS(50));

    uint8_t regs[] = {0x08,0x00, 0x02,0xFF, 0x2B,0x80, 0x00,0x05, 0x01,0x40, 0x03,0x00, 0x0A,0x00, 0x0B,0x00, 0x09,0x88, 0x0C,0x2C, 0x0D,0x02, 0x0F,0x28, 0x10,0x00, 0x11,0x00, 0x12,0xEA, 0x13,0xC0, 0x14,0x12, 0x15,0x06, 0x16,0xC3, 0x04,0x3F, 0x17,0x18, 0x18,0x02, 0x19,0x00, 0x1A,0x1E, 0x1B,0x1E, 0x26,0x00, 0x27,0x90, 0x28,0x38, 0x29,0x38, 0x2A,0x90, 0x2B,0x80, 0x2E,0x1E, 0x2F,0x1E, 0x30,0x00, 0x31,0x00, 0x02,0x00};
    for (int i = 0; i < sizeof(regs); i += 2) {
        M5.In_I2C.writeRegister8(ES8388_ADDR, regs[i], regs[i+1], 100000);
    }

    // Конфигурация STM32 для аудиомодуля M144 (CoreS3 SE)
    M5.In_I2C.writeRegister8(STM32_ADDR, 0x10, 0x00, 100000);

    audio_set_volume(g_current_volume);
}

static void decode_play_task(void *pvParameters) {
    HMP3Decoder mp3_decoder = MP3InitDecoder();
    uint8_t *inbuf = (uint8_t *)heap_caps_malloc(DEC_BUF_SIZE, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    short *outbuf = (short *)heap_caps_malloc(2304 * sizeof(short), MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);

    if (!inbuf || !outbuf) {
        ESP_LOGE(TAG, "Failed to allocate audio buffers");
        vTaskDelete(NULL);
        return;
    }

    int bytes_in_buffer = 0;

    auto do_prefill = [&]() {
        g_prefill_active = true;
        while (bytes_in_buffer < PREFILL_TARGET) {
            size_t s;
            void *d = xRingbufferReceive(s_ringbuf, &s, pdMS_TO_TICKS(2000));
            if (d) {
                if (bytes_in_buffer + s <= DEC_BUF_SIZE) {
                    memcpy(inbuf + bytes_in_buffer, d, s);
                    bytes_in_buffer += s;
                    g_rb_out += s;
                }
                vRingbufferReturnItem(s_ringbuf, d);
            } else {
                vTaskDelay(pdMS_TO_TICKS(10));
            }
        }
        g_prefill_active = false;
    };

    do_prefill();
    g_playback_started = true;

    while (1) {
        // 1. Обработка андеррана (опустошения буфера)
        if (bytes_in_buffer < REBUF_THRESHOLD) {
            ESP_LOGW(TAG, "Buffer underrun! Rebuffering...");
            g_playback_started = false;

            size_t s;
            void *d;
            while ((d = xRingbufferReceive(s_ringbuf, &s, 0)) != NULL) {
                vRingbufferReturnItem(s_ringbuf, d);
            }

            bytes_in_buffer = 0;
            memset(inbuf, 0, DEC_BUF_SIZE);
            g_rb_in = 0;
            g_rb_out = 0;

            do_prefill();
            g_playback_started = true;
            continue;
        }

        // 2. Дозаполнение буфера до целевого уровня
        while (bytes_in_buffer < PREFILL_TARGET) {
            size_t s;
            void *d = xRingbufferReceive(s_ringbuf, &s, (bytes_in_buffer < 4096) ? pdMS_TO_TICKS(500) : pdMS_TO_TICKS(20));
            if (d) {
                if (bytes_in_buffer + s <= DEC_BUF_SIZE) {
                    memcpy(inbuf + bytes_in_buffer, d, s);
                    bytes_in_buffer += s;
                    g_rb_out += s;
                }
                vRingbufferReturnItem(s_ringbuf, d);
            } else {
                break;
            }
        }

        // 3. Защита от тишины при низком буфере
        if (bytes_in_buffer < 8192 && g_i2s_enabled) {
            short sil[1024] = {0};
            size_t w;
            i2s_channel_write(tx_chan, sil, sizeof(sil), &w, pdMS_TO_TICKS(20));
            vTaskDelay(pdMS_TO_TICKS(5));
            continue;
        }

        g_dec_buf = bytes_in_buffer;
        uint8_t *rp = inbuf;
        int bl = bytes_in_buffer, fp = 0;

        // 4. Декодирование фреймов
        while (bl > 0 && fp < FRAMES_PER_PASS) {
            int off = MP3FindSyncWord(rp, bl);
            if (off < 0) {
                rp++;
                bl--;
                continue;
            }
            if (off > 0) {
                rp += off;
                bl -= off;
            }

            uint8_t *sp = rp;
            int sl = bl;
            int ret = MP3Decode(mp3_decoder, &rp, &bl, outbuf, 0);

            if (ret == ERR_MP3_INDATA_UNDERFLOW) break;
            if (ret != ERR_MP3_NONE) {
                rp = sp + 1;
                bl = sl - 1;
                continue;
            }

            MP3FrameInfo fi;
            MP3GetLastFrameInfo(mp3_decoder, &fi);
            if (fi.samprate == 0 || fi.nChans == 0) {
                rp = sp + 1;
                bl = sl - 1;
                continue;
            }

            // 5. Инициализация I2S при первом успешном фрейме
            if (!g_i2s_channel_created) {
                if (i2s_ensure_channel() != ESP_OK) {
                    vTaskDelay(pdMS_TO_TICKS(100));
                    continue;
                }
            }

            // 6. КЛЮЧЕВОЕ ИСПРАВЛЕНИЕ: Динамическое обновление частоты дискретизации
            if (g_current_sample_rate != (uint32_t)fi.samprate) {
                i2s_update_sample_rate(fi.samprate);
            }

            // 7. Включение канала (один раз)
            if (!g_i2s_enabled) {
                i2s_channel_enable(tx_chan);
                g_i2s_enabled = true;
            }

            // 8. Запись в I2S (с конвертацией моно в стерео при необходимости)
            size_t w;
            if (fi.nChans == 1) {
                short st[2304];
                for(int i = 0; i < fi.outputSamps; i++) {
                    st[i*2] = outbuf[i];
                    st[i*2+1] = outbuf[i];
                }
                i2s_channel_write(tx_chan, st, fi.outputSamps * 4, &w, portMAX_DELAY);
            } else {
                i2s_channel_write(tx_chan, outbuf, fi.outputSamps * 2, &w, portMAX_DELAY);
            }
            fp++;
        }

        // 9. Сдвиг остатков буфера
        if (bl > 0 && bl < bytes_in_buffer) {
            memmove(inbuf, rp, bl);
        } else if (bl == 0) {
            bl = 0;
        }
        bytes_in_buffer = bl;
        vTaskDelay(1);
    }
}

void audio_set_volume(int vol) {
    g_current_volume = vol < 0 ? 0 : (vol > 100 ? 100 : vol);
    uint8_t dac_vol = (uint8_t)(0xC0 - (g_current_volume * 0xC0) / 100);
    if (!g_is_muted) {
        M5.In_I2C.writeRegister8(ES8388_ADDR, 0x1A, dac_vol, 100000);
        M5.In_I2C.writeRegister8(ES8388_ADDR, 0x1B, dac_vol, 100000);
    }
}

void audio_toggle_mute() {
    g_is_muted = !g_is_muted;
    if (g_is_muted) {
        M5.In_I2C.writeRegister8(ES8388_ADDR, 0x1A, 0xC0, 100000);
        M5.In_I2C.writeRegister8(ES8388_ADDR, 0x1B, 0xC0, 100000);
    } else {
        audio_set_volume(g_current_volume);
    }
}

static bool s_headphone_connected = false;
static bool s_headphone_output_active = false;

bool audio_is_headphone_connected(void) {
    // Чтение статуса из регистра ES8388 (регистр 0x15)
    // На M5CoreS3 SE бит 7 этого регистра часто отражает статус Jack Detect
    uint8_t status = M5.In_I2C.readRegister8(ES8388_ADDR, 0x15, 100000);
    bool connected = (status & 0x80) != 0;

    // Если по каким-то причинам регистр не работает, можно временно вернуть false для теста
    s_headphone_connected = connected;
    return connected;
}

void audio_set_output_headphone(bool headphone) {
    if (s_headphone_output_active == headphone) return; // Уже в нужном режиме
    s_headphone_output_active = headphone;

    if (headphone) {
        // Включаем выход на наушники (LOUT2/ROUT2), выключаем динамик
        M5.In_I2C.writeRegister8(ES8388_ADDR, 0x28, 0x0C, 100000);
        M5.In_I2C.writeRegister8(ES8388_ADDR, 0x29, 0x00, 100000);
        ESP_LOGI(TAG, "Audio output switched to HEADPHONE");
    } else {
        // Включаем выход на динамик (LOUT1/ROUT1), выключаем наушники
        M5.In_I2C.writeRegister8(ES8388_ADDR, 0x28, 0xC0, 100000);
        M5.In_I2C.writeRegister8(ES8388_ADDR, 0x29, 0x38, 100000);
        ESP_LOGI(TAG, "Audio output switched to SPEAKER");
    }
}

bool audio_is_headphone_output(void) {
    return s_headphone_output_active;
}

static void copy_trunc(char* dst, size_t dst_size, const char* src)
{
    if (dst == nullptr || dst_size == 0) {
        return;
    }

    if (src == nullptr) {
        dst[0] = '\0';
        return;
    }

    size_t len = strlen(src);
    if (len >= dst_size) {
        len = dst_size - 1;
    }

    memcpy(dst, src, len);
    dst[len] = '\0';
}

static char* trim_in_place(char* s)
{
    if (s == nullptr) {
        return s;
    }

    while (*s != '\0' && isspace((unsigned char)*s)) {
        s++;
    }

    size_t len = strlen(s);
    while (len > 0 && isspace((unsigned char)s[len - 1])) {
        s[len - 1] = '\0';
        len--;
    }

    return s;
}

static bool starts_with(const char* s, const char* prefix)
{
    if (s == nullptr || prefix == nullptr) {
        return false;
    }

    size_t lp = strlen(prefix);
    return strncmp(s, prefix, lp) == 0;
}

static bool ends_with_ci(const char* s, const char* suffix)
{
    if (s == nullptr || suffix == nullptr) {
        return false;
    }

    size_t ls = strlen(s);
    size_t lf = strlen(suffix);

    if (ls < lf) {
        return false;
    }

    for (size_t i = 0; i < lf; i++) {
        unsigned char c1 = (unsigned char)s[ls - lf + i];
        unsigned char c2 = (unsigned char)suffix[i];

        if (tolower(c1) != tolower(c2)) {
            return false;
        }
    }

    return true;
}

static __attribute__((unused)) bool join_path(char* out, size_t out_size, const char* dir, const char* name)
{
    if (out == nullptr || out_size == 0 || dir == nullptr || name == nullptr) {
        return false;
    }

    size_t dl = strlen(dir);
    size_t nl = strlen(name);

    bool need_sep = !(dl > 0 && dir[dl - 1] == '/');

    size_t total = dl + (need_sep ? 1 : 0) + nl + 1;

    if (total > out_size) {
        return false;
    }

    memcpy(out, dir, dl);

    size_t pos = dl;

    if (need_sep) {
        out[pos++] = '/';
    }

    memcpy(out + pos, name, nl + 1);

    return true;
}

static uint32_t fnv1a32(const char* s)
{
    uint32_t h = 2166136261u;

    while (s != nullptr && *s != '\0') {
        h ^= (uint8_t)(*s++);
        h *= 16777619u;
    }

    return h;
}

static void normalize_id_input(
    const char* category,
    const char* name,
    char* out,
    size_t out_size
)
{
    if (out == nullptr || out_size == 0) {
        return;
    }

    size_t o = 0;
    bool prev_space = false;

    const char* cat = (category != nullptr) ? category : "";
    const char* nm  = (name != nullptr) ? name : "";

    for (int part = 0; part < 2; part++) {
        const char* src = (part == 0) ? cat : nm;

        if (part == 1) {
            if (o + 1 < out_size) {
                out[o++] = '|';
            }
            prev_space = false;
        } else {
            prev_space = false;
        }

        for (const unsigned char* p = (const unsigned char*)src; *p != '\0'; p++) {
            unsigned char c = *p;

            if (isspace(c)) {
                if (!prev_space && o > 0 && out[o - 1] != ' ' && out[o - 1] != '|') {
                    if (o + 1 < out_size) {
                        out[o++] = ' ';
                    }
                }
                prev_space = true;
            } else {
                prev_space = false;

                if (c >= 'A' && c <= 'Z') {
                    c = (unsigned char)(c - 'A' + 'a');
                }

                if (o + 1 < out_size) {
                    out[o++] = (char)c;
                }
            }
        }
    }

    while (o > 0 && out[o - 1] == ' ') {
        o--;
    }

    if (o < out_size) {
        out[o] = '\0';
    } else {
        out[out_size - 1] = '\0';
    }
}

uint32_t catalog_make_station_id(const char* category, const char* name)
{
    char buf[CATALOG_MAX_CATEGORY_LEN + CATALOG_MAX_NAME_LEN + 4];

    const char* cat = (category != nullptr && category[0] != '\0') ? category : "general";
    const char* nm  = (name != nullptr && name[0] != '\0') ? name : "station";

    normalize_id_input(cat, nm, buf, sizeof(buf));

    uint32_t id = fnv1a32(buf);
    if (id == 0) {
        id = 1;
    }

    return id;
}

static uint16_t parse_bitrate_from_name(const char* name)
{
    if (name == nullptr) {
        return 0;
    }

    const char* open = strrchr(name, '(');
    if (open == nullptr) {
        return 0;
    }

    const char* close = strchr(open, ')');
    const char* end = (close != nullptr) ? close : (open + strlen(open));

    const char* p = open + 1;

    while (p < end && !isdigit((unsigned char)*p)) {
        p++;
    }

    if (p >= end) {
        return 0;
    }

    uint32_t val = 0;

    while (p < end && isdigit((unsigned char)*p)) {
        val = val * 10 + (uint32_t)(*p - '0');
        if (val > 65535u) {
            val = 65535u;
            break;
        }
        p++;
    }

    return (uint16_t)val;
}

static __attribute__((unused)) void category_from_path(const char* path, char* out, size_t out_size)
{
    if (out == nullptr || out_size == 0) {
        return;
    }

    if (path == nullptr) {
        copy_trunc(out, out_size, "General");
        return;
    }

    const char* base = strrchr(path, '/');
    base = (base != nullptr) ? (base + 1) : path;

    char tmp[256];
    char work[256];

    copy_trunc(tmp, sizeof(tmp), base);
    copy_trunc(work, sizeof(work), tmp);

    if (ends_with_ci(work, ".m3u")) {
        size_t len = strlen(work);
        if (len >= 4) {
            work[len - 4] = '\0';
        }
    }

    char* par = strchr(work, '(');
    if (par != nullptr) {
        char* close = strchr(par, ')');
        if (close != nullptr) {
            *par = '\0';
        }
    }

    trim_in_place(work);

    if (work[0] == '\0') {
        copy_trunc(work, sizeof(work), tmp);
    }

    copy_trunc(out, out_size, work);
}

static bool ensure_category_capacity(size_t need)
{
    if (need <= g_catalog.category_capacity) {
        return true;
    }

    size_t new_cap = g_catalog.category_capacity ? g_catalog.category_capacity * 2 : 8;
    while (new_cap < need) {
        new_cap *= 2;
    }

    catalog_category_t* p = (catalog_category_t*)heap_caps_realloc(
        g_catalog.categories,
        new_cap * sizeof(catalog_category_t),
        MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT
    );

    if (p == nullptr) {
        ESP_LOGE(TAG, "OOM: cannot realloc categories to %u", (unsigned)new_cap);
        return false;
    }

    g_catalog.categories = p;
    g_catalog.category_capacity = new_cap;
    return true;
}

static bool ensure_station_capacity_in_category(catalog_category_t* cat, size_t need)
{
    if (cat == nullptr) {
        return false;
    }

    if (need <= cat->capacity) {
        return true;
    }

    size_t new_cap = cat->capacity ? cat->capacity * 2 : 4;
    while (new_cap < need) {
        new_cap *= 2;
    }

    catalog_station_t* p = (catalog_station_t*)heap_caps_realloc(
        cat->stations,
        new_cap * sizeof(catalog_station_t),
        MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT
    );

    if (p == nullptr) {
        ESP_LOGE(TAG, "OOM: cannot realloc stations for category '%s' to %u",
                 cat->name, (unsigned)new_cap);
        return false;
    }

    cat->stations = p;
    cat->capacity = new_cap;
    return true;
}

static catalog_category_t* find_or_create_category(const char* name)
{
    char clean[CATALOG_MAX_CATEGORY_LEN + 1];

    copy_trunc(clean, sizeof(clean), (name != nullptr && name[0] != '\0') ? name : "General");
    trim_in_place(clean);

    if (clean[0] == '\0') {
        copy_trunc(clean, sizeof(clean), "General");
    }

    for (size_t i = 0; i < g_catalog.category_count; i++) {
        if (strcmp(g_catalog.categories[i].name, clean) == 0) {
            return &g_catalog.categories[i];
        }
    }

    if (!ensure_category_capacity(g_catalog.category_count + 1)) {
        return nullptr;
    }

    catalog_category_t* cat = &g_catalog.categories[g_catalog.category_count];
    memset(cat, 0, sizeof(*cat));

    copy_trunc(cat->name, sizeof(cat->name), clean);

    g_catalog.category_count++;
    return cat;
}

static bool add_station_to_category(
    catalog_category_t* cat,
    const char* name,
    const char* url,
    uint16_t bitrate,
    uint32_t id
)
{
    if (cat == nullptr || name == nullptr || url == nullptr) {
        return false;
    }

    if (name[0] == '\0' || url[0] == '\0') {
        return false;
    }

    if (strlen(name) > CATALOG_MAX_NAME_LEN || strlen(url) > CATALOG_MAX_URL_LEN) {
        ESP_LOGW(TAG, "Station too long, skipped: %s / %s", name, url);
        return false;
    }

    for (size_t i = 0; i < cat->count; i++) {
        if (cat->stations[i].id == id) {
            return true; // уже есть, не дублируем
        }
    }

    if (!ensure_station_capacity_in_category(cat, cat->count + 1)) {
        return false;
    }

    catalog_station_t* st = &cat->stations[cat->count++];
    memset(st, 0, sizeof(*st));

    st->id = id;
    copy_trunc(st->name, sizeof(st->name), name);
    copy_trunc(st->url, sizeof(st->url), url);
    st->bitrate_kbps = bitrate;

    g_catalog.total_stations++;

    return true;
}

bool catalog_add_station(
    const char* category,
    const char* name,
    const char* url,
    uint16_t bitrate_kbps
)
{
    catalog_category_t* cat = find_or_create_category(category);
    if (cat == nullptr) {
        return false;
    }

    if (bitrate_kbps == 0) {
        bitrate_kbps = parse_bitrate_from_name(name);
    }

    uint32_t id = catalog_make_station_id(cat->name, name);
    return add_station_to_category(cat, name, url, bitrate_kbps, id);
}

void catalog_clear(void)
{
    for (size_t i = 0; i < g_catalog.category_count; i++) {
        if (g_catalog.categories[i].stations != nullptr) {
            heap_caps_free(g_catalog.categories[i].stations);
            g_catalog.categories[i].stations = nullptr;
        }
    }

    if (g_catalog.categories != nullptr) {
        heap_caps_free(g_catalog.categories);
        g_catalog.categories = nullptr;
    }

    memset(&g_catalog, 0, sizeof(g_catalog));
    g_catalog.version = 1;
}

static bool catalog_load_embedded(void)
{
    catalog_clear();

    if (g_category_count <= 0) {
        ESP_LOGW(TAG, "Embedded catalog is empty");
        return false;
    }

    for (int c = 0; c < g_category_count; c++) {
        const radio_category_t* rc = &g_categories[c];

        catalog_category_t* cat = find_or_create_category(rc->category);
        if (cat == nullptr) {
            return false;
        }

        for (int s = 0; s < rc->count; s++) {
            const char* name = rc->stations[s].name;
            const char* url  = rc->stations[s].url;

            if (name == nullptr || url == nullptr) {
                continue;
            }

            uint16_t br = parse_bitrate_from_name(name);
            uint32_t id = catalog_make_station_id(cat->name, name);

            if (!add_station_to_category(cat, name, url, br, id)) {
                ESP_LOGW(TAG, "Failed to add embedded station: %s", name);
            }
        }
    }

    g_catalog.loaded = (g_catalog.total_stations > 0);
    g_catalog.from_embedded = g_catalog.loaded;
    g_catalog.from_sd = false;

    copy_trunc(g_catalog.updated_at, sizeof(g_catalog.updated_at), "embedded");

    return g_catalog.loaded;
}

bool catalog_parse_m3u_text(const char* m3u_text, const char* category_name)
{
    if (m3u_text == nullptr) {
        return false;
    }

    catalog_category_t* cat = find_or_create_category(
        (category_name != nullptr && category_name[0] != '\0') ? category_name : "General"
    );

    if (cat == nullptr) {
        return false;
    }

    size_t len = strlen(m3u_text);
    char* buf = (char*)malloc(len + 1);
    if (buf == nullptr) {
        ESP_LOGE(TAG, "OOM for M3U buffer, len=%u", (unsigned)len);
        return false;
    }

    memcpy(buf, m3u_text, len + 1);

    char pending_name[CATALOG_MAX_NAME_LEN + 1];
    pending_name[0] = '\0';
    bool have_pending = false;

    char* saveptr = nullptr;

    for (
        char* line = strtok_r(buf, "\r\n", &saveptr);
        line != nullptr;
        line = strtok_r(nullptr, "\r\n", &saveptr)
    ) {
        char* t = trim_in_place(line);

        if (*t == '\0') {
            continue;
        }

        if (
            starts_with(t, "#CURTRACK") ||
            starts_with(t, "#EXTM3U") ||
            starts_with(t, "#EXTURL:")
        ) {
            continue;
        }

        if (starts_with(t, "#EXTINF:")) {
            char* p = t + strlen("#EXTINF:");
            p = trim_in_place(p);

            // Поддержка классического формата:
            // #EXTINF:-1,Name
            // #EXTINF:0,Name
            if (*p == '-' || isdigit((unsigned char)*p)) {
                char* comma = strchr(p, ',');
                if (comma != nullptr) {
                    p = comma + 1;
                    p = trim_in_place(p);
                }
            }

            copy_trunc(pending_name, sizeof(pending_name), p);
            have_pending = true;
            continue;
        }

        if (*t == '#') {
            continue;
        }

        // Обычная строка — URL.
        const char* url = t;

        if (have_pending) {
            uint16_t br = parse_bitrate_from_name(pending_name);
            uint32_t id = catalog_make_station_id(cat->name, pending_name);

            add_station_to_category(cat, pending_name, url, br, id);

            have_pending = false;
            pending_name[0] = '\0';
        } else {
            // Если вдруг URL без EXTINF — добавим как есть, имя = URL.
            uint32_t id = catalog_make_station_id(cat->name, url);
            add_station_to_category(cat, url, url, 0, id);
        }
    }

    free(buf);
    return true;
}

#if defined(__GNUC__)
#pragma GCC diagnostic push
#pragma GCC diagnostic ignored "-Wformat-truncation"
#endif

bool catalog_import_m3u_file(const char* path, const char* category_name)
{
    if (!sd_manager_is_mounted()) {
        ESP_LOGW(TAG, "SD not mounted, cannot import M3U: %s", path ? path : "(null)");
        return false;
    }

    if (path == nullptr) {
        return false;
    }

    char cat_buf[CATALOG_MAX_CATEGORY_LEN + 1];
    const char* cat = category_name;

    if (cat == nullptr || cat[0] == '\0') {
        category_from_path(path, cat_buf, sizeof(cat_buf));
        cat = cat_buf;
    }

    size_t len = 0;
    char* text = sd_manager_read_text(path, &len);
    if (text == nullptr) {
        ESP_LOGW(TAG, "Failed to read M3U file: %s", path);
        return false;
    }

    bool ok = catalog_parse_m3u_text(text, cat);
    free(text);

    if (ok) {
        ESP_LOGI(TAG, "Imported M3U: %s -> category '%s'", path, cat);
    }

    return ok;
}

bool catalog_import_m3u_dir(const char* dir_path)
{
    if (!sd_manager_is_mounted()) {
        ESP_LOGW(TAG, "SD not mounted, cannot import M3U dir: %s", dir_path ? dir_path : "(null)");
        return false;
    }

    if (dir_path == nullptr) {
        return false;
    }

    DIR* d = opendir(dir_path);
    if (d == nullptr) {
        ESP_LOGW(TAG, "Cannot open M3U dir: %s", dir_path);
        return false;
    }

    size_t imported = 0;
    struct dirent* ent;

    while ((ent = readdir(d)) != nullptr) {
    	if (ent->d_name[0] == '.') {
    	    continue;
    	}

        if (!ends_with_ci(ent->d_name, ".m3u")) {
            continue;
        }

        char path[512];

        if (!join_path(path, sizeof(path), dir_path, ent->d_name)) {
            ESP_LOGW(TAG, "M3U path too long, skipped");
            continue;
        }

        if (catalog_import_m3u_file(path, nullptr)) {
            imported++;
        }
    }

    closedir(d);

    ESP_LOGI(TAG, "Imported %u M3U files from %s", (unsigned)imported, dir_path);

    return imported > 0;
}

#if defined(__GNUC__)
#pragma GCC diagnostic pop
#endif

bool catalog_rebuild_from_m3u_dir(const char* dir_path)
{
    catalog_clear();

    g_catalog.version = 1;
    copy_trunc(g_catalog.updated_at, sizeof(g_catalog.updated_at), "m3u-dir");

    bool imported_any = catalog_import_m3u_dir(dir_path);

    g_catalog.loaded = imported_any && (g_catalog.total_stations > 0);
    g_catalog.from_sd = g_catalog.loaded;
    g_catalog.from_embedded = false;

    if (g_catalog.loaded && sd_manager_is_mounted()) {
        catalog_save_json_file(CATALOG_JSON_PATH);
    }

    return g_catalog.loaded;
}

static uint32_t json_get_uint(const cJSON* obj, const char* key, uint32_t def)
{
    if (obj == nullptr || key == nullptr) {
        return def;
    }

    cJSON* item = cJSON_GetObjectItemCaseSensitive(obj, key);
    if (cJSON_IsNumber(item)) {
        double d = item->valuedouble;
        if (d < 0.0) {
            d = 0.0;
        }
        if (d > 4294967295.0) {
            d = 4294967295.0;
        }
        return (uint32_t)d;
    }

    return def;
}

bool catalog_load_json_file(const char* path)
{
    if (!sd_manager_is_mounted()) {
        return false;
    }

    size_t len = 0;
    char* text = sd_manager_read_text(path, &len);
    if (text == nullptr) {
        return false;
    }

    cJSON* root = cJSON_Parse(text);
    free(text);

    if (root == nullptr) {
        ESP_LOGE(TAG, "Failed to parse JSON: %s", path);
        return false;
    }

    catalog_clear();

    g_catalog.version = (int)json_get_uint(root, "version", 1);

    cJSON* updated = cJSON_GetObjectItemCaseSensitive(root, "updated_at");
    if (cJSON_IsString(updated) && updated->valuestring != nullptr) {
        copy_trunc(g_catalog.updated_at, sizeof(g_catalog.updated_at), updated->valuestring);
    }

    cJSON* cats = cJSON_GetObjectItemCaseSensitive(root, "categories");
    if (!cJSON_IsArray(cats)) {
        ESP_LOGE(TAG, "Invalid catalog JSON: missing categories array");
        cJSON_Delete(root);
        catalog_clear();
        return false;
    }

    cJSON* cat_item = nullptr;
    cJSON_ArrayForEach(cat_item, cats) {
        cJSON* name_item = cJSON_GetObjectItemCaseSensitive(cat_item, "name");
        if (!cJSON_IsString(name_item) || name_item->valuestring == nullptr) {
            continue;
        }

        catalog_category_t* cat = find_or_create_category(name_item->valuestring);
        if (cat == nullptr) {
            continue;
        }

        cJSON* stations = cJSON_GetObjectItemCaseSensitive(cat_item, "stations");
        if (!cJSON_IsArray(stations)) {
            continue;
        }

        cJSON* st_item = nullptr;
        cJSON_ArrayForEach(st_item, stations) {
            cJSON* nm = cJSON_GetObjectItemCaseSensitive(st_item, "name");
            cJSON* url = cJSON_GetObjectItemCaseSensitive(st_item, "url");

            if (!cJSON_IsString(nm) || nm->valuestring == nullptr) {
                continue;
            }
            if (!cJSON_IsString(url) || url->valuestring == nullptr) {
                continue;
            }

            uint16_t br = (uint16_t)json_get_uint(st_item, "bitrate", 0);
            uint32_t id = json_get_uint(st_item, "id", 0);

            if (id == 0) {
                id = catalog_make_station_id(cat->name, nm->valuestring);
            }

            add_station_to_category(cat, nm->valuestring, url->valuestring, br, id);
        }
    }

    cJSON_Delete(root);

    g_catalog.loaded = (g_catalog.total_stations > 0);
    g_catalog.from_sd = g_catalog.loaded;
    g_catalog.from_embedded = false;

    if (!g_catalog.loaded) {
        catalog_clear();
    }

    return g_catalog.loaded;
}

bool catalog_save_json_file(const char* path)
{
    if (!sd_manager_is_mounted()) {
        return false;
    }

    if (!g_catalog.loaded) {
        return false;
    }

    ensure_sd_radio_dir();

    cJSON* root = cJSON_CreateObject();
    if (root == nullptr) {
        return false;
    }

    cJSON_AddNumberToObject(root, "version", g_catalog.version > 0 ? g_catalog.version : 1);
    cJSON_AddStringToObject(
        root,
        "updated_at",
        g_catalog.updated_at[0] != '\0' ? g_catalog.updated_at : "unknown"
    );

    cJSON* cats = cJSON_CreateArray();
    cJSON_AddItemToObject(root, "categories", cats);

    for (size_t ci = 0; ci < g_catalog.category_count; ci++) {
        const catalog_category_t* cat = &g_catalog.categories[ci];

        cJSON* cobj = cJSON_CreateObject();
        if (cobj == nullptr) {
            continue;
        }

        cJSON_AddStringToObject(cobj, "name", cat->name);

        cJSON* stations = cJSON_CreateArray();
        cJSON_AddItemToObject(cobj, "stations", stations);

        for (size_t si = 0; si < cat->count; si++) {
            const catalog_station_t* st = &cat->stations[si];

            cJSON* sobj = cJSON_CreateObject();
            if (sobj == nullptr) {
                continue;
            }

            cJSON_AddNumberToObject(sobj, "id", (double)st->id);
            cJSON_AddStringToObject(sobj, "name", st->name);
            cJSON_AddStringToObject(sobj, "url", st->url);
            cJSON_AddNumberToObject(sobj, "bitrate", st->bitrate_kbps);

            cJSON_AddItemToArray(stations, sobj);
        }

        cJSON_AddItemToArray(cats, cobj);
    }

    char* out = cJSON_PrintUnformatted(root);
    cJSON_Delete(root);

    if (out == nullptr) {
        ESP_LOGE(TAG, "Failed to serialize catalog JSON");
        return false;
    }

    bool ok = sd_manager_write_text(path, out);
    cJSON_free(out);

    if (ok) {
        ESP_LOGI(TAG, "Catalog saved: %s", path);
    } else {
        ESP_LOGW(TAG, "Failed to save catalog: %s", path);
    }

    return ok;
}

size_t catalog_total_stations(void)
{
    return g_catalog.total_stations;
}

const catalog_station_t* catalog_get_station(size_t category_index, size_t station_index)
{
    const catalog_category_t* cat = catalog_get_category(category_index);
    if (cat == nullptr || station_index >= cat->count) {
        return nullptr;
    }
    return &cat->stations[station_index];
}

const catalog_station_t* catalog_get_station_by_global_index(size_t global_index)
{
    size_t offset = 0;
    for (size_t ci = 0; ci < g_catalog.category_count; ci++) {
        const catalog_category_t* cat = &g_catalog.categories[ci];
        if (global_index < offset + cat->count) {
            return &cat->stations[global_index - offset];
        }
        offset += cat->count;
    }
    return nullptr;
}

size_t catalog_global_index(size_t category_index, size_t station_index)
{
    size_t offset = 0;
    for (size_t ci = 0; ci < g_catalog.category_count; ci++) {
        if (ci == category_index) {
            return offset + station_index;
        }
        offset += g_catalog.categories[ci].count;
    }
    return 0;
}

bool catalog_find_station_by_id(
    uint32_t id,
    size_t* out_category_index,
    size_t* out_station_index
)
{
    for (size_t ci = 0; ci < g_catalog.category_count; ci++) {
        const catalog_category_t* cat = &g_catalog.categories[ci];

        for (size_t si = 0; si < cat->count; si++) {
            if (cat->stations[si].id == id) {
                if (out_category_index != nullptr) {
                    *out_category_index = ci;
                }
                if (out_station_index != nullptr) {
                    *out_station_index = si;
                }
                return true;
            }
        }
    }

    return false;
}

bool catalog_init(void)
{
    catalog_clear();

    copy_trunc(g_catalog.updated_at, sizeof(g_catalog.updated_at), "init");

    if (sd_manager_is_mounted()) {
        if (catalog_load_json_file(CATALOG_JSON_PATH)) {
            ESP_LOGI(TAG, "Loaded catalog from SD: %s", CATALOG_JSON_PATH);
            return true;
        }

        ESP_LOGI(TAG, "SD catalog not found or invalid, using embedded fallback");
    }

    if (catalog_load_embedded()) {
        ESP_LOGI(TAG, "Loaded embedded fallback catalog");

        if (sd_manager_is_mounted()) {
            if (catalog_save_json_file(CATALOG_JSON_PATH)) {
                ESP_LOGI(TAG, "Initial catalog cached to SD");
            } else {
                ESP_LOGW(TAG, "Failed to cache initial catalog to SD");
            }
        }

        return true;
    }

    ESP_LOGW(TAG, "No catalog loaded");
    return false;
}

bool catalog_self_test(void)
{
    bool ok = true;

    uint32_t id1 = catalog_make_station_id("ROCK", "Наше Радио (256kbps)");
    uint32_t id2 = catalog_make_station_id("ROCK", "Наше Радио (256kbps)");

    if (id1 == 0 || id1 != id2) {
        ESP_LOGE(TAG, "Self-test failed: station ID is not stable");
        ok = false;
    }

    if (parse_bitrate_from_name("Test (128kbps)") != 128) {
        ESP_LOGE(TAG, "Self-test failed: bitrate parse 128kbps");
        ok = false;
    }

    if (parse_bitrate_from_name("Test (128-320kbps)") != 128) {
        ESP_LOGE(TAG, "Self-test failed: bitrate parse 128-320kbps");
        ok = false;
    }

    if (parse_bitrate_from_name("No bitrate") != 0) {
        ESP_LOGE(TAG, "Self-test failed: bitrate parse no bitrate");
        ok = false;
    }

    if (g_catalog.loaded && g_catalog.category_count > 0 && g_catalog.categories[0].count > 0) {
        uint32_t fid = g_catalog.categories[0].stations[0].id;
        size_t ci = 0;
        size_t si = 0;

        if (!catalog_find_station_by_id(fid, &ci, &si)) {
            ESP_LOGE(TAG, "Self-test failed: cannot find first station by ID");
            ok = false;
        }
    }

    ESP_LOGI(TAG, "Self-test: %s", ok ? "OK" : "FAIL");
    return ok;
}

bool catalog_load_favorites(void)
{
    if (!sd_manager_is_mounted()) return false;
    size_t len = 0;
    char* text = sd_manager_read_text("/sd/radio/favorites.json", &len);
    if (!text) return false;

    cJSON* root = cJSON_Parse(text);
    free(text);
    if (!root) return false;

    g_catalog.favorite_count = 0;
    cJSON* favs = cJSON_GetObjectItemCaseSensitive(root, "favorites");
    if (cJSON_IsArray(favs)) {
        cJSON* item = nullptr;
        cJSON_ArrayForEach(item, favs) {
            if (g_catalog.favorite_count < 64 && cJSON_IsNumber(item)) {
                g_catalog.favorite_ids[g_catalog.favorite_count++] = (uint32_t)item->valuedouble;
            }
        }
    }
    cJSON_Delete(root);
    return true;
}

void catalog_save_favorites(void)
{
    if (!sd_manager_is_mounted()) return;
    ensure_sd_radio_dir();
    cJSON* root = cJSON_CreateObject();
    cJSON* favs = cJSON_CreateArray();
    for (size_t i = 0; i < g_catalog.favorite_count; i++) {
        cJSON_AddItemToArray(favs, cJSON_CreateNumber(g_catalog.favorite_ids[i]));
    }
    cJSON_AddItemToObject(root, "favorites", favs);

    char* out = cJSON_PrintUnformatted(root);
    cJSON_Delete(root);
    if (out) {
        sd_manager_write_text("/sd/radio/favorites.json", out);
        cJSON_free(out);
    }
}

bool catalog_is_favorite(uint32_t station_id)
{
    for (size_t i = 0; i < g_catalog.favorite_count; i++) {
        if (g_catalog.favorite_ids[i] == station_id) return true;
    }
    return false;
}

bool catalog_toggle_favorite(uint32_t station_id)
{
    for (size_t i = 0; i < g_catalog.favorite_count; i++) {
        if (g_catalog.favorite_ids[i] == station_id) {
            // Удаляем
            for (size_t j = i; j < g_catalog.favorite_count - 1; j++) {
                g_catalog.favorite_ids[j] = g_catalog.favorite_ids[j + 1];
            }
            g_catalog.favorite_count--;
            return false; // Было в избранном, теперь удалено
        }
    }
    // Добавляем
    if (g_catalog.favorite_count < 64) {
        g_catalog.favorite_ids[g_catalog.favorite_count++] = station_id;
        return true; // Добавлено
    }
    return false; // Лимит исчерпан
}

size_t catalog_get_favorites_count(void)
{
    return g_catalog.favorite_count;
}

int catalog_get_favorite_global_index(size_t fav_index)
{
    if (fav_index >= g_catalog.favorite_count) return -1;
    uint32_t target_id = g_catalog.favorite_ids[fav_index];

    size_t offset = 0;
    for (size_t ci = 0; ci < g_catalog.category_count; ci++) {
        const catalog_category_t* cat = &g_catalog.categories[ci];
        for (size_t si = 0; si < cat->count; si++) {
            if (cat->stations[si].id == target_id) {
                return (int)(offset + si);
            }
        }
        offset += cat->count;
    }
    return -1; // Станция не найдена в текущем каталоге (возможно, удалена)
}

    // Простая проверка: если файл на SD существует, значит источник SD
	ensure_sd_radio_dir();

}

// ============================================================
// Staging Catalog & Validation (в фоне, чтобы не морозить UI)
// ============================================================
static staging_station_t* s_staging = NULL;
static size_t s_staging_count = 0;
static size_t s_staging_capacity = 0;

// Переменные для сохранения уровня логирования (чтобы заглушить красный спам при валидации)
static esp_log_level_t s_old_tls_level;
static esp_log_level_t s_old_transport_level;
static esp_log_level_t s_old_http_level;

// Переменные для отслеживания прогресса валидации UI-ем
static int s_staging_validated_count = 0;
static int s_staging_total_count = 0;
static bool s_is_validating = false;
static TaskHandle_t s_validation_task_handle = NULL;

// Forward declaration
static void catalog_validation_task(void *pvParameters);

static void staging_ensure_capacity(size_t need) {
    if (need <= s_staging_capacity) return;
    size_t new_cap = s_staging_capacity ? s_staging_capacity * 2 : 256;
    while (new_cap < need) new_cap *= 2;

    staging_station_t* p = (staging_station_t*)heap_caps_realloc(
        s_staging,
        new_cap * sizeof(staging_station_t),
        MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT
    );
    if (p) {
        s_staging = p;
        s_staging_capacity = new_cap;
    } else {
        ESP_LOGE(TAG, "OOM: cannot realloc staging catalog to %d", (int)new_cap);
    }
}

void catalog_staging_clear(void) {
    if (s_staging) {
        heap_caps_free(s_staging);
        s_staging = NULL;
    }
    s_staging_count = 0;
    s_staging_capacity = 0;
    s_staging_validated_count = 0;
    s_staging_total_count = 0;
    s_is_validating = false;
    if (s_validation_task_handle != NULL) {
        vTaskDelete(s_validation_task_handle);
        s_validation_task_handle = NULL;
    }
    ESP_LOGI(TAG, "Staging catalog cleared");
}

int catalog_staging_get_count(void) {
    return (int)s_staging_count;
}

const staging_station_t* catalog_staging_get_station(int index) {
    if (index < 0 || (size_t)index >= s_staging_count) return NULL;
    return &s_staging[index];
}

bool catalog_is_validating(void) {
    return s_is_validating;
}

int catalog_get_validated_count(void) {
    return s_staging_validated_count;
}

int catalog_get_staging_count(void) {
    return (int)s_staging_count;
}

// Функция, которую вызывает UI для запуска валидации в фоне
void catalog_start_validation(void) {
    if (s_staging_count == 0 || s_is_validating) {
        ESP_LOGW(TAG, "Nothing to validate or already validating");
        return;
    }

    // 1. Сохраняем текущие уровни логирования и отключаем спам
    s_old_tls_level = esp_log_level_get("esp-tls");
    s_old_transport_level = esp_log_level_get("transport_base");
    s_old_http_level = esp_log_level_get("HTTP_CLIENT");

    esp_log_level_set("esp-tls", ESP_LOG_NONE);
    esp_log_level_set("transport_base", ESP_LOG_NONE);
    esp_log_level_set("HTTP_CLIENT", ESP_LOG_NONE);

    s_staging_validated_count = 0;
    s_staging_total_count = (int)s_staging_count;
    s_is_validating = true;

    ESP_LOGI(TAG, "Starting background validation of %d staging stations...", s_staging_total_count);

    xTaskCreatePinnedToCore(
        catalog_validation_task,
        "cat_validate",
        4096,
        NULL,
        1, // Низкий приоритет, чтобы не мешать аудио и UI
        &s_validation_task_handle,
        1
    );
}

bool catalog_staging_load_from_m3u_dir(const char* dir_path) {
    if (!sd_manager_is_mounted()) {
        ESP_LOGW(TAG, "SD not mounted, cannot load staging");
        return false;
    }
    catalog_staging_clear(); // Очищаем предыдущий staging

    DIR* d = opendir(dir_path);
    if (!d) {
        ESP_LOGW(TAG, "Cannot open M3U dir: %s", dir_path);
        return false;
    }

    int total_parsed = 0;
    struct dirent* ent;

    while ((ent = readdir(d)) != NULL) {
        if (ent->d_name[0] == '.') continue;

        size_t len = strlen(ent->d_name);
        if (len < 5 || strcasecmp(ent->d_name + len - 4, ".m3u") != 0) continue;

        char path[512];
        snprintf(path, sizeof(path), "%s/%s", dir_path, ent->d_name);

        // Имя файла без расширения как default_group
        char group[CATALOG_MAX_CATEGORY_LEN + 1];
        strncpy(group, ent->d_name, sizeof(group) - 1);
        group[sizeof(group) - 1] = '\0';
        char* dot = strrchr(group, '.');
        if (dot) *dot = '\0';

        // Хак: используем g_catalog как временный буфер для парсинга M3U,
        // чтобы не дублировать код парсера, но сохраняем его состояние.
        catalog_category_t* saved_cats = g_catalog.categories;
        size_t saved_cat_count = g_catalog.category_count;
        size_t saved_cat_cap = g_catalog.category_capacity;
        size_t saved_total = g_catalog.total_stations;
        bool saved_loaded = g_catalog.loaded;

        g_catalog.categories = NULL;
        g_catalog.category_count = 0;
        g_catalog.category_capacity = 0;
        g_catalog.total_stations = 0;
        g_catalog.loaded = false;

        size_t file_len = 0;
        char* text = sd_manager_read_text(path, &file_len);
        bool ok = false;
        if (text) {
            ok = catalog_parse_m3u_text(text, group);
            free(text);
        }

        if (ok) {
            // Копируем всё, что распарсилось, в staging
            for (size_t ci = 0; ci < g_catalog.category_count; ci++) {
                catalog_category_t* cat = &g_catalog.categories[ci];
                for (size_t si = 0; si < cat->count; si++) {
                    catalog_station_t* st = &cat->stations[si];

                    staging_ensure_capacity(s_staging_count + 1);
                    if (s_staging_count < s_staging_capacity) {
                        staging_station_t* ss = &s_staging[s_staging_count++];
                        ss->id = st->id;
                        strncpy(ss->name, st->name, sizeof(ss->name) - 1);
                        strncpy(ss->url, st->url, sizeof(ss->url) - 1);
                        strncpy(ss->group, group, sizeof(ss->group) - 1);
                        ss->bitrate_kbps = st->bitrate_kbps;
                        ss->is_alive = false;
                        ss->last_checked = 0;
                        total_parsed++;
                    }
                }
            }
        }

        // Освобождаем память, которую напарсил M3U в g_catalog
        for (size_t i = 0; i < g_catalog.category_count; i++) {
            if (g_catalog.categories[i].stations) heap_caps_free(g_catalog.categories[i].stations);
        }
        if (g_catalog.categories) heap_caps_free(g_catalog.categories);

        // Восстанавливаем g_catalog
        g_catalog.categories = saved_cats;
        g_catalog.category_count = saved_cat_count;
        g_catalog.category_capacity = saved_cat_cap;
        g_catalog.total_stations = saved_total;
        g_catalog.loaded = saved_loaded;
    }

    closedir(d);
    ESP_LOGI(TAG, "Staging loaded: %d stations from M3U dir", total_parsed);
    return total_parsed > 0;
}

// ============================================================
// Staging Validation & Merge Implementation
// ============================================================
static bool catalog_validate_station_url(const char* url) {
    esp_http_client_config_t config = {};
    config.url = url;
    config.timeout_ms = 1500;
    config.buffer_size = 512;
    config.user_agent = "WinampMPEG/5.0";

    esp_http_client_handle_t client = esp_http_client_init(&config);
    if (!client) return false;

    esp_http_client_set_header(client, "Icy-MetaData", "1");

    if (esp_http_client_open(client, 0) != ESP_OK) {
        esp_http_client_cleanup(client);
        return false;
    }

    esp_http_client_fetch_headers(client);
    int status = esp_http_client_get_status_code(client);

    if (status != 200 && status != 206) {
        esp_http_client_close(client);
        esp_http_client_cleanup(client);
        return false;
    }

    char* ct = nullptr;
    esp_http_client_get_header(client, "Content-Type", &ct);
    if (ct && strcasestr(ct, "text/html")) {
        esp_http_client_close(client);
        esp_http_client_cleanup(client);
        return false;
    }

    uint8_t buf[512];
    int read_len = esp_http_client_read(client, (char*)buf, sizeof(buf));
    esp_http_client_close(client);
    esp_http_client_cleanup(client);

    if (read_len < 32) return false;

    // Проверка магических чисел
    if (buf[0] == 'I' && buf[1] == 'D' && buf[2] == '3') return true;
    if (buf[0] == 'O' && buf[1] == 'g' && buf[2] == 'g' && buf[3] == 'S') return true;
    if (buf[0] == 'f' && buf[1] == 'L' && buf[2] == 'a' && buf[3] == 'C') return true;
    if (buf[0] == '#' && buf[1] == 'E' && buf[2] == 'X' && buf[3] == 'T') return true;

    for (int i = 0; i < read_len - 1 && i < 512; i++) {
        if (buf[i] == 0xFF && (buf[i+1] & 0xE0) == 0xE0) {
            return true;
        }
    }

    return false;
}

int catalog_staging_validate(void) {
    if (s_staging_count == 0) {
        ESP_LOGW(TAG, "Staging catalog is empty, nothing to validate");
        return 0;
    }

    s_is_validating = true;
    s_staging_total_count = (int)s_staging_count;
    s_staging_validated_count = 0;
    uint32_t now = (uint32_t)(esp_timer_get_time() / 1000000);

    ESP_LOGI(TAG, "Starting validation of %d staging stations...", (int)s_staging_count);

    int alive_count = 0;
    for (size_t i = 0; i < s_staging_count; i++) {
        if (!s_staging[i].is_alive) {
            bool alive = catalog_validate_station_url(s_staging[i].url);
            s_staging[i].is_alive = alive;
            s_staging[i].last_checked = now;
            if (alive) alive_count++;
        }
        s_staging_validated_count++;

        // Даем 50 мс на отрисовку UI и работу аудио-декодера
        vTaskDelay(pdMS_TO_TICKS(50));
    }

    s_is_validating = false;
    ESP_LOGI(TAG, "Validation complete: %d alive out of %d checked", alive_count, (int)s_staging_count);
    return (int)s_staging_count;
}

bool catalog_staging_merge_to_working(void) {
    if (s_staging_count == 0) {
        ESP_LOGW(TAG, "Staging catalog is empty, nothing to merge");
        return false;
    }

    int added_count = 0;
    int skipped_count = 0;
    int dead_count = 0;

    ESP_LOGI(TAG, "Merging staging catalog to working catalog...");

    for (size_t i = 0; i < s_staging_count; i++) {
        if (!s_staging[i].is_alive) {
            dead_count++;
            continue;
        }

        // Проверяем, есть ли уже такая станция по ID (дедупликация по URL)
        size_t cat_idx = 0, st_idx = 0;
        if (catalog_find_station_by_id(s_staging[i].id, &cat_idx, &st_idx)) {
            skipped_count++;
            continue; // Уже есть в основном каталоге
        }

        // Добавляем в основной каталог
        if (catalog_add_station(
            s_staging[i].group[0] != '\0' ? s_staging[i].group : "Imported",
            s_staging[i].name,
            s_staging[i].url,
            s_staging[i].bitrate_kbps
        )) {
            added_count++;
        }
    }

    ESP_LOGI(TAG, "Merge complete: Added %d, Skipped (duplicate) %d, Dead %d",
             added_count, skipped_count, dead_count);

    // Сохраняем обновленный основной каталог на SD
    ensure_sd_radio_dir();
    bool ok = catalog_save_json_file(CATALOG_JSON_PATH);

    // Очищаем staging после успешного слияния
    if (ok) {
        catalog_staging_clear();
        ESP_LOGI(TAG, "Staging cleared after successful merge");
    }

    return ok;
}

static void catalog_validation_task(void *pvParameters) {
    uint32_t now = (uint32_t)(esp_timer_get_time() / 1000000);
    int alive_count = 0;

    for (size_t i = 0; i < s_staging_count; i++) {
        if (!s_is_validating) break; // Если очистили каталог во время валидации

        if (!s_staging[i].is_alive) {
            bool alive = catalog_validate_station_url(s_staging[i].url);
            s_staging[i].is_alive = alive;
            s_staging[i].last_checked = now;
            if (alive) alive_count++;
        }
        s_staging_validated_count++;

        // Логируем прогресс каждые 100 станций
        if (s_staging_validated_count % 100 == 0) {
            ESP_LOGI(TAG, "Validation progress: %d/%d", s_staging_validated_count, s_staging_total_count);
        }

        // Даём 50 мс на отрисовку UI и работу аудио-декодера
        vTaskDelay(pdMS_TO_TICKS(50));
    }

    // 2. Восстанавливаем уровни логирования обратно
    esp_log_level_set("esp-tls", s_old_tls_level);
    esp_log_level_set("transport_base", s_old_transport_level);
    esp_log_level_set("HTTP_CLIENT", s_old_http_level);

    ESP_LOGI(TAG, "Validation complete: %d alive out of %d checked", alive_count, s_staging_total_count);
    s_is_validating = false;
    s_validation_task_handle = NULL;
    vTaskDelete(NULL);
}


// ============================================================
// Загрузка текущего рабочего каталога в Staging (для очистки)
// ============================================================
bool catalog_load_working_to_staging(void) {
    if (!g_catalog.loaded || g_catalog.total_stations == 0) {
        ESP_LOGW(TAG, "Working catalog is empty, nothing to load");
        return false;
    }

    catalog_staging_clear(); // Очищаем старый буфер

    for (size_t ci = 0; ci < g_catalog.category_count; ci++) {
        const catalog_category_t* cat = &g_catalog.categories[ci];
        for (size_t si = 0; si < cat->count; si++) {
            const catalog_station_t* st = &cat->stations[si];

            staging_ensure_capacity(s_staging_count + 1);
            if (s_staging_count < s_staging_capacity) {
                staging_station_t* ss = &s_staging[s_staging_count++];
                ss->id = st->id;
                strncpy(ss->name, st->name, sizeof(ss->name) - 1);
                strncpy(ss->url, st->url, sizeof(ss->url) - 1);
                strncpy(ss->group, cat->name, sizeof(ss->group) - 1);
                ss->bitrate_kbps = st->bitrate_kbps;
                ss->is_alive = false; // Сбрасываем, чтобы валидатор проверил заново
                ss->last_checked = 0;
            }
        }
    }

    ESP_LOGI(TAG, "Working catalog loaded to staging: %d stations ready for validation", (int)s_staging_count);
    return s_staging_count > 0;
}

// ============================================================
// Групповое удаление станций
// ============================================================
bool catalog_remove_stations(const uint32_t* ids, size_t count) {
    if (ids == nullptr || count == 0) return false;

    size_t removed_count = 0;

    for (size_t ci = 0; ci < g_catalog.category_count; ci++) {
        catalog_category_t* cat = &g_catalog.categories[ci];
        size_t new_count = 0;

        for (size_t si = 0; si < cat->count; si++) {
            bool should_remove = false;
            for (size_t i = 0; i < count; i++) {
                if (cat->stations[si].id == ids[i]) {
                    should_remove = true;
                    break;
                }
            }

            if (!should_remove) {
                if (new_count != si) {
                    cat->stations[new_count] = cat->stations[si];
                }
                new_count++;
            } else {
                removed_count++;
            }
        }

        cat->count = new_count;
    }

    g_catalog.total_stations -= removed_count;

    // Удаляем эти ID из избранного
    for (size_t i = 0; i < count; i++) {
        for (size_t j = 0; j < g_catalog.favorite_count; j++) {
            if (g_catalog.favorite_ids[j] == ids[i]) {
                for (size_t k = j; k < g_catalog.favorite_count - 1; k++) {
                    g_catalog.favorite_ids[k] = g_catalog.favorite_ids[k + 1];
                }
                g_catalog.favorite_count--;
                break;
            }
        }
    }

    // Сохраняем изменения на SD
    if (sd_manager_is_mounted()) {
        ensure_sd_radio_dir();
        catalog_save_json_file(CATALOG_JSON_PATH);
        catalog_save_favorites();
    }

    ESP_LOGI(TAG, "Removed %d stations from catalog", (int)removed_count);
    return removed_count > 0;
}

// ============================================================
// Удаление категории
// ============================================================
bool catalog_remove_category(size_t category_index) {
    if (category_index >= g_catalog.category_count) return false;

    catalog_category_t* cat = &g_catalog.categories[category_index];

    // 1. Удаляем станции внутри категории
    if (cat->stations) {
        heap_caps_free(cat->stations);
        cat->stations = nullptr;
    }
    g_catalog.total_stations -= cat->count;

    // 2. Сдвигаем оставшиеся категории, чтобы закрыть дыру
    for (size_t i = category_index; i < g_catalog.category_count - 1; i++) {
        g_catalog.categories[i] = g_catalog.categories[i + 1];
    }
    g_catalog.category_count--;

    // 3. Сохраняем обновленный каталог на SD
    if (sd_manager_is_mounted()) {
        ensure_sd_radio_dir();
        catalog_save_json_file(CATALOG_JSON_PATH);
    }

    ESP_LOGI(TAG, "Category removed at index %d", (int)category_index);
    return true;
}

// ============================================================
// Пакетное удаление категорий (с сортировкой индексов)
// ============================================================
size_t catalog_remove_categories(const size_t* indices, size_t count) {
    if (indices == nullptr || count == 0) return 0;

    // Копируем индексы и сортируем по убыванию, чтобы при удалении
    // индексы не "сдвигались" и мы не удалили не те категории.
    size_t sorted[64];
    size_t actual_count = (count > 64) ? 64 : count;
    for (size_t i = 0; i < actual_count; i++) sorted[i] = indices[i];

    // Простая пузырьковая сортировка по убыванию
    for (size_t i = 0; i < actual_count - 1; i++) {
        for (size_t j = 0; j < actual_count - i - 1; j++) {
            if (sorted[j] < sorted[j + 1]) {
                size_t temp = sorted[j];
                sorted[j] = sorted[j + 1];
                sorted[j + 1] = temp;
            }
        }
    }

    size_t removed = 0;
    for (size_t i = 0; i < actual_count; i++) {
        size_t idx = sorted[i];
        if (idx >= g_catalog.category_count) continue; // Защита от невалидных индексов

        catalog_category_t* cat = &g_catalog.categories[idx];

        // Удаляем станции внутри категории
        if (cat->stations) {
            heap_caps_free(cat->stations);
            cat->stations = nullptr;
        }
        g_catalog.total_stations -= cat->count;

        // Сдвигаем оставшиеся категории
        for (size_t k = idx; k < g_catalog.category_count - 1; k++) {
            g_catalog.categories[k] = g_catalog.categories[k + 1];
        }
        g_catalog.category_count--;
        removed++;
    }

    // Сохраняем обновлённый каталог на SD
    if (removed > 0 && sd_manager_is_mounted()) {
        ensure_sd_radio_dir();
        catalog_save_json_file(CATALOG_JSON_PATH);
    }

    ESP_LOGI(TAG, "Removed %d categories", (int)removed);
    return removed;
}

static void wifi_test_task(void *pvParameters) {
    wifi_test_args_t *args = (wifi_test_args_t *)pvParameters;

    ESP_LOGI(TAG, "Wi-Fi test task: stopping stream before test");

    if (success) {
        ESP_LOGI(TAG, "Wi-Fi test SUCCESS");
        ui_set_wifi_test_result(WIFI_TEST_SUCCESS);

    } else {

        g_restart_stream_after_wifi = false;
        g_wifi_reconnect_in_progress = true;
        g_wifi_link_seen_down = true;

    free(args);

    portENTER_CRITICAL(&g_wifi_test_mux);
    g_wifi_test_active = false;
    portEXIT_CRITICAL(&g_wifi_test_mux);

    vTaskDelete(NULL);
}

extern "C" void app_main(void) {
    ESP_LOGI(TAG, "Starting GMF Radio...");

    // 1. Инициализация хранилища
    storage_init();

    // 2. Инициализация M5Stack
    auto cfg = M5.config();
    M5.begin(cfg);
    ESP_LOGI(TAG, "M5Stack initialized");

    // 2.1 Инициализация SD
    bool sd_ok = sd_manager_init();
    if (sd_ok) {
        sd_manager_self_test();
    } else {
        ESP_LOGW("RADIO", "SD card unavailable. Will use built-in fallback catalog only.");
    }

    // 2.2 Инициализация Каталога
    catalog_init();
    catalog_log_summary();
    catalog_self_test();

    // 2.2.5 Загрузка избранного
    if (catalog_load_favorites()) {
        ESP_LOGI(TAG, "Favorites loaded: %u stations", (unsigned)catalog_get_favorites_count());
    } else {
        ESP_LOGI(TAG, "No favorites file found or SD unavailable");
    }

    // 2.3 Инициализация UI
    ui_init();

    // 3. Загрузка настроек Wi-Fi
    if (!storage_load_wifi(g_wifi_ssid, sizeof(g_wifi_ssid), g_wifi_pass, sizeof(g_wifi_pass))) {
        strcpy(g_wifi_ssid, DEFAULT_WIFI_SSID);
        strcpy(g_wifi_pass, DEFAULT_WIFI_PASS);
    }
    ui_set_wifi_credentials(g_wifi_ssid, g_wifi_pass);

    // 4. Загрузка последней станции (умная логика: сначала по ID, потом fallback на индекс)
    uint32_t last_st_id = storage_load_last_station_id();
    size_t found_cat = 0, found_st = 0;
    bool found_by_id = false;

    if (last_st_id > 0) {
        found_by_id = catalog_find_station_by_id(last_st_id, &found_cat, &found_st);
    }

    if (found_by_id) {
        g_selected_station = (int)catalog_global_index(found_cat, found_st);
        ESP_LOGI(TAG, "Restored last station by ID: %d", g_selected_station);
    } else {
        // Fallback на старый индекс (если ID не найден или это первый запуск со старыми данными)
        int old_idx = storage_load_station();
        int total = (int)catalog_total_stations();
        if (total <= 0) total = g_total_stations; // fallback на config.cpp если каталог пуст

        if (old_idx >= 0 && old_idx < total) {
            g_selected_station = old_idx;
            ESP_LOGI(TAG, "Restored last station by old index: %d", g_selected_station);
        } else {
            g_selected_station = 0;
            ESP_LOGI(TAG, "Invalid last station, defaulting to 0");
        }
    }

    // 5. Подключение к Wi-Fi (с приоритетом авто-подключения)
    wifi_known_networks_load();
    if (!wifi_try_auto_connect()) {
        ESP_LOGI(TAG, "Auto-connect failed or no known networks, falling back to saved credentials");
        wifi_connect(g_wifi_ssid, g_wifi_pass);
    }

    // Сразу пробуем запустить SNTP, если Wi-Fi поднялся
    if (wifi_is_connected() && !g_sntp_started) {
        wifi_start_sntp();
        g_sntp_started = true;
    }

    // 6. Инициализация аудио
    RingbufHandle_t ringbuf = xRingbufferCreate(RINGBUF_SIZE, RINGBUF_TYPE_NOSPLIT);
    if (ringbuf == NULL) {
        ESP_LOGE(TAG, "Failed to create ring buffer");
        return;
    }
    audio_init(ringbuf);
    audio_start();

    // 7. Инициализация HTTP стрима
    http_stream_init(ringbuf);
    http_stream_start(get_station_url_from_catalog(g_selected_station));

    // 8.

    ESP_LOGI(TAG, "All modules initialized, entering main loop");

    // 9. Главный цикл приложения
    while (1) {
        // Обновляем UI
        ui_update();

        // Если Wi-Fi появился позже (например, после реконнекта) — запускаем SNTP
        if (!g_sntp_started && wifi_is_connected()) {
            wifi_start_sntp();
            g_sntp_started = true;
        }

        // 1. Если запрошен reconnect и поток уже остановлен — выполняем реконнект
        if (g_wifi_pending_reconnect &&
            !http_stream_is_running() &&
            !wifi_is_testing() &&
            !g_wifi_reconnect_in_progress) {

            ESP_LOGI(TAG, "Stream stopped, performing Wi-Fi reconnect");
            wifi_reconnect(g_wifi_ssid, g_wifi_pass);

            g_wifi_pending_reconnect = false;
            g_wifi_reconnect_in_progress = true;
            g_wifi_link_seen_down = false;
        }

        // 2. Если идёт reconnect — ждём сначала down, потом up, затем рестартим поток
        // ========================================================
        // Wi-Fi reconnect monitor + watchdog
        // ========================================================
        if (g_wifi_reconnect_in_progress) {
            bool link_up = wifi_is_connected();

            if (!link_up) {
                g_wifi_link_seen_down = true;
            } else if (g_wifi_link_seen_down) {
                ESP_LOGI(TAG, "Wi-Fi reconnect finished. Restarting stream...");

                g_wifi_reconnect_in_progress = false;
                g_wifi_link_seen_down        = false;
                g_wifi_pending_reconnect     = false;

                http_stream_restart();
                ui_set_current_mode(MODE_PLAYBACK);
            } else {
                // Watchdog: если reconnect запрошен, но Wi-Fi ни разу не падал —
                // через 15 секунд принудительно возобновляем поток.
                uint64_t now_ms = esp_timer_get_time() / 1000;

                if (now_ms - g_wifi_reconnect_started_ms > 15000) {
                    ESP_LOGW(TAG, "Reconnect watchdog fired. Link stayed up. Resuming stream.");

                    g_wifi_reconnect_in_progress = false;
                    g_wifi_link_seen_down        = false;
                    g_wifi_pending_reconnect     = false;

                    http_stream_restart();
                    ui_set_current_mode(MODE_PLAYBACK);
                }
            }
        }

        // 3. Отдельный случай: тест завершился успехом, Wi-Fi уже подключен
        if (g_restart_stream_after_wifi &&
            wifi_is_connected() &&
            !wifi_is_testing() &&
            !g_wifi_test_active &&
            !g_wifi_reconnect_in_progress &&
            !g_wifi_pending_reconnect &&
            !http_stream_is_running()) {

            ESP_LOGI(TAG, "Restarting stream after successful Wi-Fi test");
            http_stream_start(get_station_url_from_catalog(g_selected_station));
            g_restart_stream_after_wifi = false;
        }

        uint32_t starve, overflow;
        audio_get_stats(&starve, &overflow);

        ui_set_playback_info(
            get_station_name(g_selected_station),
            http_stream_get_title(),
            audio_get_buffer_level(),
            PREFILL_TARGET,
            starve,
            overflow
        );

        ui_set_network_info(
            http_stream_get_ring_fill(),
            RINGBUF_SIZE
        );

        ui_event_t event = ui_get_event();

        if (event == UI_EVENT_STATION_CHANGED) {
            g_selected_station = ui_get_selected_station();

            // Сохраняем и индекс (для обратной совместимости), и ID (для надежности)
            storage_save_station(g_selected_station);

            const catalog_station_t* st = catalog_get_station_by_global_index(g_selected_station);
            if (st != nullptr) {
                storage_save_last_station_id(st->id);
            }

            http_stream_start(get_station_url_from_catalog(g_selected_station));
            ESP_LOGI(TAG, "Station changed to %d: %s", g_selected_station, get_station_name(g_selected_station));
        }
        else if (event == UI_EVENT_WIFI_SCAN_REQUESTED) {
            ESP_LOGI(TAG, "Pausing stream and starting Wi-Fi scan...");

            http_stream_request_stop();

            int found = wifi_scan_networks();

            ESP_LOGI(TAG, "Wi-Fi scan found %d networks", found);

            ui_set_wifi_scan_ready(found > 0);

            // Если сетей не нашли — сразу возобновляем поток,
            // чтобы радио не молчало просто так.
            if (found <= 0) {
                ESP_LOGI(TAG, "No Wi-Fi networks found. Resuming stream.");
                http_stream_restart();
            }
        }
        else if (event == UI_EVENT_WIFI_CONNECT_REQUESTED) {
            ESP_LOGI(TAG, "Wi-Fi Connect requested. Saving credentials...");

            char new_ssid[33] = {0};
            char new_pass[65] = {0};

            ui_get_wifi_credentials(new_ssid, sizeof(new_ssid), new_pass, sizeof(new_pass));

            bool same_ssid = (strcmp(new_ssid, g_wifi_ssid) == 0);
            bool same_pass = (strcmp(new_pass, g_wifi_pass) == 0);
            bool link_up   = wifi_is_connected();

            ESP_LOGI(TAG, "Wi-Fi Connect check: same_ssid=%d same_pass=%d link_up=%d",
                     (int)same_ssid, (int)same_pass, (int)link_up);

            // 1. Всегда сохраняем креды
            storage_save_wifi(new_ssid, new_pass);

            strncpy(g_wifi_ssid, new_ssid, sizeof(g_wifi_ssid) - 1);
            g_wifi_ssid[sizeof(g_wifi_ssid) - 1] = '\0';

            strncpy(g_wifi_pass, new_pass, sizeof(g_wifi_pass) - 1);
            g_wifi_pass[sizeof(g_wifi_pass) - 1] = '\0';

            ui_set_wifi_credentials(g_wifi_ssid, g_wifi_pass);

            // 2. Если креды те же и Wi-Fi уже подключён — сразу возобновляем поток.
            if (same_ssid && same_pass && link_up) {
                ESP_LOGI(TAG, "Wi-Fi unchanged and link up. Resuming playback immediately.");

                http_stream_restart();

                ui_set_current_mode(MODE_PLAYBACK);

                // ВАЖНО: выходим из обработчика, чтобы не провалиться в reconnect.
                goto after_wifi_connect;
            }

            // 3. Креды изменились или link down — полноценный reconnect.
            ESP_LOGI(TAG, "Wi-Fi changed or link down. Reconnecting...");

            http_stream_request_stop();

            wifi_reconnect(g_wifi_ssid, g_wifi_pass);

            g_wifi_pending_reconnect    = true;
            g_wifi_reconnect_in_progress = true;
            g_wifi_link_seen_down        = false;
            g_wifi_reconnect_started_ms  = esp_timer_get_time() / 1000;

            ui_set_current_mode(MODE_SETTINGS);

        after_wifi_connect:
            ;
        }
        else if (event == UI_EVENT_VOLUME_CHANGED) {
            audio_set_volume(ui_get_volume());
        }
        else if (event == UI_EVENT_MUTE_TOGGLED) {
            audio_toggle_mute();
        }
        else if (event == UI_EVENT_WIFI_REVERT) {
            if (g_wifi_test_active) {
                ESP_LOGW(TAG, "Wi-Fi test active, ignoring Revert");
            } else {
                ESP_LOGI(TAG, "Wi-Fi settings reverted to last saved");

                http_stream_request_stop();
                g_wifi_pending_reconnect = true;
                g_restart_stream_after_wifi = false;
                g_wifi_reconnect_in_progress = false;
            }
        }
        else if (event == UI_EVENT_WIFI_SAVE) {
            char new_ssid[33], new_pass[65];
            ui_get_wifi_credentials(new_ssid, sizeof(new_ssid), new_pass, sizeof(new_pass));

            storage_save_wifi(new_ssid, new_pass);
            ui_set_wifi_credentials(new_ssid, new_pass);

            strcpy(g_wifi_ssid, new_ssid);
            strcpy(g_wifi_pass, new_pass);

            ESP_LOGI(TAG, "Wi-Fi settings saved to NVS (no reconnect)");
        }
        else if (event == UI_EVENT_WIFI_TEST) {
            bool start_test = false;

            portENTER_CRITICAL(&g_wifi_test_mux);
            if (!g_wifi_test_active) {
                g_wifi_test_active = true;
                start_test = true;
            }
            portEXIT_CRITICAL(&g_wifi_test_mux);

            if (!start_test) {
                ESP_LOGW(TAG, "Wi-Fi test already active, ignoring repeated Test event");
                ui_set_wifi_test_result(WIFI_TEST_IN_PROGRESS);
            } else {
                ESP_LOGI(TAG, "Starting asynchronous Wi-Fi test");

                g_restart_stream_after_wifi = false;
                g_wifi_pending_reconnect = false;
                g_wifi_reconnect_in_progress = false;

                wifi_test_args_t *args = (wifi_test_args_t *)malloc(sizeof(wifi_test_args_t));
                if (args != NULL) {
                    ui_get_wifi_credentials(args->ssid, sizeof(args->ssid), args->pass, sizeof(args->pass));

                    BaseType_t created = xTaskCreatePinnedToCore(

                        8192,
                        args,
                        6,
                        NULL,
                        1
                    );

                    if (created != pdPASS) {
                        ESP_LOGE(TAG, "Failed to create Wi-Fi test task");
                        ui_set_wifi_test_result(WIFI_TEST_FAILED_OTHER);

                        portENTER_CRITICAL(&g_wifi_test_mux);
                        g_wifi_test_active = false;
                        portEXIT_CRITICAL(&g_wifi_test_mux);

                        free(args);
                    }
                } else {
                    ESP_LOGE(TAG, "Failed to allocate Wi-Fi test args");
                    ui_set_wifi_test_result(WIFI_TEST_FAILED_OTHER);

                    portENTER_CRITICAL(&g_wifi_test_mux);
                    g_wifi_test_active = false;
                    portEXIT_CRITICAL(&g_wifi_test_mux);
                }
            }
        }
        else if (event == UI_EVENT_WIFI_CONNECT) {
            if (g_wifi_test_active) {
                ESP_LOGW(TAG, "Wi-Fi test active, ignoring Connect");
            } else {
                char new_ssid[33], new_pass[65];
                ui_get_wifi_credentials(new_ssid, sizeof(new_ssid), new_pass, sizeof(new_pass));

                strcpy(g_wifi_ssid, new_ssid);
                strcpy(g_wifi_pass, new_pass);

                storage_save_wifi(g_wifi_ssid, g_wifi_pass);
                ui_set_wifi_credentials(g_wifi_ssid, g_wifi_pass);

                ESP_LOGI(TAG, "Wi-Fi settings saved, preparing reconnect to %s", g_wifi_ssid);

                http_stream_request_stop();
                g_wifi_pending_reconnect = true;
                g_restart_stream_after_wifi = false;
                g_wifi_reconnect_in_progress = false;
            }
        }

        vTaskDelay(pdMS_TO_TICKS(50));
     }
}

// =====================================================================
// 1. Улучшенный детектор русского UTF-8 (теперь видит ВСЕ русские буквы)
// =====================================================================
static bool is_valid_utf8_russian(const char* str) {
    if (!str) return false;
    const char* p = str;
    int russian_chars = 0;
    int total_high_bytes = 0;

    while (*p) {
        unsigned char c = (unsigned char)*p;
        if (c >= 0xC0) {
            total_high_bytes++;
            bool is_russian = false;

            // Заглавные А-Я: D0 90 - D0 AF
            if (c == 0xD0 && p[1] >= 0x90 && p[1] <= 0xAF) is_russian = true;
            // Строчные а-п: D0 B0 - D0 BF (этого не хватало в старой версии!)
            if (c == 0xD0 && p[1] >= 0xB0 && p[1] <= 0xBF) is_russian = true;
            // Строчные р-я: D1 80 - D1 8F
            if (c == 0xD1 && p[1] >= 0x80 && p[1] <= 0x8F) is_russian = true;
            // Ё: D0 81, ё: D1 91
            if (c == 0xD0 && p[1] == 0x81) is_russian = true;
            if (c == 0xD1 && p[1] == 0x91) is_russian = true;

            if (is_russian) {
                russian_chars++;
                p += 2; // Пропускаем валидный 2-байтовый символ
            } else {
                // Высокий байт, но не русская буква - это не валидный русский UTF-8
                return false;
            }
        } else {
            p++; // ASCII символы (английские буквы, цифры, пробелы)
        }
    }

    // Если есть высокие байты, но ни одной русской буквы - это не русский UTF-8
    if (total_high_bytes > 0 && russian_chars == 0) {
        return false;
    }

    return true; // Строка либо чистый ASCII, либо валидный русский UTF-8
}

// =====================================================================
// 2. Математически точный конвертер CP1251 -> UTF-8
// =====================================================================
static void cp1251_to_utf8(const char* src, char* dst, size_t dst_size) {
    size_t j = 0;
    for (size_t i = 0; src[i] != '\0' && j < dst_size - 2; i++) {
        unsigned char c = (unsigned char)src[i];

        if (c >= 0xC0 && c <= 0xFF) {
            if (c == 0xA8) { // Ё
                if (j + 2 < dst_size) { dst[j++] = 0xD0; dst[j++] = 0x81; }
            }
            else if (c == 0xB8) { // ё
                if (j + 2 < dst_size) { dst[j++] = 0xD1; dst[j++] = 0x91; }
            }
            else if (c >= 0xC0 && c <= 0xEF) { // А-п (включая заглавные и часть строчных)
                if (j + 2 < dst_size) {
                    dst[j++] = 0xD0;
                    dst[j++] = c - 0xC0 + 0x90;
                }
            }
            else if (c >= 0xF0 && c <= 0xFF) { // р-я (вторая половина строчных)
                if (j + 2 < dst_size) {
                    dst[j++] = 0xD1;
                    dst[j++] = c - 0xF0 + 0x80;
                }
            }
        } else {
            dst[j++] = c; // ASCII символы копируем как есть
        }
    }
    dst[j] = '\0';
}

static esp_err_t http_event_handler(esp_http_client_event_t *evt) {
    if (evt->event_id == HTTP_EVENT_ON_HEADER &&
        strcasecmp(evt->header_key, "icy-metaint") == 0) {
        s_metaint = atoi(evt->header_value);
    }
    return ESP_OK;
}

static void http_interruptible_delay(int ms) {
    for (int elapsed = 0; elapsed < ms && !s_force_stop; elapsed += 50) {
        vTaskDelay(pdMS_TO_TICKS(50));
    }
}

static void http_clear_ringbuf(void) {
    if (s_ringbuf == NULL) return;

    size_t item_size;
    void *data;
    while ((data = xRingbufferReceive(s_ringbuf, &item_size, 0)) != NULL) {
        vRingbufferReturnItem(s_ringbuf, data);
    }
}

static void http_read_task(void *pvParameters) {
    const char *url = (const char *)pvParameters;

    s_http_task_running = true;
    ESP_LOGI(TAG, "HTTP task started, URL: %s", url);

    uint8_t *net_buf = (uint8_t *)heap_caps_malloc(4096, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    if (net_buf == NULL) {
        ESP_LOGE(TAG, "Failed to allocate net_buf");
        s_http_task_running = false;
        s_http_task_handle = NULL;
        vTaskDelete(NULL);
        return;
    }

    esp_http_client_handle_t client = NULL;
    int retry_delay_ms = 1000;

    enum { STATE_AUDIO, STATE_META_LEN, STATE_META_DATA } state = STATE_AUDIO;
    int bytes_until_meta = 0;
    int meta_bytes_remaining = 0;
    int meta_buf_pos = 0;
    char meta_buf[256] = {};

    while (s_http_task_running && !s_force_stop) {
        if (client == NULL) {
            esp_http_client_config_t config = {};
            config.url = url;
            config.buffer_size = 4096;
            config.crt_bundle_attach = esp_crt_bundle_attach;
            config.timeout_ms = 1000;
            // ============================
            config.user_agent = "WinampMPEG/5.0"; // Или "VLC/3.0.0 LibVLC/3.0.0"
            // ============================
            config.event_handler = http_event_handler;

            client = esp_http_client_init(&config);
            if (client == NULL) {
                ESP_LOGW(TAG, "HTTP client init failed, retry in %d ms", retry_delay_ms);
                http_interruptible_delay(retry_delay_ms);
                if (s_force_stop) break;
                if (retry_delay_ms < 8000) retry_delay_ms *= 2;
                continue;
            }

            esp_http_client_set_header(client, "Icy-MetaData", "1");

            if (esp_http_client_open(client, 0) != ESP_OK) {
                ESP_LOGW(TAG, "HTTP open failed, retry in %d ms", retry_delay_ms);
                esp_http_client_cleanup(client);
                client = NULL;

                http_interruptible_delay(retry_delay_ms);
                if (s_force_stop) break;

                if (retry_delay_ms < 8000) retry_delay_ms *= 2;
                continue;
            }

            esp_http_client_fetch_headers(client);

            state = STATE_AUDIO;
            bytes_until_meta = s_metaint;
            meta_bytes_remaining = 0;
            meta_buf_pos = 0;
            memset(meta_buf, 0, sizeof(meta_buf));

            retry_delay_ms = 1000;
            ESP_LOGI(TAG, "HTTP connected, metaint=%d", s_metaint);
        }

        if (s_force_stop) break;

        int read_len = esp_http_client_read(client, (char *)net_buf, 4096);
        if (read_len <= 0) {
            ESP_LOGW(TAG, "HTTP read returned %d, reconnecting in %d ms", read_len, retry_delay_ms);

            esp_http_client_close(client);
            esp_http_client_cleanup(client);
            client = NULL;

            http_interruptible_delay(retry_delay_ms);
            if (s_force_stop) break;

            if (retry_delay_ms < 8000) retry_delay_ms *= 2;
            continue;
        }

        retry_delay_ms = 1000;

        int offset = 0;

        if (s_metaint == 0) {
            xRingbufferSend(s_ringbuf, net_buf, read_len, pdMS_TO_TICKS(100));
        } else {
            while (offset < read_len) {
                if (state == STATE_AUDIO) {
                    int chunk = (read_len - offset < bytes_until_meta)
                                    ? (read_len - offset)
                                    : bytes_until_meta;

                    xRingbufferSend(s_ringbuf, net_buf + offset, chunk, pdMS_TO_TICKS(100));

                    offset += chunk;
                    bytes_until_meta -= chunk;

                    if (bytes_until_meta == 0) {
                        state = STATE_META_LEN;
                        meta_buf_pos = 0;
                    }
                } else if (state == STATE_META_LEN) {
                    meta_bytes_remaining = net_buf[offset++] * 16;
                    state = (meta_bytes_remaining == 0) ? STATE_AUDIO : STATE_META_DATA;

                    if (state == STATE_AUDIO) {
                        bytes_until_meta = s_metaint;
                    }
                } else {
                    int to_copy = (read_len - offset < meta_bytes_remaining)
                                      ? (read_len - offset)
                                      : meta_bytes_remaining;

                    if (meta_buf_pos + to_copy < 255) {
                        memcpy(meta_buf + meta_buf_pos, net_buf + offset, to_copy);
                        meta_buf_pos += to_copy;
                    }

                    offset += to_copy;
                    meta_bytes_remaining -= to_copy;

                    if (meta_bytes_remaining == 0) {
                        meta_buf[meta_buf_pos] = '\0';

                        char *t = strstr(meta_buf, "StreamTitle='");
                        if (t) {
                            t += 13;
                            char *e = strchr(t, '\'');
                            if (e) {
                                int l = e - t;
                                if (l > 127) l = 127;

                                char temp_title[128];
                                memcpy(temp_title, t, l);
                                temp_title[l] = '\0';

                                // УЛУЧШЕННОЕ ОПРЕДЕЛЕНИЕ: проверяем, является ли это валидным UTF-8 с РУССКИМИ буквами
                                if (!is_valid_utf8_russian(temp_title)) {
                                    // Это не русский UTF-8, конвертируем из CP1251
                                    cp1251_to_utf8(temp_title, s_stream_title, sizeof(s_stream_title));
                                } else {
                                    // Это валидный русский UTF-8, просто копируем
                                    strncpy(s_stream_title, temp_title, sizeof(s_stream_title) - 1);
                                    s_stream_title[sizeof(s_stream_title) - 1] = '\0';
                                }

                                ESP_LOGI(TAG, "StreamTitle: %s", s_stream_title);
                            }
                        }

                        state = STATE_AUDIO;
                        bytes_until_meta = s_metaint;
                    }
                }
            }
        }
    }

    if (client != NULL) {
        esp_http_client_close(client);
        esp_http_client_cleanup(client);
    }

    free(net_buf);

    ESP_LOGI(TAG, "HTTP task exiting");
    s_http_task_running = false;
    s_http_task_handle = NULL;
    vTaskDelete(NULL);
}

void http_stream_init(RingbufHandle_t ringbuf) {
    s_ringbuf = ringbuf;

    if (s_http_mutex == NULL) {
        s_http_mutex = xSemaphoreCreateMutex();
    }

    ESP_LOGI(TAG, "HTTP stream initialized");
}

bool http_stream_is_running(void) {
    return s_http_task_running;
}

void http_stream_request_stop(void) {
    // Неблокирующая остановка.
    // Задача сама выйдет, когда закончит текущий read/delay.
    s_force_stop = true;
}

static void http_stream_stop_unlocked(void) {
    bool need_clear = false;

    if (s_http_task_handle != NULL) {
        s_force_stop = true;

        int wait = 0;
        while (s_http_task_running && wait < 1000) {
            vTaskDelay(pdMS_TO_TICKS(10));
            wait++;
        }

        if (!s_http_task_running) {
            s_force_stop = false;
            need_clear = true;
        } else {
            // НИКОГДА не делаем vTaskDelete здесь.
            // Иначе lwIP может упасть.
            ESP_LOGE(TAG, "HTTP task did not exit in 10s; refusing force kill");
            need_clear = false;
        }
    } else {
        s_force_stop = false;
        need_clear = true;
    }

    if (need_clear) {
        http_clear_ringbuf();
    }
}

static void http_stream_start_unlocked(const char *url) {
    if (url == NULL) {
        ESP_LOGE(TAG, "Cannot start HTTP stream: URL is NULL");
        return;
    }

    http_stream_stop_unlocked();

    if (s_http_task_running) {
        ESP_LOGE(TAG, "Cannot start HTTP stream: old task is still running");
        return;
    }

    s_force_stop = false;
    s_metaint = 0;
    strcpy(s_stream_title, "Connecting...");

    http_clear_ringbuf();

    BaseType_t created = xTaskCreatePinnedToCore(
        http_read_task,
        "http_read",
        8192,
        (void *)url,
        15,
        &s_http_task_handle,
        1
    );

    if (created != pdPASS) {
        ESP_LOGE(TAG, "Failed to create HTTP task");
        s_http_task_handle = NULL;
        s_http_task_running = false;
    }
}

void http_stream_stop(void) {
    if (s_http_mutex == NULL) return;

    xSemaphoreTake(s_http_mutex, portMAX_DELAY);
    http_stream_stop_unlocked();
    xSemaphoreGive(s_http_mutex);
}

void http_stream_start(const char *url) {

	if (url != nullptr) {
	    strncpy(s_http_last_url, url, sizeof(s_http_last_url) - 1);
	    s_http_last_url[sizeof(s_http_last_url) - 1] = '\0';
	}

    if (s_http_mutex == NULL) return;

    xSemaphoreTake(s_http_mutex, portMAX_DELAY);
    http_stream_start_unlocked(url);
    xSemaphoreGive(s_http_mutex);
}

const char *http_stream_get_title(void) {
    return s_stream_title;
}

int http_stream_get_metaint(void) {
    return s_metaint;
}

uint32_t http_stream_get_ring_fill(void) {
    if (s_ringbuf == NULL) {
        return 0;
    }

    size_t free_bytes = xRingbufferGetCurFreeSize(s_ringbuf);

    if (free_bytes >= RINGBUF_SIZE) {
        return 0;
    }

    uint32_t used = (uint32_t)(RINGBUF_SIZE - free_bytes);

    if (used > RINGBUF_SIZE) {
        used = RINGBUF_SIZE;
    }

    return used;
}

void http_stream_restart(void) {
    if (s_http_last_url[0] == '\0') {
        ESP_LOGW(TAG, "http_stream_restart: no last URL stored");
        return;
    }

    ESP_LOGI(TAG, "http_stream_restart: restarting with URL: %s", s_http_last_url);
    http_stream_start(s_http_last_url);
}

#include "m3u_parser.h"

// Реализация хэша FNV-1a (32-bit)
uint32_t catalog_generate_station_id(const char* url) {
    uint32_t hash = 2166136261u;
    for (const char* p = url; *p; p++) {
        hash ^= (uint8_t)*p;
        hash *= 16777619u;
    }
    return hash;
}

int m3u_parse_file(const char* filepath, const char* default_group, catalog_station_t* out_stations, int max_stations) {
    FILE* f = fopen(filepath, "r");
    if (!f) {
        ESP_LOGE(TAG, "Failed to open M3U file: %s", filepath);
        return 0;
    }

    char line[512];
    int count = 0;
    char current_name[CATALOG_MAX_NAME_LEN + 1] = "Unknown";
    bool has_extinf = false;

    char active_group[CATALOG_MAX_CATEGORY_LEN + 1];
    if (default_group && default_group[0] != '\0') {
        strncpy(active_group, default_group, sizeof(active_group) - 1);
        active_group[sizeof(active_group) - 1] = '\0';
    } else {
        strcpy(active_group, "Imported");
    }

    while (fgets(line, sizeof(line), f) && count < max_stations) {
        // Обрезаем пробелы и переносы строк в конце
        size_t len = strlen(line);
        while (len > 0 && (line[len-1] == '\n' || line[len-1] == '\r' || line[len-1] == ' ' || line[len-1] == '\t')) {
            line[--len] = '\0';
        }
        if (len == 0) continue;

        if (strncmp(line, "#EXTINF:", 8) == 0) {
            has_extinf = true;
            strcpy(current_name, "Unknown");

            // 1. Ищем group-title="..."
            char final_group[CATALOG_MAX_CATEGORY_LEN + 1];
            char* gt = strstr(line, "group-title=\"");
            if (gt) {
                gt += 13;
                char* gt_end = strchr(gt, '"');
                if (gt_end) {
                    size_t g_len = gt_end - gt;
                    if (g_len > 0 && g_len < sizeof(final_group)) {
                        strncpy(final_group, gt, g_len);
                        final_group[g_len] = '\0';
                    } else {
                        strcpy(final_group, active_group);
                    }
                } else {
                    strcpy(final_group, active_group);
                }
            } else {
                strcpy(final_group, active_group);
            }

            // 2. Ищем имя после последней запятой
            char* comma = strrchr(line, ',');
            if (comma) {
                strncpy(current_name, comma + 1, sizeof(current_name) - 1);
                current_name[sizeof(current_name) - 1] = '\0';
            }

            // Сохраняем группу во временное поле (мы используем name как носитель, но лучше иметь отдельное,
            // однако для простоты пока сохраним группу в name, а при merge решим, или добавим поле в struct)
            // Для совместимости с текущим catalog_station_t, мы пока сохраним имя, а группу передадим в add_station
            // Но так как out_stations - это catalog_station_t, мы временно положим имя.
            // *Уточнение*: мы будем передавать группу отдельно в merge, или добавим поле в struct (см. Шаг 2).
            strncpy(out_stations[count].name, current_name, sizeof(out_stations[count].name) - 1);
            out_stations[count].name[sizeof(out_stations[count].name) - 1] = '\0';

            // Временно используем bitrate_kbps как хранение индекса группы, но это грязно.
            // Лучше добавим поле `char group[48]` в catalog_station_t на Шаге 2.
            // Пока оставим name, а группу обработаем в парсере иначе, или просто добавим поле.
            // Добавим поле `group` в catalog_station_t ниже.

        } else if (line[0] != '#') {
            // Это строка с URL
            if (has_extinf || count == 0) {
                strncpy(out_stations[count].url, line, sizeof(out_stations[count].url) - 1);
                out_stations[count].url[sizeof(out_stations[count].url) - 1] = '\0';

                // Генерируем стабильный ID
                out_stations[count].id = catalog_generate_station_id(out_stations[count].url);
                out_stations[count].bitrate_kbps = 0; // Будет определен позже или при merge

                // Базовая валидация: URL должен начинаться с http:// или https://
                if (strncmp(out_stations[count].url, "http://", 7) == 0 ||
                    strncmp(out_stations[count].url, "https://", 8) == 0) {
                    count++;
                } else {
                    ESP_LOGW(TAG, "Skipping invalid URL: %s", out_stations[count].url);
                }
            }
            has_extinf = false; // Сброс для следующей итерации
        }
    }

    fclose(f);
    ESP_LOGI(TAG, "Parsed %d valid stations from %s", count, filepath);
    return count;
}

        uint32_t in_rate = (cur_in - last_in) * 2;
        uint32_t out_rate = (cur_out - last_out) * 2;
        last_in = cur_in;
        last_out = cur_out;

        int32_t rb_level = (int32_t)(cur_in - cur_out);

            M5.Display.setCursor(10, 40);
            M5.Display.print("BUFFERING / RECONNECTING");

            target_val = g_prefill_target;

/* ================= HTTP EVENT HANDLER (для перехвата заголовков ответа) ================= */
static esp_err_t http_event_handler(esp_http_client_event_t *evt)
{
    switch (evt->event_id) {
        case HTTP_EVENT_ON_HEADER:
            // evt->header_key и evt->header_value содержат заголовок ответа
            if (strcasecmp(evt->header_key, "icy-name") == 0) {
                strncpy(g_station_name, evt->header_value, sizeof(g_station_name) - 1);
                ESP_LOGI(TAG, "📻 Станция: %s", g_station_name);
            }
            else if (strcasecmp(evt->header_key, "icy-genre") == 0) {
                ESP_LOGI(TAG, "🎵 Жанр: %s", evt->header_value);
            }
            else if (strcasecmp(evt->header_key, "icy-br") == 0) {
                ESP_LOGI(TAG, "📊 Битрейт: %s кбит/с", evt->header_value);
            }
            else if (strcasecmp(evt->header_key, "icy-metaint") == 0) {
                g_metaint = atoi(evt->header_value);
                ESP_LOGI(TAG, "📋 ICY metaint: %d байт", g_metaint);
            }
            break;
        default:
            break;
    }
    return ESP_OK;
}

/* ================= ЗАДАЧА 1: ЧТЕНИЕ ИЗ СЕТИ (С ПРАВИЛЬНЫМ ПАРСЕРОМ ICY) ================= */
static void http_read_task(void *pvParameters) {
    ESP_LOGI(TAG, "Задача чтения из сети запущена");

    esp_http_client_config_t config = {};
    config.url = STREAM_URL;
    config.buffer_size = 4096;
    config.crt_bundle_attach = esp_crt_bundle_attach;
    config.timeout_ms = 10000;
    config.event_handler = http_event_handler;  // <-- ДОБАВЛЕНО: обработчик заголовков

    esp_http_client_handle_t client = esp_http_client_init(&config);
    esp_http_client_set_header(client, "Icy-MetaData", "1");

    esp_err_t err = esp_http_client_open(client, 0);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Ошибка открытия HTTP: %s", esp_err_to_name(err));
        vTaskDelete(NULL);
        return;
    }
    esp_http_client_fetch_headers(client);

    // === УДАЛЕНО: старый код чтения заголовков через get_header ===
    // Теперь заголовки читаются через event_handler

    uint8_t *net_buf = (uint8_t *)heap_caps_malloc(4096, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    if (!net_buf) {
        ESP_LOGE(TAG, "Нет памяти для сетевого буфера");
        vTaskDelete(NULL);
        return;
    }

    ESP_LOGI(TAG, "Начинаем чтение потока...");
    int total_read = 0;

    // === State machine для парсинга ICY ===
    enum { STATE_AUDIO, STATE_META_LEN, STATE_META_DATA } state = STATE_AUDIO;
    int bytes_until_meta = g_metaint;
    int meta_bytes_remaining = 0;
    uint8_t meta_len_byte = 0;
    char meta_buf[256] = {};
    int meta_buf_pos = 0;

    while (1) {
        int read_len = esp_http_client_read(client, (char *)net_buf, 4096);
        if (read_len <= 0) {
            ESP_LOGW(TAG, "HTTP поток завершен. Всего прочитано: %d байт", total_read);
            break;
        }

        if (g_metaint == 0) {
            // === Нет метаданных — просто отправляем всё в буфер ===
            if (xRingbufferSend(s_ringbuf, net_buf, read_len, pdMS_TO_TICKS(500)) != pdTRUE) {
                g_rb_overflow = g_rb_overflow + 1;
            } else {
                g_rb_in += read_len;
            }
        } else {
            // === Парсинг ICY через state machine ===
            int offset = 0;

            while (offset < read_len) {
                if (state == STATE_AUDIO) {
                    int audio_chunk = (read_len - offset < bytes_until_meta) ?
                                      (read_len - offset) : bytes_until_meta;

                    if (xRingbufferSend(s_ringbuf, net_buf + offset, audio_chunk, pdMS_TO_TICKS(500)) != pdTRUE) {
                        g_rb_overflow = g_rb_overflow + 1;
                    } else {
                        g_rb_in += audio_chunk;
                    }

                    offset += audio_chunk;
                    bytes_until_meta -= audio_chunk;

                    if (bytes_until_meta == 0) {
                        state = STATE_META_LEN;
                        meta_len_byte = 0;
                        meta_buf_pos = 0;
                    }
                }
                else if (state == STATE_META_LEN) {
                    meta_len_byte = net_buf[offset];
                    meta_bytes_remaining = meta_len_byte * 16;
                    offset++;

                    if (meta_bytes_remaining == 0) {
                        state = STATE_AUDIO;
                        bytes_until_meta = g_metaint;
                    } else {
                        state = STATE_META_DATA;
                    }
                }
                else if (state == STATE_META_DATA) {
                    int available = read_len - offset;
                    int to_copy = (available < meta_bytes_remaining) ? available : meta_bytes_remaining;

                    if (meta_buf_pos + to_copy < (int)sizeof(meta_buf) - 1) {
                        memcpy(meta_buf + meta_buf_pos, net_buf + offset, to_copy);
                        meta_buf_pos += to_copy;
                    }

                    offset += to_copy;
                    meta_bytes_remaining -= to_copy;

                    if (meta_bytes_remaining == 0) {
                        meta_buf[meta_buf_pos] = '\0';
                        char *title_start = strstr(meta_buf, "StreamTitle='");
                        if (title_start) {
                            title_start += 13;
                            char *title_end = strchr(title_start, '\'');
                            if (title_end) {
                                int title_len = title_end - title_start;
                                if (title_len > (int)sizeof(g_stream_title) - 1) {
                                    title_len = sizeof(g_stream_title) - 1;
                                }
                                memcpy(g_stream_title, title_start, title_len);
                                g_stream_title[title_len] = '\0';
                                g_metadata_updated = true;
                                ESP_LOGI(TAG, "🎶 Сейчас играет: %s", g_stream_title);
                            }
                        }
                        state = STATE_AUDIO;
                        bytes_until_meta = g_metaint;
                    }
                }
            }
        }

        total_read += read_len;

        if (total_read % (100 * 1024) < 4096) {
            ESP_LOGI(TAG, "Сеть: прочитано %d КБ", total_read / 1024);
        }
    }

    if (net_buf) heap_caps_free(net_buf);
    esp_http_client_close(client);
    esp_http_client_cleanup(client);
    vTaskDelete(NULL);
}

/* ================= ЗАДАЧА 2: ДЕКОДИРОВАНИЕ И ВЫВОД ЗВУКА ================= */
static void decode_play_task(void *pvParameters) {
    ESP_LOGI(TAG, "Задача декодирования запущена");

    HMP3Decoder mp3_decoder = MP3InitDecoder();
    if (!mp3_decoder) {
        ESP_LOGE(TAG, "Не удалось создать MP3 декодер");
        vTaskDelete(NULL);
        return;
    }

    uint8_t *inbuf = (uint8_t *)heap_caps_malloc(DEC_BUF_SIZE, MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    short *outbuf = (short *)heap_caps_malloc(2304 * sizeof(short), MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);

    if (!inbuf || !outbuf) {
        ESP_LOGE(TAG, "Нет памяти для буферов декодера");
        vTaskDelete(NULL);
        return;
    }

    int bytes_in_buffer = 0;
    bool i2s_started = false;
    int frames_decoded = 0;
    int frames_skipped = 0;

    int last_reported = 0;
    int prefill_target = PREFILL_TARGET;

    auto do_prefill = [&]() {
        g_prefill_active = true;
        g_prefill_target = prefill_target;
        last_reported = 0;

        while (bytes_in_buffer < prefill_target) {
            size_t item_size = 0;
            void *data = xRingbufferReceive(s_ringbuf, &item_size, pdMS_TO_TICKS(2000));
            if (data != NULL) {
                if (bytes_in_buffer + item_size <= DEC_BUF_SIZE) {
                    memcpy(inbuf + bytes_in_buffer, data, item_size);
                    bytes_in_buffer += item_size;
                    g_rb_out += item_size;
                }
                vRingbufferReturnItem(s_ringbuf, data);
            }

            g_prefill_current = bytes_in_buffer;

            if (bytes_in_buffer - last_reported >= 16 * 1024) {
                int percent = (bytes_in_buffer * 100) / prefill_target;
                ESP_LOGI(TAG, "Предзаполнение: %3d%% (%d/%d)", percent, bytes_in_buffer, prefill_target);
                last_reported = bytes_in_buffer;
            }
        }
        g_prefill_active = false;
    };

    ESP_LOGI(TAG, "Предзаполнение буфера (128 КБ)...");
    do_prefill();
    ESP_LOGI(TAG, "Буфер предзаполнен: %d байт. Воспроизведение!", bytes_in_buffer);
    g_playback_started = true;

    while (1) {
        if (bytes_in_buffer < REBUF_THRESHOLD) {
            ESP_LOGW(TAG, "⚠ Буфер опустошён (%d байт). Переподключение...", bytes_in_buffer);
            g_playback_started = false;

            while (1) {
                size_t item_size = 0;
                void *data = xRingbufferReceive(s_ringbuf, &item_size, 0);
                if (data == NULL) break;
                vRingbufferReturnItem(s_ringbuf, data);
            }

            g_rb_in = 0;
            g_rb_out = 0;
            bytes_in_buffer = 0;

            ESP_LOGI(TAG, "Повторное предзаполнение буфера (128 КБ)...");
            do_prefill();

            ESP_LOGI(TAG, "Буфер восстановлен: %d байт. Продолжаем воспроизведение!", bytes_in_buffer);
            g_playback_started = true;
            continue;
        }

        while (bytes_in_buffer < PREFILL_TARGET) {
            size_t item_size = 0;
            TickType_t timeout = (bytes_in_buffer < 4 * 1024) ? pdMS_TO_TICKS(500) : pdMS_TO_TICKS(50);

            void *data = xRingbufferReceive(s_ringbuf, &item_size, timeout);
            if (data != NULL) {
                if (bytes_in_buffer + item_size <= DEC_BUF_SIZE) {
                    memcpy(inbuf + bytes_in_buffer, data, item_size);
                    bytes_in_buffer += item_size;
                    g_rb_out += item_size;
                }
                vRingbufferReturnItem(s_ringbuf, data);
            } else {
                if (bytes_in_buffer < 4 * 1024) {
                    g_rb_starve = g_rb_starve + 1;
                }
                break;
            }
        }

        // === ЗАЩИТА ОТ ЩЕЛЧКОВ: если буфер почти пуст, отправляем тишину ===
        if (bytes_in_buffer < 2048 && i2s_started) {
            static short silence_buf[2048];
            memset(silence_buf, 0, sizeof(silence_buf));
            size_t bytes_written = 0;
            i2s_channel_write(tx_chan, silence_buf, sizeof(silence_buf),
                              &bytes_written, pdMS_TO_TICKS(10));
            g_rb_starve = g_rb_starve + 1;
            continue; // Возвращаемся к пополнению буфера
        }

        g_dec_buf = bytes_in_buffer;

        uint8_t *read_ptr = inbuf;
        int bytes_left = bytes_in_buffer;
        int frames_this_pass = 0;

        while (bytes_left > 0 && frames_this_pass < FRAMES_PER_PASS) {
            int offset = MP3FindSyncWord(read_ptr, bytes_left);
            if (offset < 0) {
                bytes_left = 0;
                break;
            }
            if (offset > 0) {
                read_ptr += offset;
                bytes_left -= offset;
            }

            uint8_t *saved_ptr = read_ptr;
            int saved_left = bytes_left;

            int ret = MP3Decode(mp3_decoder, &read_ptr, &bytes_left, outbuf, 0);

            if (ret == ERR_MP3_INDATA_UNDERFLOW) {
                break;
            } else if (ret != ERR_MP3_NONE) {
                read_ptr = saved_ptr + 1;
                bytes_left = saved_left - 1;
                frames_skipped++;
                vTaskDelay(pdMS_TO_TICKS(26));
                continue;
            }

            MP3FrameInfo fi;
            MP3GetLastFrameInfo(mp3_decoder, &fi);

            if (fi.samprate == 0 || fi.nChans == 0) {
                read_ptr = saved_ptr + 1;
                bytes_left = saved_left - 1;
                frames_skipped++;
                vTaskDelay(pdMS_TO_TICKS(26));
                continue;
            }

            if (!i2s_started) {
                ESP_LOGI(TAG, "Настройка I2S на %d Гц, %d каналов", fi.samprate, fi.nChans);
                if (i2s_setup(fi.samprate) == ESP_OK) {
                    i2s_started = true;
                } else {
                    break;
                }
            }

            if (fi.nChans == 1) {
                static short stereo_buf[2304];
                for (int i = 0; i < fi.outputSamps; i++) {
                    stereo_buf[i * 2]     = outbuf[i];
                    stereo_buf[i * 2 + 1] = outbuf[i];
                }
                size_t bytes_written = 0;
                i2s_channel_write(tx_chan, stereo_buf, fi.outputSamps * 2 * sizeof(short),
                                  &bytes_written, portMAX_DELAY);
            } else {
                size_t bytes_written = 0;
                i2s_channel_write(tx_chan, outbuf, fi.outputSamps * sizeof(short),
                                  &bytes_written, portMAX_DELAY);
            }
            frames_decoded++;
            frames_this_pass++;
        }

        if (bytes_left > 0 && bytes_left < bytes_in_buffer) {
            memmove(inbuf, read_ptr, bytes_left);
        }
        bytes_in_buffer = bytes_left;
    }

    MP3FreeDecoder(mp3_decoder);
    if (inbuf) heap_caps_free(inbuf);
    if (outbuf) heap_caps_free(outbuf);
    vTaskDelete(NULL);
}

/* ================= ES8388 (Через встроенный I2C от M5Unified) ================= */
static void es8388_write(uint8_t reg, uint8_t val)
{
    // Добавлен 4-й аргумент: частота 100000 Гц (100 кГц)
    // Приведен адрес к uint8_t
    M5.In_I2C.writeRegister8((uint8_t)ES8388_ADDR, reg, val, 100000);
}

static uint8_t es8388_read(uint8_t reg)
{
    // Добавлен 3-й аргумент: частота 100000 Гц
    return M5.In_I2C.readRegister8((uint8_t)ES8388_ADDR, reg, 100000);
}

static void set_trrs_mux(uint8_t mode)
{
    ESP_LOGI(TAG, "Установка MUX: %s", mode ? "CTIA (American)" : "OMTP (National)");
    // Добавлен 4-й аргумент: частота 100000 Гц
    M5.In_I2C.writeRegister8((uint8_t)STM32_ADDR, 0x10, mode, 100000);
}

/* ================= I2S ================= */
static esp_err_t i2s_setup(uint32_t sample_rate)
{
    ESP_LOGI(TAG, "Настройка I2S: %lu Гц", (unsigned long)sample_rate);

    i2s_chan_config_t chan_cfg = I2S_CHANNEL_DEFAULT_CONFIG(I2S_NUM_0, I2S_ROLE_MASTER);
    chan_cfg.auto_clear = true;
    chan_cfg.dma_desc_num = 24;
    chan_cfg.dma_frame_num = 1023;

    esp_err_t err = i2s_new_channel(&chan_cfg, &tx_chan, NULL);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "i2s_new_channel: %s", esp_err_to_name(err));
        return err;
    }

    i2s_std_config_t std_cfg = {};
    std_cfg.clk_cfg = I2S_STD_CLK_DEFAULT_CONFIG(sample_rate);
    std_cfg.slot_cfg = I2S_STD_PHILIP_SLOT_DEFAULT_CONFIG(I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_STEREO);

    std_cfg.gpio_cfg.mclk = I2S_MCLK_PIN;
    std_cfg.gpio_cfg.bclk = I2S_BCK_PIN;
    std_cfg.gpio_cfg.ws   = I2S_WS_PIN;
    std_cfg.gpio_cfg.dout = I2S_DOUT_PIN;
    std_cfg.gpio_cfg.din  = I2S_DIN_PIN;
    std_cfg.gpio_cfg.invert_flags.mclk_inv = false;
    std_cfg.gpio_cfg.invert_flags.bclk_inv = false;
    std_cfg.gpio_cfg.invert_flags.ws_inv   = false;

    std_cfg.clk_cfg.mclk_multiple = I2S_MCLK_MULTIPLE_256;

    err = i2s_channel_init_std_mode(tx_chan, &std_cfg);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "i2s_channel_init_std_mode: %s", esp_err_to_name(err));
        return err;
    }

    err = i2s_channel_enable(tx_chan);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "i2s_channel_enable: %s", esp_err_to_name(err));
        return err;
    }

    ESP_LOGI(TAG, "✅ I2S запущен");
    return ESP_OK;
}

/* ================= WiFi ================= */
static void wifi_event_handler(void *arg, esp_event_base_t event_base,
                               int32_t event_id, void *event_data)
{
    if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_START) {
        esp_wifi_connect();
    } else if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_DISCONNECTED) {
        ESP_LOGW(TAG, "WiFi: дисконнект, переподключаюсь...");
        esp_wifi_connect();
        xEventGroupClearBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
    } else if (event_base == IP_EVENT && event_id == IP_EVENT_STA_GOT_IP) {
        ip_event_got_ip_t *event = (ip_event_got_ip_t *)event_data;
        ESP_LOGI(TAG, "WiFi: получен IP " IPSTR, IP2STR(&event->ip_info.ip));
        xEventGroupSetBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
    }
}

static void wifi_init_sta(void)
{
    s_wifi_event_group = xEventGroupCreate();
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());
    esp_netif_create_default_wifi_sta();

    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));

    esp_event_handler_instance_t instance_any_id;
    esp_event_handler_instance_t instance_got_ip;
    ESP_ERROR_CHECK(esp_event_handler_instance_register(WIFI_EVENT, ESP_EVENT_ANY_ID,
                                                         &wifi_event_handler, NULL, &instance_any_id));
    ESP_ERROR_CHECK(esp_event_handler_instance_register(IP_EVENT, IP_EVENT_STA_GOT_IP,
                                                         &wifi_event_handler, NULL, &instance_got_ip));

    wifi_config_t wifi_config = {};
    strncpy((char*)wifi_config.sta.ssid, WIFI_SSID, sizeof(wifi_config.sta.ssid));
    strncpy((char*)wifi_config.sta.password, WIFI_PASSWORD, sizeof(wifi_config.sta.password));
    wifi_config.sta.threshold.authmode = WIFI_AUTH_WPA2_PSK;

    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_STA));
    ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_STA, &wifi_config));
    ESP_ERROR_CHECK(esp_wifi_start());

    // === ОТКЛЮЧАЕМ POWER SAVE для стабильного потока ===
    ESP_ERROR_CHECK(esp_wifi_set_ps(WIFI_PS_NONE));
    ESP_LOGI(TAG, "WiFi: Power Save отключен");

    ESP_LOGI(TAG, "WiFi: подключение к \"%s\"...", WIFI_SSID);
    xEventGroupWaitBits(s_wifi_event_group, WIFI_CONNECTED_BIT, pdFALSE, pdFALSE, portMAX_DELAY);
}

#include "sd_manager.h"

//#include <M5Unified.h>

#include <esp_log.h>

static const char* TAG = "SD";

static const char* MOUNT_POINT = "/sd";
static const char* RADIO_DIR = "/sd/radio";
static const char* SELFTEST_FILE = "/sd/radio/selftest.txt";

static bool s_mounted = false;
static bool s_own_spi_bus = false;
static sdmmc_card_t* s_card = nullptr;

// Проверенные пины SD для M5Stack CoreS3 SE из старого рабочего диалога.
#define SD_SPI_HOST   SPI2_HOST
#define SD_CS_PIN     GPIO_NUM_4
#define SD_MOSI_PIN   GPIO_NUM_37
#define SD_MISO_PIN   GPIO_NUM_35
#define SD_CLK_PIN    GPIO_NUM_36

// Для стабильности временно используем низкую скорость.
// 20 MHz работал один раз, но для SD 2GB + общий SPI с дисплеем лучше 1 MHz.
#define SD_SPI_MAX_FREQ_KHZ 1000

const char* sd_manager_mount_point(void)
{
    return MOUNT_POINT;
}

bool sd_manager_is_mounted(void)
{
    return s_mounted;
}

uint64_t sd_manager_free_bytes(void)
{
    if (!s_mounted) {
        return 0;
    }

    uint64_t total_bytes = 0;
    uint64_t free_bytes = 0;

    esp_err_t err = esp_vfs_fat_info(MOUNT_POINT, &total_bytes, &free_bytes);
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "esp_vfs_fat_info failed: %s", esp_err_to_name(err));
        return 0;
    }

    return free_bytes;
}

bool sd_manager_ensure_dir(const char* path)
{
    if (!s_mounted || path == nullptr) {
        return false;
    }

    struct stat st;
    if (stat(path, &st) == 0) {
        return S_ISDIR(st.st_mode);
    }

    if (mkdir(path, 0777) == 0) {
        ESP_LOGI(TAG, "Created directory: %s", path);
        return true;
    }

    ESP_LOGE(TAG, "Failed to create directory: %s", path);
    return false;
}

bool sd_manager_write_text(const char* path, const char* text)
{
    if (!s_mounted || path == nullptr || text == nullptr) {
        return false;
    }

    FILE* f = fopen(path, "wb");
    if (f == nullptr) {
        ESP_LOGE(TAG, "Failed to open for write: %s", path);
        return false;
    }

    size_t len = strlen(text);
    size_t written = fwrite(text, 1, len, f);
    fclose(f);

    if (written != len) {
        ESP_LOGE(TAG, "Short write: %s (%u/%u)",
                 path,
                 (unsigned)written,
                 (unsigned)len);
        return false;
    }

    return true;
}

char* sd_manager_read_text(const char* path, size_t* out_len)
{
    if (out_len != nullptr) {
        *out_len = 0;
    }

    if (!s_mounted || path == nullptr) {
        return nullptr;
    }

    FILE* f = fopen(path, "rb");
    if (f == nullptr) {
        ESP_LOGE(TAG, "Failed to open for read: %s", path);
        return nullptr;
    }

    if (fseek(f, 0, SEEK_END) != 0) {
        fclose(f);
        return nullptr;
    }

    long size = ftell(f);
    if (size < 0) {
        fclose(f);
        return nullptr;
    }

    rewind(f);

    char* buf = (char*)malloc((size_t)size + 1);
    if (buf == nullptr) {
        ESP_LOGE(TAG, "OOM for file: %s", path);
        fclose(f);
        return nullptr;
    }

    size_t read_bytes = fread(buf, 1, (size_t)size, f);
    buf[read_bytes] = '\0';
    fclose(f);

    if (out_len != nullptr) {
        *out_len = read_bytes;
    }

    return buf;
}

bool sd_manager_init(void)
{
    if (s_mounted) {
        return true;
    }

    ESP_LOGI(TAG, "Initializing SD card via SPI (CoreS3 SE proven pins)...");
    ESP_LOGI(TAG, "SPI host: %d", (int)SD_SPI_HOST);
    ESP_LOGI(TAG, "Pins: CS=%d MOSI=%d MISO=%d CLK=%d",
             (int)SD_CS_PIN,
             (int)SD_MOSI_PIN,
             (int)SD_MISO_PIN,
             (int)SD_CLK_PIN);
    ESP_LOGI(TAG, "Max SD SPI frequency: %d kHz", SD_SPI_MAX_FREQ_KHZ);

    spi_bus_config_t bus_cfg = {};
    bus_cfg.mosi_io_num = SD_MOSI_PIN;
    bus_cfg.miso_io_num = SD_MISO_PIN;
    bus_cfg.sclk_io_num = SD_CLK_PIN;
    bus_cfg.quadwp_io_num = -1;
    bus_cfg.quadhd_io_num = -1;
    bus_cfg.max_transfer_sz = 4096;

    // M5GFX уже мог инициализировать SPI2 для дисплея.
    // Глушим шум именно на время попытки инициализации шины.
    esp_log_level_set("spi", ESP_LOG_NONE);
    esp_err_t ret = spi_bus_initialize(SD_SPI_HOST, &bus_cfg, SPI_DMA_CH_AUTO);
    esp_log_level_set("spi", ESP_LOG_WARN);

    if (ret == ESP_OK) {
        s_own_spi_bus = true;
        ESP_LOGI(TAG, "SPI bus initialized for SD");
    } else if (ret == ESP_ERR_INVALID_STATE) {
        s_own_spi_bus = false;
        ESP_LOGW(TAG, "SPI bus already initialized. Reusing it for SD.");
        ESP_LOGW(TAG, "WARNING: this assumes M5GFX uses the same MOSI/MISO/CLK pins as SD.");
    } else {
        ESP_LOGE(TAG, "spi_bus_initialize failed: %s", esp_err_to_name(ret));
        s_own_spi_bus = false;
        return false;
    }

    sdmmc_host_t host = SDSPI_HOST_DEFAULT();
    host.slot = SD_SPI_HOST;
    host.max_freq_khz = SD_SPI_MAX_FREQ_KHZ;

    sdspi_device_config_t slot_config = SDSPI_DEVICE_CONFIG_DEFAULT();
    slot_config.gpio_cs = SD_CS_PIN;
    slot_config.host_id = SD_SPI_HOST;

    esp_vfs_fat_sdmmc_mount_config_t mount_config = {};
    mount_config.format_if_mount_failed = false;
    mount_config.max_files = 5;
    mount_config.allocation_unit_size = 16 * 1024;
    mount_config.disk_status_check_enable = false;

    // Для отладки можно временно поставить ESP_LOG_INFO.
    // Потом вернём WARN.
    esp_log_level_set("sdspi_transaction", ESP_LOG_WARN);

    ret = esp_vfs_fat_sdspi_mount(
        MOUNT_POINT,
        &host,
        &slot_config,
        &mount_config,
        &s_card
    );

    if (ret != ESP_OK) {
        if (ret == ESP_FAIL) {
            ESP_LOGW(TAG, "Failed to mount filesystem. "
                          "Ensure SD card is inserted and formatted as FAT32/FAT16.");
        } else if (ret == ESP_ERR_TIMEOUT) {
            ESP_LOGW(TAG, "SD card timeout. Check physical insertion and card format.");
        } else if (ret == ESP_ERR_INVALID_RESPONSE) {
            ESP_LOGW(TAG, "SD card invalid response. Try lower SPI frequency or check pins.");
        } else if (ret == ESP_ERR_INVALID_STATE) {
            ESP_LOGW(TAG, "SD card invalid state. Try full power-cycle of the board.");
        } else {
            ESP_LOGW(TAG, "Failed to initialize SD card: %s", esp_err_to_name(ret));
        }

        s_card = nullptr;
        s_mounted = false;

        // Освобождаем SPI bus только если его инициализировали мы.
        if (s_own_spi_bus) {
            esp_err_t free_ret = spi_bus_free(SD_SPI_HOST);
            if (free_ret != ESP_OK) {
                ESP_LOGW(TAG, "spi_bus_free failed: %s", esp_err_to_name(free_ret));
            }
            s_own_spi_bus = false;
        }

        return false;
    }

    s_mounted = true;

    ESP_LOGI(TAG, "SD mounted at %s", MOUNT_POINT);
    ESP_LOGI(TAG, "Free space: %llu MB",
             (unsigned long long)(sd_manager_free_bytes() / (1024ULL * 1024ULL)));

    if (s_card != nullptr) {
        sdmmc_card_print_info(stdout, s_card);
    }

    if (!sd_manager_ensure_dir(RADIO_DIR)) {
        ESP_LOGW(TAG, "Failed to ensure directory %s", RADIO_DIR);
    }

    return true;
}

bool sd_manager_self_test(void)
{
    if (!s_mounted) {
        ESP_LOGW(TAG, "Self-test skipped: SD not mounted");
        return false;
    }

    if (!sd_manager_ensure_dir(RADIO_DIR)) {
        ESP_LOGE(TAG, "Self-test failed: cannot ensure %s", RADIO_DIR);
        return false;
    }

    const char* payload = "OK\n";

    if (!sd_manager_write_text(SELFTEST_FILE, payload)) {
        ESP_LOGE(TAG, "Self-test failed: write error");
        return false;
    }

    size_t len = 0;
    char* readback = sd_manager_read_text(SELFTEST_FILE, &len);

    bool ok = (readback != nullptr) &&
              (len == strlen(payload)) &&
              (strcmp(readback, payload) == 0);

    if (readback != nullptr) {
        free(readback);
    }

    if (remove(SELFTEST_FILE) != 0) {
        ESP_LOGW(TAG, "Self-test warning: failed to remove %s", SELFTEST_FILE);
    }

    ESP_LOGI(TAG, "Self-test: %s", ok ? "OK" : "FAIL");
    return ok;
}

void storage_save_wifi(const char* ssid, const char* pass) {
    nvs_handle_t handle;
    esp_err_t err = nvs_open(NVS_NAMESPACE_WIFI, NVS_READWRITE, &handle);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Failed to open NVS namespace '%s': %s", NVS_NAMESPACE_WIFI, esp_err_to_name(err));
        return;
    }

    err = nvs_set_str(handle, NVS_KEY_SSID, ssid);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Failed to save SSID: %s", esp_err_to_name(err));
    }

    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Failed to save password: %s", esp_err_to_name(err));
    }

    err = nvs_commit(handle);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Failed to commit NVS: %s", esp_err_to_name(err));
    }

    nvs_close(handle);
    ESP_LOGI(TAG, "Wi-Fi credentials saved: SSID='%s'", ssid);
}

        return false;

        ESP_LOGW(TAG, "Failed to load password: %s", esp_err_to_name(err));

        return false;

// ============================================================
// Чтение HMI
// ============================================================
static int32_t read_hmi_register_32(uint8_t reg) {
    uint8_t buf[4] = {0};
    if (!M5.In_I2C.readRegister(HMI_I2C_ADDR, reg, buf, 4, 100000)) return 0;
    return (int32_t)(buf[0] | (buf[1] << 8) | (buf[2] << 16) | (buf[3] << 24));
}

static bool read_hmi_register_8(uint8_t reg) {
    uint8_t val = 0;
    if (!M5.In_I2C.readRegister(HMI_I2C_ADDR, reg, &val, 1, 100000)) return false;
    return val == 0;
}

static bool is_cat_marked(size_t idx) {
    for (size_t i = 0; i < s_marked_cat_count; i++) {
        if (s_marked_cat_indices[i] == idx) return true;
    }
    return false;
}

static void toggle_cat_mark(size_t idx) {
    for (size_t i = 0; i < s_marked_cat_count; i++) {
        if (s_marked_cat_indices[i] == idx) {
            for (size_t j = i; j < s_marked_cat_count - 1; j++) {
                s_marked_cat_indices[j] = s_marked_cat_indices[j + 1];
            }
            s_marked_cat_count--;
            return;
        }
    }
    if (s_marked_cat_count < 64) {
        s_marked_cat_indices[s_marked_cat_count++] = idx;
    }
}

    s_marked_cat_count = 0;
}

    // Снимаем отметку, если есть

            for (size_t j = i; j < s_marked_count - 1; j++) {

        s_marked_ids[s_marked_count++] = id;
    }
}

// ============================================================
// Статус-бар: WiFi / время / батарея
// ============================================================
template <typename T>
static void draw_status_bar_impl(T& lcd) {
    lcd.setFont(&fonts::Font0);

    // ---------- WiFi RSSI слева ----------
    lcd.setTextDatum(TL_DATUM);

    if (wifi_is_connected()) {
        int rssi = wifi_get_rssi();
        char buf[24];
        snprintf(buf, sizeof(buf), "WiFi %ddBm", rssi);

        if (rssi >= -50) {
            lcd.setTextColor(0x07E0);       // отличный сигнал
        } else if (rssi >= -65) {
            lcd.setTextColor(TFT_YELLOW);   // средний
        } else {
            lcd.setTextColor(TFT_RED);      // слабый
        }

        lcd.drawString(buf, 2, 4);
    } else {
        lcd.setTextColor(TFT_RED);
        lcd.drawString("WiFi --", 2, 4);
    }

    // ---------- Время по центру ----------
    // ВАЖНО: при MC_DATUM y — это центр текста, поэтому ставим y=8
    lcd.setTextDatum(MC_DATUM);

    time_t now = time(nullptr);

    // Время считается синхронизированным, если timestamp разумный.
    // 1700000000 ~= ноябрь 2023.
    if (now > 1700000000) {
        struct tm tm_info;
        localtime_r(&now, &tm_info);

        char buf[24];
        strftime(buf, sizeof(buf), "%H:%M %d.%m", &tm_info);

        lcd.setTextColor(TFT_WHITE);
        lcd.drawString(buf, 160, 8);
    } else {
        lcd.setTextColor(TFT_LIGHTGRAY);
        lcd.drawString("--:-- --.--", 160, 8);
    }

    // ---------- Батарея / USB справа ----------
    bool charging = M5.Power.isCharging();
    int bat = (int)M5.Power.getBatteryLevel();
    int mv = (int)M5.Power.getBatteryVoltage();

    // Диагностика в лог каждые ~5 секунд
    static uint32_t power_dbg_counter = 0;
    if (++power_dbg_counter >= 100) {
        power_dbg_counter = 0;
        ESP_LOGI(TAG, "PWR: charging=%d level=%d voltage=%dmV",
                 (int)charging, bat, mv);
    }

    lcd.setTextDatum(TR_DATUM);

    if (charging) {
        lcd.setTextColor(TFT_YELLOW);
        lcd.drawString("USB", 318, 4);
    }
    else if (mv >= 2800 && mv <= 4600) {
        // Li-Po/Li-Ion примерная карта:
        // 3.00V ~ 0%
        // 4.20V ~ 100%
        int pct = ((mv - 3000) * 100) / 1200;

        if (pct < 0) pct = 0;
        if (pct > 100) pct = 100;

        char buf[16];
        snprintf(buf, sizeof(buf), "%d%%", pct);

        if (pct > 50) {
            lcd.setTextColor(0x07E0);
        } else if (pct > 20) {
            lcd.setTextColor(TFT_YELLOW);
        } else {
            lcd.setTextColor(TFT_RED);
        }

        lcd.drawString(buf, 318, 4);
    }
    else if (bat >= 0 && bat < 100) {
        char buf[16];
        snprintf(buf, sizeof(buf), "%d%%", bat);

        if (bat > 50) {
            lcd.setTextColor(0x07E0);
        } else if (bat > 20) {
            lcd.setTextColor(TFT_YELLOW);
        } else {
            lcd.setTextColor(TFT_RED);
        }

        lcd.drawString(buf, 318, 4);
    }
    else {
        lcd.setTextColor(TFT_LIGHTGRAY);
        lcd.drawString("--%", 318, 4);
    }

    lcd.setTextDatum(TL_DATUM);
}

// ============================================================
// Мини-график сетевого буфера
// ============================================================
template <typename T>
static void draw_net_sparkline_impl(T& lcd, int x, int y, int w, int h) {
    if (s_net_history_count == 0 || s_net_capacity == 0) {
        return;
    }

    lcd.drawRect(x, y, w, h, TFT_DARKGREY);

    const int total_slots = 64;
    int step = w / total_slots;

    if (step < 1) {
        step = 1;
    }

    int samples = s_net_history_count;
    int start = (s_net_history_pos + total_slots - samples) % total_slots;

    for (int i = 0; i < samples; i++) {
        int idx = (start + i) % total_slots;
        uint32_t val = s_net_history[idx];

        int bar_h = (int)(((uint64_t)val * (uint64_t)(h - 2)) / (uint64_t)s_net_capacity);

        if (bar_h < 1 && val > 0) {
            bar_h = 1;
        }

        int bx = x + 1 + i * step;
        int bw = step - 1;

        if (bw < 1) {
            bw = 1;
        }

        if (bx + bw > x + w - 1) {
            break;
        }

        uint16_t color;

        if (val == 0) {
            color = TFT_RED;
        } else if (((uint64_t)val * 100ULL) / (uint64_t)s_net_capacity < 10ULL) {
            color = TFT_YELLOW;
        } else {
            color = 0x07E0;
        }

        lcd.fillRect(bx, y + h - 1 - bar_h, bw, bar_h, color);
    }
}

// ============================================================
// Wi-Fi manual input / on-screen keyboard helpers
// ============================================================

static bool s_keyboard_shift = false;

static int wifi_scan_view_start(void) {
    int start = s_wifi_scan_selected_idx;

    if (start > 2) {
        start = s_wifi_scan_selected_idx - 2;
    }

    if (start < 0) {
        start = 0;
    }

    return start;
}

static void wifi_kb_append_char(char c) {
    char* buf = (s_keyboard_target == KB_TARGET_PASS) ? s_wifi_pass : s_wifi_ssid;

    size_t max_len = (s_keyboard_target == KB_TARGET_PASS)
                         ? (sizeof(s_wifi_pass) - 1)
                         : (sizeof(s_wifi_ssid) - 1);

    size_t len = strlen(buf);

    if (len < max_len) {
        buf[len] = c;
        buf[len + 1] = '\0';
    }
}

static void wifi_kb_backspace(void) {
    char* buf = (s_keyboard_target == KB_TARGET_PASS) ? s_wifi_pass : s_wifi_ssid;
    size_t len = strlen(buf);

    if (len > 0) {
        buf[len - 1] = '\0';
    }
}

static void wifi_kb_close(void) {
    s_keyboard_active = false;
    s_keyboard_target = KB_TARGET_NONE;
    s_keyboard_shift = false;
}

static void wifi_kb_open(kb_target_t target) {
    s_keyboard_active = true;
    s_keyboard_target = target;
    s_keyboard_shift = false;

    // Если был открыт список скана — закрываем его
    s_wifi_show_scan_results = false;
    s_wifi_scan_selected_idx = -1;

    if (target == KB_TARGET_SSID) {
        s_wifi_focus = 0;
    } else {
        s_wifi_focus = 1;
    }

    ESP_LOGI(TAG, "Keyboard opened for target=%d", (int)target);
}

static void wifi_back_action(void) {
    if (s_keyboard_active) {
        wifi_kb_close();
        ESP_LOGI(TAG, "Keyboard closed by back action");
        return;
    }

    if (s_wifi_show_scan_results) {
        s_wifi_show_scan_results = false;
        s_wifi_scan_selected_idx = -1;
        s_wifi_focus = 0;
        s_keyboard_shift = false;
        ESP_LOGI(TAG, "Scan results closed by back action");
        return;
    }

    s_wifi_focus = 0;
    s_keyboard_shift = false;
    s_current_mode = MODE_SETTINGS;
    ESP_LOGI(TAG, "Exited Wi-Fi mode to SETTINGS");
}

static void wifi_select_scan_result(int idx) {
    int count = wifi_scan_get_count();

    if (count == 0) {
        s_wifi_show_scan_results = false;
        s_wifi_scan_selected_idx = -1;
        wifi_kb_open(KB_TARGET_SSID);
        return;
    }

    if (idx < 0 || idx >= count) {
        idx = 0;
    }

    const wifi_scan_record_t* rec = wifi_scan_get_record(idx);

    if (rec != nullptr) {
        strncpy(s_wifi_ssid, rec->ssid, sizeof(s_wifi_ssid) - 1);
        s_wifi_ssid[sizeof(s_wifi_ssid) - 1] = '\0';

        ESP_LOGI(TAG, "Wi-Fi SSID selected from scan: %s", s_wifi_ssid);
    }

    s_wifi_show_scan_results = false;
    s_wifi_scan_selected_idx = -1;

    // Сразу открываем ввод пароля
    wifi_kb_open(KB_TARGET_PASS);
}

static void handle_wifi_touch(int32_t x, int32_t y) {
    ESP_LOGI(TAG, "handle_wifi_touch x=%d y=%d scan=%d kb=%d focus=%d",
             (int)x, (int)y,
             (int)s_wifi_show_scan_results,
             (int)s_keyboard_active,
             s_wifi_focus);

    // ------------------------------------------------------------
    // 1. Если показан список найденных сетей — работаем только с ним
    // ------------------------------------------------------------
    if (s_wifi_show_scan_results) {
        int count = wifi_scan_get_count();

        if (count == 0) {
            // Нет сетей — тап по нижней области открывает ручной ввод SSID
            if (y >= 76) {
                s_wifi_show_scan_results = false;
                s_wifi_scan_selected_idx = -1;
                wifi_kb_open(KB_TARGET_SSID);
            }
            return;
        }

        const int list_y = 76;
        const int item_h = 26;
        const int visible = 5;

        int start = wifi_scan_view_start();

        if (y >= list_y && y < list_y + visible * item_h) {
            int idx = start + (y - list_y) / item_h;

            if (idx >= 0 && idx < count) {
                wifi_select_scan_result(idx);
            }
        }

        return;
    }

    // ------------------------------------------------------------
    // 2. Тап по полям SSID / Password
    //    Работает и когда клавиатура открыта, и когда закрыта.
    // ------------------------------------------------------------
    for (int i = 0; i < 2; i++) {
        int fy = WIFI_FIELD_Y[i] - 2;

        if (x >= 5 && x < 315 && y >= fy && y < fy + WIFI_FIELD_H + 4) {
            wifi_kb_open(i == 0 ? KB_TARGET_SSID : KB_TARGET_PASS);
            return;
        }
    }

    // ------------------------------------------------------------
    // 3. Если клавиатура открыта — обрабатываем тапы по клавиатуре
    // ------------------------------------------------------------
    if (s_keyboard_active) {
        if (y >= WIFI_KB_Y && y < WIFI_KB_Y + WIFI_KB_ROWS * WIFI_KB_H) {
            int row = (y - WIFI_KB_Y) / WIFI_KB_H;

            // Буквенные/цифровые ряды
            if (row >= 0 && row < 4) {
                if (x >= WIFI_KB_X && x < WIFI_KB_X + 10 * WIFI_KB_STEP) {
                    int col = (x - WIFI_KB_X) / WIFI_KB_STEP;

                    const char* row_str = WIFI_KB_LETTER_ROWS[row];
                    int len = (int)strlen(row_str);

                    if (col >= 0 && col < len) {
                        char c = row_str[col];

                        if (s_keyboard_shift && c >= 'a' && c <= 'z') {
                            c = (char)(c - 'a' + 'A');
                        }

                        wifi_kb_append_char(c);

                        // После ввода символа Shift автоматически выключается
                        s_keyboard_shift = false;

                        ESP_LOGI(TAG, "Keyboard char: '%c' target=%d",
                                 c, (int)s_keyboard_target);
                    }
                }

                return;
            }

            // Нижний ряд действий: Shift / Space / Del / OK
            if (row == 4) {
                int ay = WIFI_KB_Y + 4 * WIFI_KB_H;

                if (y >= ay && y < ay + WIFI_KB_H - 2) {
                    for (int i = 0; i < 4; i++) {
                        int ax = WIFI_KB_ACTION_X[i];
                        int aw = WIFI_KB_ACTION_W[i];

                        if (x >= ax && x < ax + aw) {
                            switch (i) {
                                case 0: // Shift
                                    s_keyboard_shift = !s_keyboard_shift;
                                    ESP_LOGI(TAG, "Keyboard shift=%d", (int)s_keyboard_shift);
                                    break;

                                case 1: // Space
                                    wifi_kb_append_char(' ');
                                    ESP_LOGI(TAG, "Keyboard space");
                                    break;

                                case 2: // Del
                                    wifi_kb_backspace();
                                    ESP_LOGI(TAG, "Keyboard backspace");
                                    break;

                                case 3: // OK
                                    wifi_kb_close();
                                    ESP_LOGI(TAG, "Keyboard OK/close");
                                    break;
                            }

                            return;
                        }
                    }
                }
            }
        }

        return;
    }

    // ------------------------------------------------------------
    // 4. Если клавиатура закрыта — тапы по кнопкам Scan / Connect
    // ------------------------------------------------------------
    int by = WIFI_BTN_Y - 2;

    if (y >= by && y < by + WIFI_BTN_H + 4) {
        if (x >= 5 && x < 155) {
            ESP_LOGI(TAG, "Touch Scan pressed");
            ui_trigger_wifi_scan();
            return;
        }

        if (x >= 165 && x < 315) {
            ESP_LOGI(TAG, "Touch Connect pressed");
            ui_trigger_wifi_connect();
            return;
        }
    }
}
template <typename T>
static void draw_wifi_keyboard_impl(T& lcd) {
    lcd.setFont(&fonts::Font0);

    // Буквенные и цифровые ряды
    for (int row = 0; row < 4; row++) {
        const char* row_str = WIFI_KB_LETTER_ROWS[row];
        int len = (int)strlen(row_str);
        int y = WIFI_KB_Y + row * WIFI_KB_H;

        for (int col = 0; col < 10; col++) {
            if (col >= len) {
                continue;
            }

            char ch = row_str[col];

            if (s_keyboard_shift && ch >= 'a' && ch <= 'z') {
                ch = (char)(ch - 'a' + 'A');
            }

            int x = WIFI_KB_X + col * WIFI_KB_STEP;

            lcd.fillRoundRect(x, y, WIFI_KB_W, WIFI_KB_H - 2, 3, TFT_BLACK);
            lcd.drawRoundRect(x, y, WIFI_KB_W, WIFI_KB_H - 2, 3, 0x07E0);

            char buf[2] = {ch, '\0'};

            lcd.setTextColor(TFT_WHITE);
            lcd.setTextDatum(MC_DATUM);
            lcd.drawString(buf, x + WIFI_KB_W / 2, y + (WIFI_KB_H - 2) / 2);
        }
    }

    // Нижний ряд: Shift / Space / Del / OK
    int ay = WIFI_KB_Y + 4 * WIFI_KB_H;

    for (int i = 0; i < 4; i++) {
        bool active = (i == 0 && s_keyboard_shift);

        uint16_t bg = active ? 0x07E0 : TFT_BLACK;
        uint16_t fg = active ? TFT_BLACK : 0x07E0;

        int x = WIFI_KB_ACTION_X[i];
        int w = WIFI_KB_ACTION_W[i];

        lcd.fillRoundRect(x, ay, w, WIFI_KB_H - 2, 3, bg);
        lcd.drawRoundRect(x, ay, w, WIFI_KB_H - 2, 3, 0x07E0);

        lcd.setTextColor(fg);
        lcd.setTextDatum(MC_DATUM);
        lcd.drawString(WIFI_KB_ACTION_LABEL[i], x + w / 2, ay + (WIFI_KB_H - 2) / 2);
    }

    lcd.setTextDatum(TL_DATUM);
}

template <typename T>
static void about_row(T& lcd, int y, const char* label, const char* value,
                      uint16_t color = ABOUT_VALUE) {
    lcd.setFont(&fonts::efontCN_14);
    // подпись слева, тускло
    lcd.setTextDatum(TL_DATUM);
    lcd.setTextColor(ABOUT_LABEL);
    lcd.drawString(label, 18, y);
    // значение справа, акцентным цветом
    lcd.setTextColor(color);
    lcd.setTextDatum(TR_DATUM);
    lcd.drawString(value, 306, y);
    lcd.setTextDatum(TL_DATUM);
}
// ============================================================
// Отрисовка UI (template)
// ============================================================
template <typename T>
static void draw_ui_impl(T& lcd) {
    lcd.fillScreen(TFT_BLACK);

    // Верхний статус-бар
    draw_status_bar_impl(lcd);


    const char* modes[] = {"PLAY", "FAV", "CAT", "STA", "SET"};
    int menu_y = 18;
    int menu_h = 22;

    int active_tab = (int)s_current_mode;

    // Wi-Fi, About и Catalog — это суб-меню настроек,
    // поэтому визуально подсвечиваем вкладку SET.
    if (active_tab == (int)MODE_WIFI ||
        active_tab == (int)MODE_ABOUT ||
        active_tab == (int)MODE_CATALOG) {
        active_tab = (int)MODE_SETTINGS;
    }

    for (int i = 0; i < 5; i++) {
        if (i == active_tab) {
            lcd.fillRect(i * 64, menu_y, 64, menu_h, 0x07E0);
            lcd.setTextColor(TFT_BLACK);
        } else {
            lcd.fillRect(i * 64, menu_y, 64, menu_h, TFT_BLACK);
            lcd.setTextColor(TFT_LIGHTGRAY);
        }

        lcd.setFont(&fonts::Font0);
        lcd.setTextDatum(MC_DATUM);
        lcd.drawString(modes[i], i * 64 + 32, menu_y + menu_h / 2 + 1);
    }
    lcd.setTextDatum(TL_DATUM);

    if (s_is_muted) {
        lcd.setFont(&fonts::Font0);
        lcd.setTextColor(TFT_RED);
        lcd.setTextDatum(TR_DATUM);
        lcd.drawString("MUTE", 318, menu_y + 4);
        lcd.setTextDatum(TL_DATUM);
    }

    // ========================================================
    // Wi-Fi Settings
    // ========================================================
    // ========================================================
    // Wi-Fi Settings & Known Networks
    // ========================================================
    if (s_current_mode == MODE_WIFI) {
        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(0x07E0);
        if (s_wifi_show_scan_results) lcd.drawString("Найденные сети", 10, 42);
        else if (s_keyboard_active) lcd.drawString("Ввод Wi-Fi", 10, 42);
        else lcd.drawString("Настройка Wi-Fi", 10, 42);

        if (s_wifi_show_scan_results) {
            lcd.setFont(&fonts::efontCN_14);
            lcd.setTextColor(TFT_LIGHTGRAY);
            lcd.drawString("Тап/энкодер - выбрать, A - назад", 10, 58);
            int count = wifi_scan_get_count();
            if (count == 0) {
                lcd.setTextColor(TFT_YELLOW);
                lcd.drawString("Сети не найдены", 15, 90);
                lcd.drawString("Тапни или нажми A и введи SSID вручную", 15, 112);
            } else {
                const int list_y = 76;
                const int item_h = 26;
                const int visible = 5;
                int start = wifi_scan_view_start();
                int y = list_y;
                for (int i = start; i < count && i < start + visible; i++) {
                    const wifi_scan_record_t* rec = wifi_scan_get_record(i);
                    if (rec == nullptr) continue;
                    bool selected = (i == s_wifi_scan_selected_idx);
                    if (selected) {
                        lcd.setTextColor(TFT_BLACK);
                        lcd.fillRoundRect(5, y, 310, item_h - 2, 4, 0x07E0);
                    } else {
                        lcd.setTextColor(TFT_WHITE);
                        lcd.fillRoundRect(5, y, 310, item_h - 2, 4, TFT_BLACK);
                        lcd.drawRoundRect(5, y, 310, item_h - 2, 4, 0x07E0);
                    }
                    char line[64];
                    snprintf(line, sizeof(line), "%s (%d)", rec->ssid, (int)rec->rssi);
                    lcd.setFont(&fonts::efontCN_14);
                    lcd.drawString(line, 12, y + 4);
                    y += item_h;
                }
            }
        } else {
            const char* labels[2] = {"SSID:", "Пароль:"};
            const char* values[2] = {s_wifi_ssid, s_wifi_pass};
            for (int i = 0; i < 2; i++) {
                bool focused = s_keyboard_active ? (s_keyboard_target == (i == 0 ? KB_TARGET_SSID : KB_TARGET_PASS)) : (s_wifi_focus == i);
                int fy = WIFI_FIELD_Y[i] - 2;
                if (focused) {
                    lcd.setTextColor(TFT_BLACK);
                    lcd.fillRoundRect(5, fy, 310, WIFI_FIELD_H, 4, 0x07E0);
                } else {
                    lcd.setTextColor(TFT_WHITE);
                    lcd.fillRoundRect(5, fy, 310, WIFI_FIELD_H, 4, TFT_BLACK);
                    lcd.drawRoundRect(5, fy, 310, WIFI_FIELD_H, 4, TFT_LIGHTGRAY);
                }
                char disp[72]; disp[0] = '\0';
                if (i == 1 && !focused) {
                    int n = (int)strlen(values[i]);
                    if (n > 30) n = 30;
                    for (int k = 0; k < n; k++) disp[k] = '*';
                    disp[n] = '\0';
                } else {
                    strncpy(disp, values[i], sizeof(disp) - 1);
                    disp[sizeof(disp) - 1] = '\0';
                }
                char line[96];
                snprintf(line, sizeof(line), "%s %s", labels[i], disp);
                lcd.setFont(&fonts::efontCN_14);
                lcd.setTextColor(focused ? TFT_BLACK : TFT_WHITE);
                lcd.drawString(line, 10, WIFI_FIELD_Y[i] + 2);

                bool cursor_here = false;
                if (s_keyboard_active) {
                    if (i == 0 && s_keyboard_target == KB_TARGET_SSID) cursor_here = true;
                    if (i == 1 && s_keyboard_target == KB_TARGET_PASS) cursor_here = true;
                }
                if (cursor_here && s_cursor_blink) {
                    int text_w = lcd.textWidth(line);
                    int cursor_x = 10 + text_w + 2;
                    if (cursor_x > 306) cursor_x = 306;
                    int cursor_y = WIFI_FIELD_Y[i] + 3;
                    int cursor_h = WIFI_FIELD_H - 8;
                    uint16_t cursor_color = focused ? TFT_BLACK : TFT_WHITE;
                    lcd.fillRect(cursor_x, cursor_y, 2, cursor_h, cursor_color);
                }
            }

            if (!s_keyboard_active) {
                const char* btn_labels[4] = {"Scan", "Connect", "Networks", "Back"};
                const int btn_x[4] = {5, 165, 5, 165};
                const int btn_y[4] = {WIFI_BTN_Y, WIFI_BTN_Y, WIFI_BTN_ROW2_Y, WIFI_BTN_ROW2_Y};

                for (int i = 0; i < 4; i++) {
                    int focus_idx = i + 2; // 2=Scan, 3=Connect, 4=Networks, 5=Back
                    bool focused = (s_wifi_focus == focus_idx);
                    int bx = btn_x[i];
                    int by = btn_y[i];

                    if (focused) {
                        lcd.setTextColor(TFT_BLACK);
                        lcd.fillRoundRect(bx, by, WIFI_BTN_W, WIFI_BTN_H, 4, 0x07E0);
                    } else {
                        lcd.setTextColor(0x07E0);
                        lcd.fillRoundRect(bx, by, WIFI_BTN_W, WIFI_BTN_H, 4, TFT_BLACK);
                        lcd.drawRoundRect(bx, by, WIFI_BTN_W, WIFI_BTN_H, 4, 0x07E0);
                    }
                    lcd.setFont(&fonts::efontCN_14);
                    lcd.setTextDatum(MC_DATUM);
                    lcd.drawString(btn_labels[i], bx + WIFI_BTN_W / 2, by + WIFI_BTN_H / 2 + 1);
                }
                lcd.setTextDatum(TL_DATUM);
            } else {
                draw_wifi_keyboard_impl(lcd);
            }
        }
    }
    // ========================================================
    // Known Networks Management
    // ========================================================
    else if (s_current_mode == MODE_KNOWN_NETWORKS) {
        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(0x07E0);
        lcd.drawString("Управление сетями", 10, 48);

        int count = wifi_known_networks_get_count();
        if (count == 0) {
            lcd.setTextColor(TFT_LIGHTGRAY);
            lcd.drawString("Нет сохраненных сетей", 15, 80);
            lcd.drawString("Подключитесь к Wi-Fi, чтобы", 15, 100);
            lcd.drawString("добавить его в список", 15, 120);
        } else {
            int visible = 5;
            int start = s_settings_idx;
            if (start > 2) start = s_settings_idx - 2;
            if (start < 0) start = 0;

            int y = 68;
            for (int i = start; i < count && i < start + visible; i++) {
                const known_network_t* net = wifi_known_networks_get(i);
                if (!net) continue;

                if (i == s_settings_idx) {
                    lcd.setTextColor(TFT_BLACK);
                    lcd.fillRoundRect(5, y - 5, 310, 26, 4, 0x07E0);
                } else {
                    lcd.setTextColor(TFT_WHITE);
                }

                char line[64];
                snprintf(line, sizeof(line), "%s %s", net->auto_connect ? "[A]" : "[ ]", net->ssid);
                lcd.drawString(line, 15, y);
                y += 32;
            }

            // ИСПРАВЛЕНИЕ: используем efontCN_14 вместо Font0 для поддержки кириллицы
            lcd.setTextColor(TFT_LIGHTGRAY);
            lcd.setFont(&fonts::efontCN_14);
            lcd.drawString("A: Удалить сеть | B: Назад", 10, 216);
        }
    }

    // ========================================================
    // Favorites (Реальный список)
    // ========================================================
    if (s_current_mode == MODE_FAVORITES) {
        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(0x07E0);
        lcd.drawString("Избранное", 10, 48);

        int visible_items = 6;
        int fav_count = (int)catalog_get_favorites_count();
        int start_idx = s_fav_station_idx;
        if (start_idx > 3) {
            start_idx = s_fav_station_idx - 3;
        }

        int y = 68;
        if (fav_count == 0) {
            lcd.setTextColor(TFT_LIGHTGRAY);
            lcd.setFont(&fonts::efontCN_14);
            lcd.drawString("Список пуст", 15, y);
            lcd.drawString("Долгое нажатие в STA", 15, y + 24);
            lcd.drawString("добавит станцию", 15, y + 48);
        } else {
            for (int i = start_idx; i < fav_count && i < start_idx + visible_items; i++) {
                int global_idx = catalog_get_favorite_global_index(i);
                const catalog_station_t* st = catalog_get_station_by_global_index(global_idx);
                if (st == nullptr) continue; // Станция могла быть удалена из каталога

                if (i == s_fav_station_idx) {
                    lcd.setTextColor(TFT_BLACK);
                    lcd.fillRoundRect(5, y - 5, 310, 24, 4, 0x07E0);
                } else {
                    lcd.setTextColor(TFT_WHITE);
                }
                lcd.setFont(&fonts::efontCN_16);
                lcd.drawString(st->name, 15, y);
                y += 28;
            }
        }
    }
    // ========================================================
    // Settings Menu
    // ========================================================
    else if (s_current_mode == MODE_SETTINGS) {
        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(0x07E0);
        lcd.drawString("Настройки", 10, 48);

        const char* items[] = {"1. Wi-Fi Setup", "2. Catalog", "3. About", "4. Back"};
        int count = 4;
        int y = 68;

        for (int i = 0; i < count; i++) {
            if (i == s_settings_idx) {
                lcd.setTextColor(TFT_BLACK);
                lcd.fillRoundRect(5, y - 5, 310, 26, 4, 0x07E0);
            } else {
                lcd.setTextColor(TFT_WHITE);
            }

            // Для пункта "2. Catalog" показываем прогресс валидации
            if (i == 1 && catalog_is_validating()) {
                int done = catalog_get_validated_count();
                int total = catalog_get_staging_count();
                int pct = (total > 0) ? (done * 100 / total) : 0;
                char line[80];
                snprintf(line, sizeof(line), "Catalog - %d/%d (%d%%)", done, total, pct);
                lcd.drawString(line, 15, y);
            } else {
                lcd.drawString(items[i], 15, y);
            }

            y += 32;
        }
    }

    // ========================================================
    // About Screen (Без заголовка, компактная вёрстка)
    // ========================================================
    else if (s_current_mode == MODE_ABOUT) {
        char line[64];

        // ---- СИСТЕМА ----
        // Начинаем сразу под верхними вкладками (y=40)
        about_section(lcd, 42, "СИСТЕМА");

        const esp_app_desc_t* app = esp_app_get_description();
        about_row(lcd, 59, "FW", app->version, ABOUT_ACCENT);

        snprintf(line, sizeof(line), "%s %s", app->date, app->time);
        about_row(lcd, 74, "Build", line);

        uint64_t up_s = (uint64_t)(esp_timer_get_time() / 1000000);
        uint32_t uh = (uint32_t)(up_s / 3600);
        uint32_t um = (uint32_t)((up_s / 60) % 60);
        uint32_t us = (uint32_t)(up_s % 60);
        if (uh > 0) {
            snprintf(line, sizeof(line), "%uh %um %us", (unsigned int)uh, (unsigned int)um, (unsigned int)us);
        } else {
            snprintf(line, sizeof(line), "%um %us", (unsigned int)um, (unsigned int)us);
        }
        about_row(lcd, 89, "Аптайм", line);

        int  mv  = (int)M5.Power.getBatteryVoltage();
        int  pct = (int)M5.Power.getBatteryLevel();
        bool chg = M5.Power.isCharging();
        if (chg) {
            snprintf(line, sizeof(line), "%dmV %d%% [USB]", mv, pct);
        } else {
            snprintf(line, sizeof(line), "%dmV %d%%", mv, pct);
        }
        uint16_t bat_col = (pct > 50) ? ABOUT_ACCENT : (pct > 20 ? TFT_YELLOW : TFT_RED);
        about_row(lcd, 104, "Батарея", line, bat_col);

        // ---- СЕТЬ ----
        // Отступ 16 пикселей от последней строки
        about_section(lcd, 122, "СЕТЬ");

        uint8_t mac[6];
        esp_read_mac(mac, ESP_MAC_WIFI_STA);
        snprintf(line, sizeof(line), "%02X:%02X:%02X:%02X:%02X:%02X",
                 mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
        about_row(lcd, 139, "MAC", line);

        if (wifi_is_connected()) {
            esp_netif_t* netif = esp_netif_get_handle_from_ifkey("WIFI_STA_DEF");
            esp_netif_ip_info_t ip_info;
            if (netif != nullptr && esp_netif_get_ip_info(netif, &ip_info) == ESP_OK) {
                snprintf(line, sizeof(line), IPSTR, IP2STR(&ip_info.ip));
            } else {
                snprintf(line, sizeof(line), "нет IP");
            }
            about_row(lcd, 154, "IP", line, ABOUT_ACCENT);

            int rssi = wifi_get_rssi();
            snprintf(line, sizeof(line), "%d dBm", rssi);
            uint16_t rssi_col = (rssi >= -60) ? ABOUT_ACCENT
                                : (rssi >= -75) ? TFT_YELLOW : TFT_RED;
            about_row(lcd, 169, "RSSI", line, rssi_col);
        } else {
            about_row(lcd, 154, "IP",   "не подключено", TFT_RED);
            about_row(lcd, 169, "RSSI", "--", TFT_LIGHTGRAY);
        }

        // ---- КАТАЛОГ ----
        // Отступ 14 пикселей
        about_section(lcd, 187, "КАТАЛОГ");

        const char* src = catalog_get_source_str();
        int total_st = (int)catalog_total_stations();
        int fav_count = (int)catalog_get_favorites_count();

        // Объединяем в одну строку, чтобы точно влезло
        snprintf(line, sizeof(line), "%s (%d) | Избр: %d", src, total_st, fav_count);
        about_row(lcd, 204, "Каталог", line, ABOUT_ACCENT);

        // Итоговая нижняя точка: 204 + 14 (высота шрифта) = 218 пикселей.
        // До края экрана (240) остаётся 22 пикселя запаса!
    }

    // ========================================================
    // Catalog Management (Финальная версия с прогресс-баром)
    // ========================================================
    else if (s_current_mode == MODE_CATALOG) {
        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(0x07E0);
        lcd.drawString("Управление каталогом", 10, 48);

        // Если идет валидация, показываем прогресс-бар вместо меню
        if (catalog_is_validating()) {
            int done = catalog_get_validated_count();
            int total = catalog_get_staging_count();
            int pct = (total > 0) ? (done * 100 / total) : 0;

            lcd.setTextColor(TFT_WHITE);
            lcd.drawString("Проверка живых станций...", 15, 80);

            lcd.drawRoundRect(15, 110, 290, 24, 6, TFT_DARKGREY);
            int bar_w = (290 * pct) / 100;
            if (bar_w > 0) {
                lcd.fillRoundRect(15, 110, bar_w, 24, 6, 0x07E0);
            }

            char buf[48];
            snprintf(buf, sizeof(buf), "%d / %d (%d%%)", done, total, pct);
            lcd.setTextDatum(MC_DATUM);
            lcd.setTextColor(TFT_WHITE); // Белый текст для читаемости
            lcd.drawString(buf, 160, 122);
            lcd.setTextDatum(TL_DATUM);

            lcd.setTextColor(TFT_LIGHTGRAY);
            lcd.drawString("Можно слушать радио...", 15, 150);
            return; // Выходим, не рисуем меню поверх
        }

        // Обычное меню (когда валидация не идет)
        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(TFT_WHITE);

        int staging_count = catalog_staging_get_count();
        char line[64];
        snprintf(line, sizeof(line), "В буфере: %d станций", staging_count);
        lcd.drawString(line, 15, 76);

        int y = 100;
        const char* items[] = {
            "1. Загрузить РАБОЧИЙ каталог",
            "2. Импортировать M3U с SD",
            "3. Проверить живые (Validate)",
            "4. Применить изменения (Merge)",
            "5. Назад"
        };
        int count = 5;

        for (int i = 0; i < count; i++) {
            if (i == s_settings_idx) {
                lcd.setTextColor(TFT_BLACK);
                lcd.fillRoundRect(5, y - 5, 310, 26, 4, 0x07E0);
            } else {
                lcd.setTextColor(TFT_WHITE);
            }
            lcd.drawString(items[i], 15, y);
            y += 32;
        }
    }

    // ========================================================
    // Playback
    // ========================================================
    if (s_current_mode == MODE_PLAYBACK) {
        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(0x07E0);
        const catalog_category_t* play_cat = catalog_get_category(s_selected_category);
        lcd.drawString(play_cat ? play_cat->name : "---", 10, 48);

        lcd.setFont(&fonts::efontCN_16);
        lcd.setTextColor(TFT_WHITE);
        lcd.drawString(s_station_name, 10, 68);

        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(TFT_LIGHTGRAY);
        lcd.drawString(s_track_title, 10, 98);

        lcd.setFont(&fonts::Font0);
        lcd.setTextColor(0x07E0);
        lcd.drawString("Buffer:", 10, 128);

        int bar_w = 280;
        int bar_h = 16;
        lcd.drawRoundRect(10, 146, bar_w, bar_h, 4, TFT_WHITE);
        int fill_pct = 0;
        if (s_buffer_target > 0) {
            fill_pct = (int)((s_buffer_level * 100) / s_buffer_target);
        }
        if (fill_pct > 100) fill_pct = 100;
        int fill_w = (fill_pct * bar_w) / 100;
        if (fill_w > 0) {
            lcd.fillRoundRect(10, 146, fill_w, bar_h, 4, 0x07E0);
        }

        char buf[48];
        lcd.setTextColor(TFT_WHITE);
        snprintf(buf, sizeof(buf), "Buf: %d%% (%lu KB)",
                 fill_pct,
                 (unsigned long)(s_buffer_level / 1024));
        lcd.drawString(buf, 10, 168);

        // ========================================================
        // Если идет валидация — показываем прогресс-бар вместо Net/sparkline
        // ========================================================
        if (catalog_is_validating()) {
            int done = catalog_get_validated_count();
            int total = catalog_get_staging_count();
            int pct = (total > 0) ? (done * 100 / total) : 0;

            lcd.setFont(&fonts::Font0);
            lcd.setTextColor(0x07E0);
            lcd.drawString("Validate:", 10, 186);

            // Рамка прогресс-бара (та же геометрия, что и у Buffer)
            lcd.drawRoundRect(10, 198, 280, 16, 4, TFT_DARKGREY);
            int vbar_w = (280 * pct) / 100;
            if (vbar_w > 0) {
                lcd.fillRoundRect(10, 198, vbar_w, 16, 4, 0x07E0);
            }

            snprintf(buf, sizeof(buf), "%d / %d (%d%%)", done, total, pct);
            lcd.setTextColor(TFT_WHITE);
            lcd.drawString(buf, 10, 218);

        } else {
            // ========================================================
            // Обычная сетевая телеметрия
            // ========================================================
            int net_pct = 0;
            if (s_net_capacity > 0) {
                net_pct = (int)((s_net_fill * 100) / s_net_capacity);
            }
            if (net_pct > 100) net_pct = 100;

            if (net_pct == 0) {
                lcd.setTextColor(TFT_RED);
            } else if (net_pct < 10) {
                lcd.setTextColor(TFT_YELLOW);
            } else {
                lcd.setTextColor(0x07E0);
            }

            snprintf(buf, sizeof(buf), "Net: %d%% (%lu KB)",
                     net_pct,
                     (unsigned long)(s_net_fill / 1024));
            lcd.drawString(buf, 10, 186);

            // Мини-график сетевого буфера
            draw_net_sparkline_impl(lcd, 10, 204, 300, 18);

            lcd.setTextColor(TFT_RED);
            snprintf(buf, sizeof(buf), "Starve: %lu | Overflow: %lu",
                     (unsigned long)s_starve_count,
                     (unsigned long)s_overflow_count);
            lcd.drawString(buf, 10, 226);
        }
    }

    // ========================================================
    // Categories (со скроллингом и режимом удаления)
    // ========================================================
    else if (s_current_mode == MODE_CATEGORIES) {
        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(0x07E0);

        // Заголовок: в режиме выделения показываем счётчик
        if (s_cat_selection_mode) {
            char title[80];
            snprintf(title, sizeof(title), "Категории [ВЫДЕЛЕНИЕ: %d]", (int)s_marked_cat_count);
            lcd.drawString(title, 10, 48);
        } else {
            lcd.drawString("Выберите категорию:", 10, 48);
        }

        int visible_items = 5;
        int cat_count = (int)catalog_category_count();
        int total_items = cat_count + 1; // +1 для служебного пункта

        int start_idx = s_selected_category;
        if (start_idx > 2) start_idx = s_selected_category - 2;
        if (start_idx < 0) start_idx = 0;

        int y = 68;
        for (int i = start_idx; i < total_items && i < start_idx + visible_items; i++) {
            if (i < cat_count) {
                // ---- Обычная категория ----
                const catalog_category_t* c = catalog_get_category(i);
                if (c == nullptr) continue;

                bool is_marked = is_cat_marked((size_t)i);

                if (i == s_selected_category) {
                    lcd.setTextColor(TFT_BLACK);
                    lcd.fillRoundRect(5, y - 5, 310, 26, 4, 0x07E0);
                } else {
                    lcd.setTextColor(is_marked ? TFT_RED : TFT_WHITE);
                }

                char display_name[CATALOG_MAX_CATEGORY_LEN + 16];
                if (s_cat_selection_mode) {
                    snprintf(display_name, sizeof(display_name), is_marked ? "[x] %s" : "[ ] %s", c->name);
                } else {
                    snprintf(display_name, sizeof(display_name), "%s", c->name);
                }
                lcd.drawString(display_name, 15, y);
            } else {
                // ---- Служебный пункт ----
                if (i == s_selected_category) {
                    lcd.setTextColor(TFT_BLACK);
                    lcd.fillRoundRect(5, y - 5, 310, 26, 4, TFT_YELLOW);
                } else {
                    lcd.setTextColor(TFT_YELLOW);
                }

                char label[64];
                if (s_cat_selection_mode) {
                    if (s_marked_cat_count > 0) {
                        snprintf(label, sizeof(label), "Удалить отмеченные (%d)", (int)s_marked_cat_count);
                    } else {
                        snprintf(label, sizeof(label), "Выход из режима выделения");
                    }
                } else {
                    snprintf(label, sizeof(label), ">> Режим выделения");
                }
                lcd.drawString(label, 15, y);
            }
            y += 32;
        }
    }

    // ========================================================
    // Stations (со скроллингом, избранным и режимом выделения)
    // ========================================================
    else if (s_current_mode == MODE_STATIONS) {
        const catalog_category_t* cur_cat = catalog_get_category(s_selected_category);
        lcd.setFont(&fonts::efontCN_14);
        lcd.setTextColor(0x07E0);

        // Заголовок: в режиме выделения показываем счётчик
        if (s_selection_mode) {
            char title[80];
            snprintf(title, sizeof(title), "%s [ВЫДЕЛЕНИЕ: %d]",
                     cur_cat ? cur_cat->name : "---", (int)s_marked_count);
            lcd.drawString(title, 10, 48);
        } else {
            lcd.drawString(cur_cat ? cur_cat->name : "---", 10, 48);
        }

        int visible_items = 6;
        int station_count = cur_cat ? (int)cur_cat->count : 0;
        int total_items = station_count + 1; // +1 для служебного пункта

        int start_idx = s_ui_station_idx;
        if (start_idx > 3) {
            start_idx = s_ui_station_idx - 3;
        }
        int y = 68;

        for (int i = start_idx; i < total_items && i < start_idx + visible_items; i++) {
            if (i < station_count) {
                // ---- Обычная станция ----
                const catalog_station_t* st = catalog_get_station(s_selected_category, i);
                if (st == nullptr) continue;

                bool is_fav = catalog_is_favorite(st->id);
                bool is_marked = is_station_marked(st->id);

                if (i == s_ui_station_idx) {
                    lcd.setTextColor(TFT_BLACK);
                    lcd.fillRoundRect(5, y - 5, 310, 24, 4, 0x07E0);
                } else {
                    if (is_marked) {
                        lcd.setTextColor(TFT_RED); // Отмеченные — красные
                    } else {
                        lcd.setTextColor(TFT_WHITE);
                    }
                }

                lcd.setFont(&fonts::efontCN_16);

                char display_name[CATALOG_MAX_NAME_LEN + 16];
                if (s_selection_mode) {
                    // В режиме выделения показываем чекбоксы
                    if (is_marked) {
                        snprintf(display_name, sizeof(display_name), "[x] %s", st->name);
                    } else {
                        snprintf(display_name, sizeof(display_name), "[ ] %s", st->name);
                    }
                } else {
                    // В обычном режиме показываем звёздочку для избранных
                    if (is_fav) {
                        snprintf(display_name, sizeof(display_name), "★ %s", st->name);
                    } else {
                        snprintf(display_name, sizeof(display_name), "  %s", st->name);
                    }
                }

                lcd.drawString(display_name, 15, y);
            } else {
                // ---- Служебный пункт ----
                if (i == s_ui_station_idx) {
                    lcd.setTextColor(TFT_BLACK);
                    lcd.fillRoundRect(5, y - 5, 310, 24, 4, 0x07E0);
                } else {
                    lcd.setTextColor(TFT_YELLOW);
                }

                lcd.setFont(&fonts::efontCN_14);
                char label[64];
                if (s_selection_mode) {
                    if (s_marked_count > 0) {
                        snprintf(label, sizeof(label), "Удалить отмеченные (%d)", (int)s_marked_count);
                    } else {
                        snprintf(label, sizeof(label), "Выход из режима выделения");
                    }
                } else {
                    snprintf(label, sizeof(label), ">> Режим выделения");
                }
                lcd.drawString(label, 15, y);
            }
            y += 28;
        }
    }

    // ========================================================
    // Volume
    // ========================================================
    else if (s_current_mode == MODE_VOLUME) {
        lcd.setFont(&fonts::Font4);
        lcd.setTextColor(TFT_WHITE);
        lcd.setTextDatum(MC_DATUM);

        char vol_str[16];
        snprintf(vol_str, sizeof(vol_str), "%d%%", s_current_volume);
        lcd.drawString(vol_str, 160, 88);

        lcd.setTextDatum(TL_DATUM);

        int bar_w = 280;
        int bar_h = 20;

        lcd.drawRoundRect(20, 138, bar_w, bar_h, 5, TFT_WHITE);

        int fill_w = (s_current_volume * bar_w) / 100;
        if (fill_w > 0) {
            lcd.fillRoundRect(20, 138, fill_w, bar_h, 5, 0x07E0);
        }
    }
}

// ============================================================
// Wrapper-функции отрисовки
// ============================================================
static void draw_ui() {
    if (s_use_sprite && s_ui_sprite != nullptr) {
        draw_ui_impl(*s_ui_sprite);
        s_ui_sprite->pushSprite(0, 0);
    } else {
        draw_ui_impl(M5.Lcd);
    }
}

// ============================================================
// Публичные функции
// ============================================================
void ui_init() {
    s_use_sprite = false;

    if (s_ui_sprite != nullptr) {
        delete s_ui_sprite;
        s_ui_sprite = nullptr;
    }

    if (!SPRITE_DEBUG_FORCE_DIRECT) {
        s_ui_sprite = new (std::nothrow) m5gfx::LGFX_Sprite(&M5.Lcd);

        if (s_ui_sprite != nullptr) {
            s_ui_sprite->setPsram(true);

            if (s_ui_sprite->createSprite(320, 240)) {
                s_ui_sprite->setTextDatum(TL_DATUM);
                s_use_sprite = true;
                ESP_LOGI(TAG, "UI sprite created: 320x240, psram=true");
            } else {
                ESP_LOGW(TAG, "UI sprite createSprite failed");
                delete s_ui_sprite;
                s_ui_sprite = nullptr;
            }
        } else {
            ESP_LOGW(TAG, "UI sprite allocation failed");
        }
    } else {
        ESP_LOGW(TAG, "UI sprite debug forced direct drawing");
    }

    // Явно зажигаем подсветку
    M5.Display.setBrightness(DISPLAY_BRIGHTNESS);

    ESP_LOGI(TAG, "UI initialized, sprite_enabled=%d, brightness=%d",
             s_use_sprite, DISPLAY_BRIGHTNESS);
}

void ui_update() {
    static bool last_btn_a = false, last_btn_b = false, enc_is_pressed = false;
    static uint32_t enc_press_start_time = 0, redraw_counter = 0;
    static bool ui_needs_redraw = true;

    // Состояние сна экрана
    static uint64_t s_last_activity_us = esp_timer_get_time();
    static bool     s_display_awake    = true;

    M5.update();

    // ========================================================
    // Headphone detection (проверяем каждые 200 мс)
    // ========================================================
    s_headphone_check_counter++;
    if (s_headphone_check_counter >= 4) { // 4 * 50 мс = 200 мс
        s_headphone_check_counter = 0;

        bool hp_connected = audio_is_headphone_connected();

        if (hp_connected != s_last_headphone_state) {
            s_last_headphone_state = hp_connected;

            // Переключаем аудиовыход
            audio_set_output_headphone(hp_connected);

            // RGB LED индикация через M5Unified (индекс 0, затем R, G, B)
            if (hp_connected) {
                M5.Led.setColor(0, 0, 255, 0); // index=0, R=0, G=255, B=0 (Зелёный)
                ESP_LOGI(TAG, "Headphones connected -> GREEN LED");
            } else {
                M5.Led.setColor(0, 0, 0, 255); // index=0, R=0, G=0, B=255 (Синий)
                ESP_LOGI(TAG, "Headphones disconnected -> BLUE LED");
            }
        }
    }

    // ========================================================
    // 1. Собираем весь ввод в один снимок кадра
    // ========================================================
    auto touch          = M5.Touch.getDetail();
    bool touch_pressed  = touch.wasPressed();

    if (touch_pressed) {
        ESP_LOGI(TAG, "Touch raw: x=%d y=%d mode=%d",
                 (int)touch.x, (int)touch.y, (int)s_current_mode);
    }

    int32_t increment   = read_hmi_register_32(0x10);
    bool btn_a          = read_hmi_register_8(0x21);
    bool btn_b          = read_hmi_register_8(0x22);
    bool btn_enc        = read_hmi_register_8(0x20);

    bool btn_a_edge       = btn_a && !last_btn_a;
    bool btn_b_edge       = btn_b && !last_btn_b;
    bool enc_just_pressed = btn_enc && !enc_is_pressed;
    bool enc_just_released = !btn_enc && enc_is_pressed;

    bool activity = touch_pressed || btn_a_edge || btn_b_edge ||
                    enc_just_pressed || enc_just_released || (increment != 0);

    uint64_t now_us = esp_timer_get_time();

    // ========================================================
    // 2. Управление сном / пробуждением экрана
    // ========================================================
    if (activity) {
        s_last_activity_us = now_us;
        if (!s_display_awake) {
            M5.Display.setBrightness(DISPLAY_BRIGHTNESS);
            s_display_awake = true;
            ui_needs_redraw = true;
            ESP_LOGI(TAG, "Display wakeup");
        }
    } else if (s_display_awake &&
               (now_us - s_last_activity_us) > (uint64_t)DISPLAY_IDLE_TIMEOUT_MS * 1000ULL) {
        M5.Display.setBrightness(0);
        s_display_awake = false;
        ESP_LOGI(TAG, "Display sleep (idle %u ms)", (unsigned)DISPLAY_IDLE_TIMEOUT_MS);
    }

    // Пока экран спит — ничего не рисуем и не обрабатываем,
    // но синхронизируем флаги кнопок, чтобы после пробуждения
    // не генерировались ложные front-edges.
    if (!s_display_awake) {
        last_btn_a     = btn_a;
        last_btn_b     = btn_b;
        enc_is_pressed = btn_enc;
        return;
    }

    // ========================================================
    // 3. Мигание курсора
    // ========================================================
    s_cursor_blink_counter++;
    if (s_cursor_blink_counter >= 10) {
        s_cursor_blink = !s_cursor_blink;
        s_cursor_blink_counter = 0;
        if (s_keyboard_active) ui_needs_redraw = true;
    }

    redraw_counter++;
    if (redraw_counter >= 10) {
        ui_needs_redraw = true;
        redraw_counter = 0;
    }

    // ========================================================
    // 4. Обработка тача в режиме Wi-Fi
    // ========================================================
    if (s_current_mode == MODE_WIFI && touch_pressed) {
        ESP_LOGI(TAG, "WiFi touch: x=%d y=%d kb=%d scan=%d focus=%d",
                 (int)touch.x,
                 (int)touch.y,
                 (int)s_keyboard_active,
                 (int)s_wifi_show_scan_results,
                 s_wifi_focus);

        handle_wifi_touch(touch.x, touch.y);
        ui_needs_redraw = true;
    }

    // ========================================================
    // 5. Энкодер: ВРАЩЕНИЕ
    // ========================================================
    if (increment != 0) {
        s_encoder_accumulator += increment;

        int32_t effective = 0;

        if (s_encoder_accumulator >= ENCODER_THRESHOLD) {
            effective = s_encoder_accumulator / ENCODER_THRESHOLD;
            s_encoder_accumulator %= ENCODER_THRESHOLD;
        } else if (s_encoder_accumulator <= -ENCODER_THRESHOLD) {
            effective = s_encoder_accumulator / ENCODER_THRESHOLD;
            s_encoder_accumulator %= ENCODER_THRESHOLD;
        }

        if (effective != 0) {
        	if (s_current_mode == MODE_CATEGORIES) {
        	    int cat_count = (int)catalog_category_count();
        	    int total_items = cat_count + 1; // Учитываем служебный пункт

        	    s_selected_category += effective;
        	    if (s_selected_category < 0) s_selected_category = 0;
        	    if (total_items > 0 && s_selected_category >= total_items) {
        	        s_selected_category = total_items - 1;
        	    }
        	    ui_needs_redraw = true;
        	}
            else if (s_current_mode == MODE_STATIONS) {
                const catalog_category_t* ec = catalog_get_category(s_selected_category);
                int station_count = ec ? (int)ec->count : 0;
                int total_items = station_count + 1; // +1 для служебного пункта

                s_ui_station_idx += effective;
                if (s_ui_station_idx < 0) s_ui_station_idx = 0;
                if (total_items > 0 && s_ui_station_idx >= total_items) {
                    s_ui_station_idx = total_items - 1;
                }
                ui_needs_redraw = true;
            }
            else if (s_current_mode == MODE_FAVORITES) {
                int fav_count = (int)catalog_get_favorites_count();

                s_fav_station_idx += effective;

                if (s_fav_station_idx < 0) s_fav_station_idx = 0;
                if (fav_count > 0 && s_fav_station_idx >= fav_count) {
                    s_fav_station_idx = fav_count - 1;
                }

                ui_needs_redraw = true;
            }
            else if (s_current_mode == MODE_SETTINGS || s_current_mode == MODE_CATALOG) {
                s_settings_idx += effective;
                if (s_settings_idx < 0) s_settings_idx = 0;
                if (s_current_mode == MODE_SETTINGS && s_settings_idx >= 4) s_settings_idx = 3;
                if (s_current_mode == MODE_CATALOG && s_settings_idx >= 5) s_settings_idx = 4; // <-- Важно!
                ui_needs_redraw = true;
            }
            else if (s_current_mode == MODE_WIFI && !s_keyboard_active) {
                if (s_wifi_show_scan_results) {
                    int count = wifi_scan_get_count();
                    if (count > 0) {
                        if (s_wifi_scan_selected_idx < 0) s_wifi_scan_selected_idx = 0;
                        s_wifi_scan_selected_idx += effective;
                        if (s_wifi_scan_selected_idx < 0) s_wifi_scan_selected_idx = 0;
                        if (s_wifi_scan_selected_idx >= count) s_wifi_scan_selected_idx = count - 1;
                    } else {
                        s_wifi_scan_selected_idx = -1;
                    }
                } else {
                    // 0=SSID, 1=PASS, 2=Scan, 3=Connect, 4=Networks, 5=Back
                    s_wifi_focus += effective;
                    if (s_wifi_focus < 0) s_wifi_focus = 0;
                    if (s_wifi_focus > 5) s_wifi_focus = 5;
                }
                ui_needs_redraw = true;
            }
            else if (s_current_mode == MODE_VOLUME || s_current_mode == MODE_PLAYBACK) {
                int new_vol = s_current_volume + effective * 5;

                if (new_vol < 0) new_vol = 0;
                if (new_vol > 100) new_vol = 100;

                if (new_vol != s_current_volume) {
                    s_current_volume = new_vol;
                    s_new_volume = new_vol;
                    s_pending_event = UI_EVENT_VOLUME_CHANGED;
                    ui_needs_redraw = true;
                }
            }
        }
    }

    // ========================================================
    // 6. Кнопка A: назад / отмена
    // ========================================================
    if (btn_a_edge) {
        if (s_current_mode == MODE_STATIONS && s_selection_mode) {
            s_selection_mode = false;
            clear_marks();
            ESP_LOGI(TAG, "Station selection mode cancelled by button A");
        }
        else if (s_current_mode == MODE_CATEGORIES && s_cat_selection_mode) {
            s_cat_selection_mode = false;
            clear_cat_marks();
            ESP_LOGI(TAG, "Category selection mode cancelled by button A");
        }
        else if (s_current_mode == MODE_WIFI) {
            wifi_back_action();
        }
        else if (s_current_mode == MODE_KNOWN_NETWORKS) {
            s_current_mode = MODE_WIFI;
            s_wifi_focus = 4;
        }
        else if (s_current_mode == MODE_PLAYBACK) {
            s_current_mode = MODE_SETTINGS;
        }
        else if (s_current_mode == MODE_SETTINGS) {
            s_current_mode = MODE_STATIONS;
        }
        else if (s_current_mode == MODE_STATIONS) {
            s_current_mode = MODE_CATEGORIES;
        }
        else if (s_current_mode == MODE_CATEGORIES) {
            s_current_mode = MODE_FAVORITES;
        }
        else if (s_current_mode == MODE_FAVORITES) {
            s_current_mode = MODE_PLAYBACK;
        }
        else if (s_current_mode == MODE_ABOUT || s_current_mode == MODE_CATALOG) {
            s_current_mode = MODE_SETTINGS;
        }

        s_encoder_accumulator = 0;
        ui_needs_redraw = true;
    }
    last_btn_a = btn_a;

    // ========================================================
    // 7. Кнопка B: вперёд
    //    В Wi-Fi тоже делаем безопасный выход, чтобы не было ловушки.
    // ========================================================
    if (btn_b_edge) { // <--- ИСПРАВЛЕНО: здесь должно быть btn_b_edge, а не btn_a_edge!
        if (s_current_mode == MODE_WIFI) {
            wifi_back_action();
        }
        else if (s_current_mode == MODE_KNOWN_NETWORKS) {
            s_current_mode = MODE_WIFI;
            s_wifi_focus = 4; // Возвращаем фокус на кнопку Networks
        }
        else if (s_current_mode == MODE_PLAYBACK) {
            s_current_mode = MODE_FAVORITES;
        }
        else if (s_current_mode == MODE_FAVORITES) {
            s_current_mode = MODE_CATEGORIES;
        }
        else if (s_current_mode == MODE_CATEGORIES) {
            s_current_mode = MODE_STATIONS;
        }
        else if (s_current_mode == MODE_STATIONS) {
            s_current_mode = MODE_SETTINGS;
        }
        else if (s_current_mode == MODE_SETTINGS) {
            s_current_mode = MODE_PLAYBACK;
        }
        else if (s_current_mode == MODE_ABOUT || s_current_mode == MODE_CATALOG) {
            s_current_mode = MODE_SETTINGS;
        }

        s_encoder_accumulator = 0;
        ui_needs_redraw = true;
    }
    last_btn_b = btn_b;

    // ========================================================
    // 8. Кнопка энкодера: короткое и долгое нажатие
    // ========================================================
    if (btn_enc) {
        if (!enc_is_pressed) {
            enc_is_pressed = true;
            enc_press_start_time = esp_timer_get_time() / 1000;
            s_encoder_accumulator = 0;
            enc_long_press_handled = false;
        } else {
            uint32_t now = esp_timer_get_time() / 1000;

            if (now - enc_press_start_time >= 1000 && !enc_long_press_handled) {
                if (s_current_mode == MODE_STATIONS) {
                    int global_idx = catalog_global_index(s_selected_category, s_ui_station_idx);
                    const catalog_station_t* st = catalog_get_station_by_global_index(global_idx);

                    if (st != nullptr) {
                        bool added = catalog_toggle_favorite(st->id);
                        catalog_save_favorites();
                        ESP_LOGI(TAG, "Favorite %s: %s", added ? "ADDED" : "REMOVED", st->name);
                    }
                }
                else if (s_current_mode == MODE_FAVORITES) {
                    int global_idx = catalog_get_favorite_global_index(s_fav_station_idx);
                    const catalog_station_t* st = catalog_get_station_by_global_index(global_idx);

                    if (st != nullptr) {
                        bool added = catalog_toggle_favorite(st->id);
                        catalog_save_favorites();
                        ESP_LOGI(TAG, "Favorite %s: %s", added ? "ADDED" : "REMOVED", st->name);

                        if (!added) {
                            int fav_count = (int)catalog_get_favorites_count();

                            if (s_fav_station_idx >= fav_count && fav_count > 0) {
                                s_fav_station_idx = fav_count - 1;
                            } else if (fav_count == 0) {
                                s_fav_station_idx = 0;
                            }
                        }
                    }
                }
                else if (s_current_mode == MODE_WIFI) {
                    wifi_back_action();
                }
                else {
                    // В остальных режимах долгое нажатие = MUTE
                    s_is_muted = !s_is_muted;
                    s_pending_event = UI_EVENT_MUTE_TOGGLED;
                }

                enc_long_press_handled = true;
                ui_needs_redraw = true;
            }
        }
    } else {
        if (enc_is_pressed) {
                if (!enc_long_press_handled) {
                	if (s_current_mode == MODE_STATIONS) {
                	    const catalog_category_t* cat = catalog_get_category(s_selected_category);
                	    int station_count = cat ? (int)cat->count : 0;

                	    if (s_ui_station_idx < station_count) {
                	        // Обычная станция
                	        if (s_selection_mode) {
                	            // В режиме выделения: отметить/снять отметку
                	            const catalog_station_t* st = catalog_get_station(s_selected_category, s_ui_station_idx);
                	            if (st) {
                	                toggle_mark_station(st->id);
                	                ESP_LOGI(TAG, "Station %s %s (marked: %d)",
                	                         st->name,
                	                         is_station_marked(st->id) ? "MARKED" : "UNMARKED",
                	                         (int)s_marked_count);
                	            }
                	        } else {
                	            // Обычный режим: воспроизведение
                	            s_new_station_idx = (int)catalog_global_index(s_selected_category, s_ui_station_idx);
                	            s_pending_event = UI_EVENT_STATION_CHANGED;
                	            s_current_mode = MODE_PLAYBACK;
                	        }
                	    } else {
                	        // Служебный пункт
                	        if (s_selection_mode) {
                	            if (s_marked_count > 0) {
                	                ESP_LOGI(TAG, "Deleting %d marked stations...", (int)s_marked_count);
                	                catalog_remove_stations(s_marked_ids, s_marked_count);
                	                clear_marks();
                	                // Корректируем индекс, если вышли за границы
                	                const catalog_category_t* new_cat = catalog_get_category(s_selected_category);
                	                int new_count = new_cat ? (int)new_cat->count : 0;
                	                if (s_ui_station_idx >= new_count) {
                	                    s_ui_station_idx = new_count > 0 ? new_count - 1 : 0;
                	                }
                	            }
                	            s_selection_mode = false;
                	        } else {
                	            // Входим в режим выделения
                	            s_selection_mode = true;
                	            clear_marks();
                	            ESP_LOGI(TAG, "Selection mode enabled");
                	        }
                	    }
                	}
                    else if (s_current_mode == MODE_FAVORITES) {
                        int global_idx = catalog_get_favorite_global_index(s_fav_station_idx);
                        if (global_idx >= 0) {
                            s_new_station_idx = global_idx;
                            s_pending_event = UI_EVENT_STATION_CHANGED;
                            s_current_mode = MODE_PLAYBACK;
                        }
                    }
                    else if (s_current_mode == MODE_CATEGORIES) {
                        int cat_count = (int)catalog_category_count();

                        if (s_selected_category < cat_count) {
                            // ---- Нажали на обычную категорию ----
                            if (s_cat_selection_mode) {
                                // В режиме выделения: отметить/снять отметку
                                toggle_cat_mark((size_t)s_selected_category);
                                ESP_LOGI(TAG, "Category %s %s (marked: %d)",
                                         catalog_get_category(s_selected_category)->name,
                                         is_cat_marked((size_t)s_selected_category) ? "MARKED" : "UNMARKED",
                                         (int)s_marked_cat_count);
                            } else {
                                // Обычный режим: переход в список станций
                                s_current_mode = MODE_STATIONS;
                                s_ui_station_idx = 0;
                            }
                        } else {
                            // ---- Нажали на служебный пункт ----
                            if (s_cat_selection_mode) {
                                if (s_marked_cat_count > 0) {
                                    ESP_LOGI(TAG, "Deleting %d marked categories...", (int)s_marked_cat_count);
                                    catalog_remove_categories(s_marked_cat_indices, s_marked_cat_count);
                                    clear_cat_marks();
                                    // Корректируем индекс, если вышли за границы
                                    int new_count = (int)catalog_category_count();
                                    if (s_selected_category >= new_count) {
                                        s_selected_category = new_count > 0 ? new_count - 1 : 0;
                                    }
                                }
                                s_cat_selection_mode = false;
                            } else {
                                // Входим в режим выделения
                                s_cat_selection_mode = true;
                                clear_cat_marks();
                                ESP_LOGI(TAG, "Category selection mode enabled");
                            }
                        }
                    }
                    else if (s_current_mode == MODE_SETTINGS) {
                        if (s_settings_idx == 0) {
                            s_current_mode = MODE_WIFI;
                            s_wifi_focus = 0;
                            s_keyboard_active = false;
                            s_keyboard_target = KB_TARGET_NONE;
                            s_wifi_show_scan_results = false;
                            s_wifi_scan_selected_idx = -1;
                        }
                        else if (s_settings_idx == 1) {
                            s_current_mode = MODE_CATALOG;
                            s_settings_idx = 0; // Сбрасываем индекс для подменю
                        }
                        else if (s_settings_idx == 2) {
                            s_current_mode = MODE_ABOUT;
                        }
                        else if (s_settings_idx == 3) {
                            s_current_mode = MODE_PLAYBACK;
                        }
                    }
                    else if (s_current_mode == MODE_CATALOG) {
                        if (s_settings_idx == 0) {
                            ESP_LOGI(TAG, "Catalog: Loading working catalog to staging...");
                            catalog_load_working_to_staging();
                        }
                        else if (s_settings_idx == 1) {
                            ESP_LOGI(TAG, "Catalog: Importing M3U from SD...");
                            catalog_staging_load_from_m3u_dir(CATALOG_M3U_DIR);
                        }
                        else if (s_settings_idx == 2) {
                            ESP_LOGI(TAG, "Catalog: Starting background validation...");
                            catalog_start_validation();
                        }
                        else if (s_settings_idx == 3) {
                            ESP_LOGI(TAG, "Catalog: Merging staging to working...");
                            catalog_staging_merge_to_working();
                        }
                        else if (s_settings_idx == 4) {
                            s_current_mode = MODE_SETTINGS;
                            s_settings_idx = 1; // Возвращаемся на пункт "Catalog" в настройках
                        }
                    }
                    else if (s_current_mode == MODE_ABOUT) {
                        s_current_mode = MODE_SETTINGS;
                    }
                    else if (s_current_mode == MODE_WIFI) {
                        if (s_wifi_show_scan_results) {
                            wifi_select_scan_result(s_wifi_scan_selected_idx);
                        }
                        else if (s_keyboard_active) {
                            wifi_kb_close();
                        }
                        else {
                            if (s_wifi_focus == 0) wifi_kb_open(KB_TARGET_SSID);
                            else if (s_wifi_focus == 1) wifi_kb_open(KB_TARGET_PASS);
                            else if (s_wifi_focus == 2) {
                                ESP_LOGI(TAG, "Wi-Fi Scan requested via encoder");
                                ui_trigger_wifi_scan();
                            }
                            else if (s_wifi_focus == 3) {
                                ESP_LOGI(TAG, "Wi-Fi Connect requested via encoder");
                                ui_trigger_wifi_connect();
                            }
                            else if (s_wifi_focus == 4) {
                                s_current_mode = MODE_KNOWN_NETWORKS;
                                s_settings_idx = 0;
                            }
                            else if (s_wifi_focus == 5) {
                                wifi_back_action();
                            }
                        }
                    }
                }

            enc_is_pressed = false;
            s_encoder_accumulator = 0;
            ui_needs_redraw = true;
        }
    }

    // ========================================================
    // 9. Перерисовка
    // ========================================================
    if (ui_needs_redraw) {
        draw_ui();
        ui_needs_redraw = false;
    }
}

static int s_known_networks_count = 0;
static char s_last_attempted_ssid[33] = {0};
static char s_last_attempted_pass[65] = {0};

static const char* TAG = "WIFI";

static EventGroupHandle_t s_wifi_event_group = NULL;
static const int WIFI_CONNECTED_BIT = BIT0;
static volatile bool s_is_connected = false;

static int s_connection_attempts = 0;
static const int MAX_CONNECTION_ATTEMPTS = 3;

static volatile bool s_testing_mode = false;
static volatile bool s_test_passed = false;
static volatile bool s_test_ignore_disconnect = false;
static volatile wifi_test_result_t s_last_error = WIFI_TEST_NOT_STARTED;

static volatile bool s_manual_reconnect = false;
static volatile uint8_t s_last_disconnect_reason = 0;
static volatile bool s_test_running = false;
static bool s_sntp_started = false;

static wifi_scan_record_t s_scan_results[20];
static int s_scan_count = 0;

static void wifi_event_handler(void* arg, esp_event_base_t base, int32_t id, void* data) {
    if (base == WIFI_EVENT && id == WIFI_EVENT_STA_START) {
        ESP_LOGI(TAG, "Wi-Fi started, connecting...");
        esp_wifi_connect();
    }
    else if (base == WIFI_EVENT && id == WIFI_EVENT_STA_DISCONNECTED) {
        wifi_event_sta_disconnected_t* disconnected = (wifi_event_sta_disconnected_t*)data;
        s_is_connected = false;
        s_last_disconnect_reason = disconnected->reason;

        if (disconnected->reason == WIFI_REASON_AUTH_FAIL || disconnected->reason == WIFI_REASON_4WAY_HANDSHAKE_TIMEOUT) {
            s_last_error = WIFI_TEST_FAILED_AUTH;
        } else if (disconnected->reason == WIFI_REASON_NO_AP_FOUND) {
            s_last_error = WIFI_TEST_FAILED_NOT_FOUND;
        } else {
            s_last_error = WIFI_TEST_FAILED_OTHER;
        }

        if (s_testing_mode) {
            if (s_test_ignore_disconnect && disconnected->reason == WIFI_REASON_ASSOC_LEAVE) {
                s_test_ignore_disconnect = false;
                return;
            }
            ESP_LOGW(TAG, "Wi-Fi disconnected during test (reason: %d)", disconnected->reason);
            s_test_passed = false;
            xEventGroupSetBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
            return;
        }

        if (s_manual_reconnect) {
            ESP_LOGI(TAG, "Manual reconnect in progress, skipping auto reconnect (reason: %d)", disconnected->reason);
            return;
        }

        s_connection_attempts++;
        if (s_connection_attempts <= MAX_CONNECTION_ATTEMPTS) {
            ESP_LOGW(TAG, "Wi-Fi disconnected, reconnecting... (attempt %d/%d, reason: %d)", s_connection_attempts, MAX_CONNECTION_ATTEMPTS, disconnected->reason);
            esp_wifi_connect();
        } else {
            ESP_LOGE(TAG, "Wi-Fi connection failed after %d attempts", MAX_CONNECTION_ATTEMPTS);
            xEventGroupSetBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
        }
    }
    else if (base == IP_EVENT && id == IP_EVENT_STA_GOT_IP) {
        ip_event_got_ip_t* event = (ip_event_got_ip_t*) data;
        ESP_LOGI(TAG, "Got IP: " IPSTR, IP2STR(&event->ip_info.ip));
        s_is_connected = true;
        s_connection_attempts = 0;
        s_last_error = WIFI_TEST_NOT_STARTED;
        s_last_disconnect_reason = 0;

        // АВТО-СОХРАНЕНИЕ: если подключились успешно, добавляем в знакомые сети
        if (s_last_attempted_ssid[0] != '\0') {
            wifi_known_networks_add(s_last_attempted_ssid, s_last_attempted_pass);
        }

        if (s_testing_mode) {
            s_test_passed = true;
        }
        xEventGroupSetBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
    }
}

void wifi_connect(const char* ssid, const char* pass) {
    ESP_LOGI(TAG, "Connecting to Wi-Fi: SSID='%s'", ssid);
    strncpy(s_last_attempted_ssid, ssid, sizeof(s_last_attempted_ssid) - 1);
    strncpy(s_last_attempted_pass, pass, sizeof(s_last_attempted_pass) - 1);
    s_connection_attempts = 0;

    if (s_wifi_event_group == NULL) {
        s_wifi_event_group = xEventGroupCreate();
    }

    esp_netif_init();
    esp_event_loop_create_default();
    esp_netif_create_default_wifi_sta();

    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    esp_wifi_init(&cfg);

    esp_event_handler_instance_register(WIFI_EVENT, ESP_EVENT_ANY_ID, &wifi_event_handler, NULL, NULL);
    esp_event_handler_instance_register(IP_EVENT, IP_EVENT_STA_GOT_IP, &wifi_event_handler, NULL, NULL);

    wifi_config_t wc = {};
    strncpy((char*)wc.sta.ssid, ssid, sizeof(wc.sta.ssid) - 1);
    strncpy((char*)wc.sta.password, pass, sizeof(wc.sta.password) - 1);

    esp_wifi_set_mode(WIFI_MODE_STA);
    esp_wifi_set_config(WIFI_IF_STA, &wc);
    esp_wifi_start();
    esp_wifi_set_ps(WIFI_PS_NONE);

    ESP_LOGI(TAG, "Waiting for connection...");
    xEventGroupWaitBits(s_wifi_event_group, WIFI_CONNECTED_BIT, pdFALSE, pdFALSE, portMAX_DELAY);

    if (s_is_connected) {
        ESP_LOGI(TAG, "Wi-Fi connected successfully");
    } else {
        ESP_LOGE(TAG, "Wi-Fi connection failed - user can fix settings via UI");
    }
}

void wifi_reconnect(const char* ssid, const char* pass) {
    if (s_testing_mode || s_test_running) {
        ESP_LOGW(TAG, "Wi-Fi test active, ignoring reconnect request");
        return;
    }

    ESP_LOGI(TAG, "Reconnecting to Wi-Fi: SSID='%s'", ssid);
    s_connection_attempts = 0;
    s_manual_reconnect = true;

    if (s_wifi_event_group != NULL) {
        xEventGroupClearBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
    }

    wifi_config_t wc = {};
    strncpy((char*)wc.sta.ssid, ssid, sizeof(wc.sta.ssid) - 1);
    strncpy((char*)wc.sta.password, pass, sizeof(wc.sta.password) - 1);

    esp_err_t err = esp_wifi_set_config(WIFI_IF_STA, &wc);
    if (err != ESP_OK) {
        ESP_LOGW(TAG, "esp_wifi_set_config failed: %s", esp_err_to_name(err));
    }

    esp_wifi_disconnect();
    vTaskDelay(pdMS_TO_TICKS(300));
    esp_wifi_connect();
    vTaskDelay(pdMS_TO_TICKS(300));
    s_manual_reconnect = false;
}

bool wifi_is_connected() { return s_is_connected; }

bool wifi_wait_connected(uint32_t timeout_ms) {
    if (s_is_connected) return true;
    if (s_wifi_event_group == NULL) return false;
    EventBits_t bits = xEventGroupWaitBits(s_wifi_event_group, WIFI_CONNECTED_BIT, pdFALSE, pdFALSE, pdMS_TO_TICKS(timeout_ms));
    return s_is_connected && (bits & WIFI_CONNECTED_BIT);
}

bool wifi_test_connection(const char* ssid, const char* pass) {
    if (s_test_running) {
        ESP_LOGW(TAG, "Wi-Fi test already running, refusing concurrent test");
        return false;
    }
    s_test_running = true;
    ESP_LOGI(TAG, "Testing Wi-Fi connection: SSID='%s'", ssid);
    s_testing_mode = true;
    bool result = false;

    for (int attempt = 1; attempt <= 2; attempt++) {
        s_test_passed = false;
        s_test_ignore_disconnect = true;
        s_last_error = WIFI_TEST_NOT_STARTED;
        s_last_disconnect_reason = 0;

        if (s_wifi_event_group != NULL) {
            xEventGroupClearBits(s_wifi_event_group, WIFI_CONNECTED_BIT);
        }

        wifi_config_t wc = {};
        strncpy((char*)wc.sta.ssid, ssid, sizeof(wc.sta.ssid) - 1);
        strncpy((char*)wc.sta.password, pass, sizeof(wc.sta.password) - 1);

        esp_err_t err = esp_wifi_set_config(WIFI_IF_STA, &wc);
        if (err != ESP_OK) ESP_LOGW(TAG, "Test set_config failed: %s", esp_err_to_name(err));

        esp_wifi_disconnect();
        vTaskDelay(pdMS_TO_TICKS(200));
        esp_wifi_connect();

        xEventGroupWaitBits(s_wifi_event_group, WIFI_CONNECTED_BIT, pdTRUE, pdFALSE, pdMS_TO_TICKS(10000));

        if (s_test_passed) {
            ESP_LOGI(TAG, "Wi-Fi test: SUCCESS (attempt %d)", attempt);
            result = true;
            break;
        }

        ESP_LOGW(TAG, "Wi-Fi test attempt %d failed: error=%d, reason=%d", attempt, s_last_error, s_last_disconnect_reason);
        if (attempt == 1 && s_last_error != WIFI_TEST_FAILED_NOT_FOUND) {
            ESP_LOGW(TAG, "Retrying Wi-Fi test once...");
            vTaskDelay(pdMS_TO_TICKS(1000));
            continue;
        }
        break;
    }

    s_testing_mode = false;
    s_test_ignore_disconnect = false;
    if (!result && s_last_error == WIFI_TEST_NOT_STARTED) s_last_error = WIFI_TEST_FAILED_TIMEOUT;
    if (!result) ESP_LOGW(TAG, "Wi-Fi test: FAILED (error=%d)", s_last_error);
    s_test_running = false;
    return result;
}

int wifi_scan_networks(void) {
    ESP_LOGI(TAG, "Starting Wi-Fi scan...");
    wifi_scan_config_t scan_config = { .ssid = NULL, .bssid = NULL, .channel = 0, .show_hidden = false };
    esp_err_t err = esp_wifi_scan_start(&scan_config, true);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Wi-Fi scan start failed: %s", esp_err_to_name(err));
        return -1;
    }

    uint16_t ap_count = 20;
    wifi_ap_record_t *ap_info = (wifi_ap_record_t *)heap_caps_malloc(20 * sizeof(wifi_ap_record_t), MALLOC_CAP_INTERNAL);
    if (!ap_info) {
        ESP_LOGE(TAG, "Failed to allocate memory for Wi-Fi scan results");
        return -1;
    }

    err = esp_wifi_scan_get_ap_records(&ap_count, ap_info);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Wi-Fi scan get records failed: %s", esp_err_to_name(err));
        free(ap_info);
        return -1;
    }

    s_scan_count = ap_count;
    for (int i = 0; i < ap_count; i++) {
        memcpy(s_scan_results[i].ssid, ap_info[i].ssid, 33);
        s_scan_results[i].rssi = ap_info[i].rssi;
    }
    free(ap_info);

    for (int i = 0; i < s_scan_count - 1; i++) {
        for (int j = 0; j < s_scan_count - i - 1; j++) {
            if (s_scan_results[j].rssi < s_scan_results[j + 1].rssi) {
                wifi_scan_record_t temp = s_scan_results[j];
                s_scan_results[j] = s_scan_results[j + 1];
                s_scan_results[j + 1] = temp;
            }
        }
    }
    ESP_LOGI(TAG, "Wi-Fi scan completed, found %d networks", s_scan_count);
    return s_scan_count;
}

int wifi_scan_get_count(void) { return s_scan_count; }
const wifi_scan_record_t* wifi_scan_get_record(int index) {
    if (index < 0 || index >= s_scan_count) return NULL;
    return &s_scan_results[index];
}

bool wifi_known_networks_load(void) {
    if (!sd_manager_is_mounted()) return false;
    size_t len = 0;
    char* text = sd_manager_read_text("/sd/radio/known_networks.json", &len);
    if (!text) return false;

    cJSON* root = cJSON_Parse(text);
    free(text);
    if (!root) return false;

    s_known_networks_count = 0;
    cJSON* arr = cJSON_GetObjectItemCaseSensitive(root, "networks");
    if (cJSON_IsArray(arr)) {
        cJSON* item = NULL;
        cJSON_ArrayForEach(item, arr) {
            if (s_known_networks_count < MAX_KNOWN_NETWORKS) {
                cJSON* ssid = cJSON_GetObjectItemCaseSensitive(item, "ssid");
                cJSON* pass = cJSON_GetObjectItemCaseSensitive(item, "pass");
                cJSON* auto_conn = cJSON_GetObjectItemCaseSensitive(item, "auto");

                if (cJSON_IsString(ssid) && cJSON_IsString(pass)) {
                    strncpy(s_known_networks[s_known_networks_count].ssid, ssid->valuestring, 32);
                    strncpy(s_known_networks[s_known_networks_count].pass, pass->valuestring, 64);
                    s_known_networks[s_known_networks_count].auto_connect = cJSON_IsTrue(auto_conn);
                    s_known_networks_count++;
                }
            }
        }
    }
    cJSON_Delete(root);
    return true;
}

bool wifi_known_networks_save(void) {
    if (!sd_manager_is_mounted()) return false;
    mkdir("/sd/radio", 0777);

    cJSON* root = cJSON_CreateObject();
    cJSON* arr = cJSON_CreateArray();
    for (int i = 0; i < s_known_networks_count; i++) {
        cJSON* item = cJSON_CreateObject();
        cJSON_AddStringToObject(item, "ssid", s_known_networks[i].ssid);
        cJSON_AddStringToObject(item, "pass", s_known_networks[i].pass);
        cJSON_AddBoolToObject(item, "auto", s_known_networks[i].auto_connect);
        cJSON_AddItemToArray(arr, item);
    }
    cJSON_AddItemToObject(root, "networks", arr);

    char* out = cJSON_PrintUnformatted(root);
    cJSON_Delete(root);

    bool ok = sd_manager_write_text("/sd/radio/known_networks.json", out);
    cJSON_free(out);
    return ok;
}

bool wifi_known_networks_add(const char* ssid, const char* pass) {
    if (!ssid || !pass) return false;
    for (int i = 0; i < s_known_networks_count; i++) {
        if (strcmp(s_known_networks[i].ssid, ssid) == 0) {
            strncpy(s_known_networks[i].pass, pass, sizeof(s_known_networks[i].pass) - 1);
            s_known_networks[i].auto_connect = true;
            return wifi_known_networks_save();
        }
    }
    if (s_known_networks_count < MAX_KNOWN_NETWORKS) {
        strncpy(s_known_networks[s_known_networks_count].ssid, ssid, sizeof(s_known_networks[s_known_networks_count].ssid) - 1);
        strncpy(s_known_networks[s_known_networks_count].pass, pass, sizeof(s_known_networks[s_known_networks_count].pass) - 1);
        s_known_networks[s_known_networks_count].auto_connect = true;
        s_known_networks_count++;
        return wifi_known_networks_save();
    }
    return false;
}

bool wifi_known_networks_remove(int index) {
    if (index < 0 || index >= s_known_networks_count) return false;
    for (int i = index; i < s_known_networks_count - 1; i++) {
        s_known_networks[i] = s_known_networks[i + 1];
    }
    s_known_networks_count--;
    return wifi_known_networks_save();
}

int wifi_known_networks_get_count(void) { return s_known_networks_count; }
const known_network_t* wifi_known_networks_get(int index) {
    if (index < 0 || index >= s_known_networks_count) return NULL;
    return &s_known_networks[index];
}

bool wifi_try_auto_connect(void) {
    if (s_known_networks_count == 0) return false;
    ESP_LOGI(TAG, "Scanning for known networks...");
    int count = wifi_scan_networks();
    if (count <= 0) return false;

    int best_rssi = -1000;
    int best_net_idx = -1;

    for (int i = 0; i < count; i++) {
        const wifi_scan_record_t* rec = wifi_scan_get_record(i);
        if (!rec) continue;
        for (int j = 0; j < s_known_networks_count; j++) {
            if (s_known_networks[j].auto_connect && strcmp(rec->ssid, s_known_networks[j].ssid) == 0) {
                if (rec->rssi > best_rssi) {
                    best_rssi = rec->rssi;
                    best_net_idx = j;
                }
            }
        }
    }

    if (best_net_idx >= 0) {
        ESP_LOGI(TAG, "Auto-connecting to best known network: %s (RSSI: %d)", s_known_networks[best_net_idx].ssid, best_rssi);
        wifi_connect(s_known_networks[best_net_idx].ssid, s_known_networks[best_net_idx].pass);
        return true;
    }
    return false;
}
